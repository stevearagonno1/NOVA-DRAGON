"""Module 4 — online Naive-Bayes order-flow classifier + CVD engine.

Live: true tick-level delta from @aggTrade (buyer_is_maker → aggressive sell).
Backtest: wick-split delta proxy from OHLCV. Both feed identical features.

Zero-Threshold rule: the classifier NEVER blocks a trade — it emits
(probability %, class) which is appended to telemetry only.
"""
from __future__ import annotations

import time
from collections import deque

import numpy as np

from . import config as C

_EPS = 1e-12


class NaiveBayesOrderFlow:
    """Rolling 3-class Gaussian Naive Bayes with O(1) updates.

    Per class c and feature k we keep (n_c, ΣF, ΣF²) → μ, σ on the fly.
    A 1-unit multiplicative decay per update keeps the model *rolling*
    (exponentially forgetting) — still strictly O(1) per tick.
    Classes: 0=Bull, 1=Bear, 2=Diverged.
    """

    CLASSES = ("Bull", "Bear", "Diverged")

    def __init__(self, n_features: int = 3, decay: float = 0.999):
        self.n = np.zeros(3, dtype=np.float64)
        self.s1 = np.zeros((3, n_features), dtype=np.float64)
        self.s2 = np.zeros((3, n_features), dtype=np.float64)
        self.decay = float(decay)

    @staticmethod
    def _sanitize(feats: np.ndarray) -> np.ndarray:
        return np.clip(np.nan_to_num(np.asarray(feats, dtype=np.float64),
                                     nan=0.0, posinf=10.0, neginf=-10.0),
                       -10.0, 10.0)

    def update(self, feats: np.ndarray, cls: int) -> None:
        f = self._sanitize(feats)
        d = self.decay
        if d < 1.0:                                  # rolling window stats
            self.n *= d
            self.s1 *= d
            self.s2 *= d
        self.n[cls] += 1.0
        self.s1[cls] += f
        self.s2[cls] += f * f

    def predict(self, feats: np.ndarray) -> tuple[str, float]:
        """Returns (class_name, probability_pct). Never raises on cold start."""
        total = self.n.sum()
        if total < 5.0:                              # cold start — uninformative
            return "Unclassified", 50.0
        f = self._sanitize(feats)
        prior = (self.n + 1.0) / (total + 3.0)       # Laplace smoothing
        mu = self.s1 / np.maximum(self.n, 1.0)[:, None]
        var = self.s2 / np.maximum(self.n, 1.0)[:, None] - mu * mu
        sig = np.sqrt(np.clip(var, 1e-9, None))
        logp = np.log(prior)
        z = (f[None, :] - mu) / sig
        logp = logp + (-0.5 * z * z - np.log(sig)).sum(axis=1)
        logp -= logp.max()
        post = np.exp(logp)
        post /= post.sum()
        i = int(np.argmax(post))
        return self.CLASSES[i], float(post[i] * 100.0)


class OrderFlowEngine:
    """Per-symbol CVD tracker + 50-bar z-scored feature extraction."""

    def __init__(self):
        self.cvd = 0.0
        self.bar_delta = 0.0
        self.prices: deque[float] = deque(maxlen=120)
        self.cvds: deque[float] = deque(maxlen=120)
        self.raw_hist: deque[np.ndarray] = deque(maxlen=50)
        self.last_label = 2
        # ---- whale-wake: O(1) Welford stats on per-trade quote volume ------
        self.qv_n = 0.0
        self.qv_mean = 0.0
        self.qv_m2 = 0.0
        self.whale: tuple[int, float] | None = None   # (side, monotonic_ts)

    # -------------------------------------------------- live tick ingestion
    def on_agg_trade(self, price: float, qty: float, buyer_is_maker: bool) -> None:
        """True wick-split tick delta: maker buyer ⇒ aggressive seller.
        Also maintains online quote-volume stats and flags extreme single
        prints (whales) — an O(1) Welford accumulator feeds the detector."""
        d = -qty if buyer_is_maker else qty
        self.cvd += d
        self.bar_delta += d

        qv = price * qty                                # quote volume (USD)
        self.qv_n += 1.0
        dd = qv - self.qv_mean
        self.qv_mean += dd / self.qv_n
        self.qv_m2 += dd * (qv - self.qv_mean)
        if self.qv_n > C.WHALE_MIN_SAMPLES and qv > C.WHALE_MIN_QUOTE:
            sd = (self.qv_m2 / self.qv_n) ** 0.5
            if qv > self.qv_mean + C.WHALE_SIGMA * max(sd, _EPS):
                self.whale = (1 if d > 0 else -1, time.monotonic())

    def whale_recent(self, side: int, ttl: float = C.WHALE_TTL_SEC) -> bool:
        """True if an extreme same-side whale print happened within `ttl`."""
        return bool(self.whale and self.whale[0] == side
                    and time.monotonic() - self.whale[1] <= ttl)

    # ------------------------------------------------- bar close (both modes)
    def on_bar_close(self, close: float, bar_delta: float | None = None) -> None:
        """`bar_delta=None` → live (delta accumulated from ticks); otherwise the
        caller supplies the wick-split proxy (backtest)."""
        d = self.bar_delta if bar_delta is None else float(bar_delta)
        if bar_delta is not None:
            self.cvd += d
        self.prices.append(float(close))
        self.cvds.append(self.cvd)
        prev_close = self.prices[-2] if len(self.prices) > 1 else close
        self._last_price_chg = close - prev_close
        self._last_delta = d
        self.bar_delta = 0.0

    # ------------------------------------------------------------ features
    def features(self) -> np.ndarray | None:
        """Returns z-scored [cvdRoc, priceRoc − cvdRoc, slopeR] or None.

        Robustness notes vs. the naive version:
        * CVD is a cumulative signed sum that crosses/hovers near 0 — a fixed
          |cvd[-k]| denominator explodes. CVD features are therefore scaled
          by mean |ΔCVD| over the window (volume units, strictly stationary).
        * slopeR scale uses max(σ, mean|Δ|) so flat CVD can't divide by ~0.
        * Returns None (no learning) until the 50-bar z-window is warm —
          pre-warm raw values must never contaminate the class statistics.
        """
        if len(self.prices) < 25:
            return None
        p = np.asarray(self.prices)
        v = np.asarray(self.cvds)

        dv_scale = max(float(np.abs(np.diff(v[-21:])).mean()), _EPS)
        cvd_roc = float((v[-1] - v[-10]) / (9.0 * dv_scale))
        price_roc = float((p[-1] - p[-10]) / (abs(p[-10]) + _EPS))
        divergence = price_roc - cvd_roc
        x = np.arange(20, dtype=float)
        slope = np.polyfit(x, v[-20:], 1)[0]
        slope_scale = max(float(v[-20:].std()), dv_scale, _EPS)
        slope_r = float(slope / slope_scale)
        raw = np.clip(np.array([cvd_roc, divergence, slope_r]), -50.0, 50.0)

        self.raw_hist.append(raw)
        if len(self.raw_hist) < 10:
            return None                                # warm-up: no learning yet
        H = np.asarray(self.raw_hist)
        mu, sd = H.mean(axis=0), H.std(axis=0) + _EPS
        return (raw - mu) / sd

    def self_label(self) -> int:
        """Self-supervision: bar outcome vs. order-flow agreement."""
        pc = getattr(self, "_last_price_chg", 0.0)
        d = getattr(self, "_last_delta", 0.0)
        if pc == 0.0 and d == 0.0:
            return self.last_label
        agree = np.sign(pc) == np.sign(d) and pc != 0.0
        self.last_label = (0 if pc > 0 else 1) if agree else 2
        return self.last_label
