"""Module 7 — dynamic position sizing + the Symbol Governor."""
from __future__ import annotations

from . import config as C


class PositionSizer:
    """Confidence- and volatility-driven sizing; abandons fixed lots."""

    @staticmethod
    def notional(confidence: float, vr: float) -> float:
        """Returns USD notional; 0.0 ⇒ trade rejected (<50 confidence)."""
        if confidence < C.CONF_REJECT:
            return 0.0
        base = 0.5 * (C.BASE_NOTIONAL_MIN + C.BASE_NOTIONAL_MAX)   # $11 buffer
        size = base / 2.0 if confidence <= C.CONF_HALF else base   # 50–70 half, >70 full
        if vr and vr > C.VR_SHOCK:                                  # extreme volatility guard
            size /= 2.0
        return size


class SymbolGovernor:
    """Per-symbol consecutive-loss cooldown: 3 net losses → 600 s timeout.
    Only the offending symbol is parked; the rest of the universe keeps trading."""

    def __init__(self):
        self._streak: dict[str, int] = {}
        self._cooldown_until: dict[str, float] = {}

    def allowed(self, symbol: str, now: float) -> bool:
        until = self._cooldown_until.get(symbol)
        if until is not None and now < until:
            return False
        if until is not None and now >= until:
            self._cooldown_until.pop(symbol, None)
            self._streak[symbol] = 0
        return True

    def record(self, symbol: str, net_pnl: float, now: float) -> bool:
        """Returns True if this record triggered a fresh cooldown."""
        if net_pnl < 0:
            streak = self._streak.get(symbol, 0) + 1
            self._streak[symbol] = streak
            if streak >= C.GOVERNOR_MAX_LOSSES:
                self._cooldown_until[symbol] = now + C.GOVERNOR_COOLDOWN_SEC
                self._streak[symbol] = 0
                return True
        else:
            self._streak[symbol] = 0
        return False

    def reset(self) -> None:
        self._streak.clear()
        self._cooldown_until.clear()
