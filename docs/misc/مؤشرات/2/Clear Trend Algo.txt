//@version=5
indicator('Clear Trend Algo', shorttitle = 'Clear Trend v3.2', overlay = true)

// UI Option: Auto Trend Detector (Only for UI, No Functional Impact)
autoTrendDetector = input.bool(true, title='Auto Trend Detector')

// Core Trend Inputs
price       = input(close, 'Source')
length      = input.int(200, 'Length', minval=1)
smooth      = input.int(3, 'Extra Smooth [1 = None]', minval=1)

// New Input for Double Trend Line Gap (Percentage of ATR)
trendGapMult = input.float(0.1, "Trend Line Gap Multiplier (ATR %)", minval=0.01, maxval=1.0, step=0.01, group="Trend Line Settings")
atrForGap    = ta.atr(14)  // Using ATR(14) for dynamic gap scaling
trendGap     = atrForGap * trendGapMult  // Gap is now proportional to ATR

mult        = input.float(0.3, minval=0.05, maxval=3, step=0.05,    
  title='Width',  inline = 'SD Channel', group='Volatility Filter (SD Channel)')
sd_len      = input.int(5, minval=1,                                  
  title='Length', inline = 'SD Channel', group='Volatility Filter (SD Channel)')

baseline    = ta.wma(price, sd_len)
dev         = mult * ta.stdev(price, sd_len)
upper       = baseline + dev
lower       = baseline - dev

cprice      = price > upper ? upper : price < lower ? lower : price
REMA        = ta.wma(ta.wma(cprice, length), smooth)

// Double Trend Lines
REMA_upper  = REMA + trendGap
REMA_lower  = REMA - trendGap

c_up        = color.new(#33ff00, 0)
c_dn        = color.new(#ff1111, 0)
glow_up     = color.new(#33ff00, 60)
glow_dn     = color.new(#ff1111, 60)

REMA_up     = REMA > REMA[1]

// Plot Double Clear Trend Lines with Glow Effect
plot(REMA_upper, title='Clear Trend Upper', color=REMA_up ? c_up : c_dn, linewidth=4)
plot(REMA_lower, title='Clear Trend Lower', color=REMA_up ? c_up : c_dn, linewidth=4)
plot(REMA_upper, title='Glow Effect Upper', color=REMA_up ? glow_up : glow_dn, linewidth=7, transp=50)
plot(REMA_lower, title='Glow Effect Lower', color=REMA_up ? glow_up : glow_dn, linewidth=7, transp=50)

// Buy/Sell Signals
newBuySignal = REMA_up and not REMA_up[1] and barstate.isconfirmed
plotshape(newBuySignal, title="BUY", style=shape.labelup, location=location.belowbar, color=#00e600, text="🚀BUY", textcolor=color.white, size=size.small)

newSellSignal = not REMA_up and REMA_up[1] and barstate.isconfirmed
plotshape(newSellSignal, title="SELL", style=shape.labeldown, location=location.abovebar, color=#fc0404, text="🔥SELL", textcolor=color.white, size=size.small)

// Stop-Loss Logic with Manual & Auto Options
atrRisk     = input.int(3, "Initial Risk % (Manual)", minval=1, group="Stop Loss Settings")
atrLen      = input.int(14, "ATR Length (Manual)", minval=1, group="Stop Loss Settings")
manualOffset = input.float(0.0, "Manual SL Offset (Points)", step=0.1, group="Stop Loss Settings")
useAutoSL   = input.bool(false, "Use Auto Stop-Loss", group="Stop Loss Settings")
autoSLPips  = input.float(30, "Auto SL Pips (10-30)", minval=10, maxval=30, step=5, group="Stop Loss Settings")

// New UI Inputs for Colors and Thickness
slLineColor   = input.color(color.red, "Stop-Loss Line Color", group="Line Settings")
buyLineColor  = input.color(color.green, "Buy Signal Line Color", group="Line Settings")
sellLineColor = input.color(color.orange, "Sell Signal Line Color", group="Line Settings")

slLineWidth   = input.int(2, "Stop-Loss Line Thickness", minval=1, maxval=5, group="Line Settings")
buyLineWidth  = input.int(2, "Buy Signal Line Thickness", minval=1, maxval=5, group="Line Settings")
sellLineWidth = input.int(2, "Sell Signal Line Thickness", minval=1, maxval=5, group="Line Settings")

// Take Profit Settings (Adjusted for Proper Pip Scaling)
takeProfitPips = input.float(80.0, title="Take Profit (in pips)", minval=1.0, step=1.0, group="Take Profit Settings")
tickSize = syminfo.mintick > 0 ? syminfo.mintick : 0.01 // Prevent zero tick size issues
takeProfitAmount = takeProfitPips * tickSize * 10  // Adjusted for proper pip scaling (e.g., 80 pips = 0.0080 for forex)

// Track Entry Prices and Take Profit Signals
var float entryPriceBuy = na
var float entryPriceSell = na
var bool takeProfitSignalBuy = false
var bool takeProfitSignalSell = false

var int trigger = 0
trigger := newBuySignal ? 1 : newSellSignal ? 0 : nz(trigger[1])

// Calculate Initial SL (Manual Mode)
atrBand = ta.atr(atrLen) * atrRisk
manualSL = trigger == 1 ? low - atrBand : high + atrBand
manualAdjustedSL = trigger == 1 ? manualSL + manualOffset : manualSL - manualOffset

// Calculate Auto SL (in points/pips)
pipValue = syminfo.mintick
autoSLPoints = autoSLPips * pipValue * 10
autoSL = trigger == 1 ? low - autoSLPoints : high + autoSLPoints

// Choose SL Based on Mode
finalSL = useAutoSL ? autoSL : manualAdjustedSL

// Variables to Store Lines and Last Signal Levels
var line sl_line = na
var line buy_line = na
var line sell_line = na
var float lastBuyPrice = na
var float lastSellPrice = na
var float lastSL = na
var int lastSignalBar = na

// Buy Signal Logic with Take Profit
if newBuySignal
    lastBuyPrice := close
    lastSellPrice := na
    lastSL := finalSL
    lastSignalBar := bar_index
    entryPriceBuy := close // Store entry price for Take Profit
    takeProfitSignalBuy := false // Reset Take Profit signal
    if not na(sl_line)
        line.delete(sl_line)
    sl_line := line.new(lastSignalBar, finalSL, bar_index, finalSL, color=slLineColor, style=line.style_dashed, width=slLineWidth)
    if not na(buy_line)
        line.delete(buy_line)
    buy_line := line.new(lastSignalBar, close, bar_index, close, color=buyLineColor, style=line.style_dashed, width=buyLineWidth)
    if not na(sell_line)
        line.delete(sell_line)
    sell_line := na
    label.new(bar_index, finalSL, "SL: " + str.tostring(finalSL, "#.##"), xloc.bar_index, yloc.price, color=slLineColor, textcolor=color.white, style=label.style_label_up, size=size.small)

// Sell Signal Logic with Take Profit
if newSellSignal
    lastSellPrice := close
    lastBuyPrice := na
    lastSL := finalSL
    lastSignalBar := bar_index
    entryPriceSell := close // Store entry price for Take Profit
    takeProfitSignalSell := false // Reset Take Profit signal
    if not na(sl_line)
        line.delete(sl_line)
    sl_line := line.new(lastSignalBar, finalSL, bar_index, finalSL, color=slLineColor, style=line.style_dashed, width=slLineWidth)
    if not na(sell_line)
        line.delete(sell_line)
    sell_line := line.new(lastSignalBar, close, bar_index, close, color=sellLineColor, style=line.style_dashed, width=sellLineWidth)
    if not na(buy_line)
        line.delete(buy_line)
    buy_line := na
    label.new(bar_index, finalSL, "SL: " + str.tostring(finalSL, "#.##"), xloc.bar_index, yloc.price, color=slLineColor, textcolor=color.white, style=label.style_label_down, size=size.small)

// Take Profit Logic for Buy (Trigger Only Once)
if (not na(entryPriceBuy) and close >= entryPriceBuy + takeProfitAmount and not takeProfitSignalBuy[1])
    takeProfitSignalBuy := true
    entryPriceBuy := na // Reset entry price after take profit

// Take Profit Logic for Sell (Trigger Only Once)
if (not na(entryPriceSell) and close <= entryPriceSell - takeProfitAmount and not takeProfitSignalSell[1])
    takeProfitSignalSell := true
    entryPriceSell := na // Reset entry price after take profit

// Update Existing Lines Until New Signal
if not (newBuySignal or newSellSignal) and not na(lastSignalBar)
    if not na(buy_line) and not na(lastBuyPrice)
        line.set_x2(buy_line, bar_index)
    if not na(sell_line) and not na(lastSellPrice)
        line.set_x2(sell_line, bar_index)
    if not na(sl_line) and not na(lastSL)
        line.set_x2(sl_line, bar_index)

// Plot Take Profit Signals (Only Once)
plotshape(takeProfitSignalBuy and not takeProfitSignalBuy[1], title="Take Profit Buy", style=shape.labeldown, location=location.abovebar, color=color.rgb(235, 252, 4), text="Take Profit", textcolor=#050505, size=size.small)
plotshape(takeProfitSignalSell and not takeProfitSignalSell[1], title="Take Profit Sell", style=shape.labelup, location=location.belowbar, color=color.rgb(2, 18, 245), text="Take Profit", textcolor=color.white, size=size.small)

// Alerts
SwingDn     = REMA_up[1] and not(REMA_up) and barstate.isconfirmed
SwingUp     = REMA_up    and not(REMA_up[1]) and barstate.isconfirmed

alertcondition(SwingUp, "Clear Trend Swing Up", "Clear Trend Swing Up Detected!")
alertcondition(SwingDn, "Clear Trend Swing Down", "Clear Trend Swing Down Detected!")
alertcondition(SwingUp or SwingDn, "Clear Trend Swing", "Clear Trend Swing Detected!")
