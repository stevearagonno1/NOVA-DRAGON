//@version=6
indicator("Liquidity Sweep Auto TP (RR 1:2)", overlay=true, max_lines_count=500, max_labels_count=500)

//=============================
// Inputs
//=============================
pivotLen = input.int(5, "Swing Length", minval=2)
rr = input.float(2.0, "Risk Reward", minval=1.0)

//=============================
// Swing High / Low
//=============================
ph = ta.pivothigh(high, pivotLen, pivotLen)
pl = ta.pivotlow(low, pivotLen, pivotLen)

var float buyLiquidity = na
var float sellLiquidity = na

if not na(ph)
    buyLiquidity := ph

if not na(pl)
    sellLiquidity := pl

//=============================
// Liquidity Rays
//=============================
var line buyLine = na
var line sellLine = na

if not na(ph)
    line.delete(buyLine)
    buyLine := line.new(
         bar_index-pivotLen, ph,
         bar_index, ph,
         extend=extend.right,
         color=color.red,
         width=2)

if not na(pl)
    line.delete(sellLine)
    sellLine := line.new(
         bar_index-pivotLen, pl,
         bar_index, pl,
         extend=extend.right,
         color=color.lime,
         width=2)

//=============================
// Sweep Detection
//=============================
bullSweep = not na(sellLiquidity) and low < sellLiquidity and close > sellLiquidity
bearSweep = not na(buyLiquidity) and high > buyLiquidity and close < buyLiquidity

//=============================
// === Dashboard with Telegram Link ===
var table myTable = table.new(position.bottom_right, 1, 1, border_width=1, frame_color=color.black, bgcolor=color.white)

// Add Telegram Message to Dashboard
table.cell(myTable, 0, 0, "Join Telegram @mrexpert_ai", bgcolor=color.blue, text_color=color.white, text_size=size.normal)
// Trade Lines
//=============================
var line entryLine = na
var line slLine = na
var line tpLine = na

var label entryLbl = na
var label slLbl = na
var label tpLbl = na

if bullSweep or bearSweep

    line.delete(entryLine)
    line.delete(slLine)
    line.delete(tpLine)

    label.delete(entryLbl)
    label.delete(slLbl)
    label.delete(tpLbl)

    float entry = close
    float sl = na
    float tp = na

    if bullSweep
        sl := low
        risk = entry - sl
        tp := entry + risk * rr

    if bearSweep
        sl := high
        risk = sl - entry
        tp := entry - risk * rr

    entryLine := line.new(
         bar_index, entry,
         bar_index+100, entry,
         extend=extend.right,
         color=color.blue,
         width=2)

    slLine := line.new(
         bar_index, sl,
         bar_index+100, sl,
         extend=extend.right,
         color=color.red,
         width=2)

    tpLine := line.new(
         bar_index, tp,
         bar_index+100, tp,
         extend=extend.right,
         color=color.green,
         width=2)

    entryLbl := label.new(bar_index, entry,
         "ENTRY",
         style=label.style_label_left,
         color=color.blue,
         textcolor=color.white)

    slLbl := label.new(bar_index, sl,
         "SL",
         style=label.style_label_left,
         color=color.red,
         textcolor=color.white)

    tpLbl := label.new(bar_index, tp,
         "TP 1:2",
         style=label.style_label_left,
         color=color.green,
         textcolor=color.white)

//=============================
// Alerts
//=============================
alertcondition(bullSweep, title="Bullish Liquidity Sweep", message="Bullish Liquidity Sweep")
alertcondition(bearSweep, title="Bearish Liquidity Sweep", message="Bearish Liquidity Sweep")
