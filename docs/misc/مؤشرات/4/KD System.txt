fib_update() =>
    var settings        = zigzag.Settings.new(fib_devMult, fib_depth, color(na), false, false, false, false, "Absolute", true)
    var zigzag.ZigZag zz = zigzag.newInstance(settings)
    var zigzag.Pivot lP  = na
    var float sp         = na
    var float ht         = na
    settings.devThreshold := ta.atr(10) / close * 100 * fib_devMult
    if zz.update()
        lP := zz.lastPivot()
        if not na(lP)
            var line ll = na
            if switch_fibonacci
                if na(ll)
                    ll := line.new(lP.start, lP.end, xloc=xloc.bar_time, color=color.gray, width=1, style=line.style_dashed)
                else
                    line.set_first_point(ll, lP.start)
                    line.set_second_point(ll, lP.end)
            sp := fib_reverse ? lP.start.price : lP.end.price
            ep  = fib_reverse ? lP.end.price   : lP.start.price
            ht := (sp > ep ? -1 : 1) * math.abs(sp - ep)
    [lP, sp, ht]

[fibLastP, fibSP, fibHT] = fib_update()

    var id = label.new(x=x, y=price, xloc=xloc.bar_time, text=lStr, textcolor=txtColor, style=lStyle, textalign=aln, color=#00000000)
    label.set_xy(id, x, price)
    label.set_text(id, lStr)
    label.set_textcolor(id, txtColor)

fib_cross(series float sr, series float r) =>
    (r > sr and r < sr[1]) or (r < sr and r > sr[1])

fib_processLevel(bool show, float value, color col, line prevId) =>
    r = fibSP + fibHT * value
    _crossed = fib_cross(close, r)
    if show and not na(fibLastP)
        lId = fib_line(r, col)
        fib_lbl(r, fib_lblTxt(value, r), col)
        if _crossed
            alert("Autofib: " + syminfo.ticker + " crossing " + str.tostring(value))
        if not na(prevId)
            linefill.new(lId, prevId, color=color.new(col, fib_bgTrans))
        lId
    else
        prevId

    var float rngfilt = na
    if x > nz(rngfilt[1])
        rngfilt := x - r < nz(rngfilt[1]) ? nz(rngfilt[1]) : x - r
    else
        rngfilt := x + r > nz(rngfilt[1]) ? nz(rngfilt[1]) : x + r
    rngfilt

HMA_f(s, l)  => ta.wma(2 * ta.wma(s, l / 2) - ta.wma(s, l), math.round(math.sqrt(l)))
EHMA_f(s, l) => ta.ema(2 * ta.ema(s, l / 2) - ta.ema(s, l), math.round(math.sqrt(l)))
THMA_f(s, l) => ta.wma(ta.wma(s, l / 3) * 3 - ta.wma(s, l / 2) - ta.wma(s, l), l)

HullCalc(m, s, l) =>
    switch m
        "Hma"  => HMA_f(s, l)
        "Ehma" => EHMA_f(s, l)
        "Thma" => THMA_f(s, l / 2)

percWidth(l, p) => (ta.highest(l) - ta.lowest(l)) * p / 100

wavetrend(ws, chl, avl) =>
    esa  = ta.ema(ws, chl)
    d    = ta.ema(math.abs(ws - esa), chl)
    ci   = (ws - esa) / (0.015 * d)
    w1   = ta.ema(ci, avl)
    w2   = ta.sma(w1, 3)
    [w1, w2]

f_topFrac(s) => s[4] < s[2] and s[3] < s[2] and s[2] > s[1] and s[2] > s[0]
f_botFrac(s) => s[4] > s[2] and s[3] > s[2] and s[2] < s[1] and s[2] < s[0]
f_frac(s)    => f_topFrac(s) ? 1 : f_botFrac(s) ? -1 : 0

f_findDivs(fd, topLim, botLim) =>
    fTopBool = f_frac(fd) > 0 and fd[2] >= topLim
    fBotBool = f_frac(fd) < 0 and fd[2] <= botLim
    fTop     = fTopBool ? fd[2] : na
    fBot     = fBotBool ? fd[2] : na
    hPrev    = ta.valuewhen(fTopBool, fd[2],   0)[2]
    hPrice   = ta.valuewhen(fTopBool, high[2], 0)[2]
    lPrev    = ta.valuewhen(fBotBool, fd[2],   0)[2]
    lPrice   = ta.valuewhen(fBotBool, low[2],  0)[2]
    bear     = fTopBool and high[2] > hPrice and fd[2] < hPrev
    bull     = fBotBool and low[2]  < lPrice and fd[2] > lPrev
    [bear, bull]

var array<int>      cvdSupLefts = array.new<int>()

cvdPolyUpdate(bl, zT, zB, oldLine, lCol, fCol, isDem) =>
    ln  = math.max(0, math.min(int(bar_index - bl), math.min(2900, bar_index - 1)))
    cd  = 0.0
    mxD = 0.0
    for k = ln to 0
        d = close[k] > open[k] ? volume[k] : close[k] < open[k] ? -volume[k] : 0.0
        cd  += d
        mxD := math.max(mxD, math.abs(cd))
    pts  = array.new<chart.point>()
    bxH  = zT - zB
    midY = (zT + zB) / 2
    array.push(pts, chart.point.from_index(bar_index - ln, midY))
    cd := 0.0
    for k = ln to 0
        d = close[k] > open[k] ? volume[k] : close[k] < open[k] ? -volume[k] : 0.0
        cd += d
        rawY = isDem ? zB + (mxD > 0 ? (cd / mxD) * bxH : 0) : zT + (mxD > 0 ? (cd / mxD) * bxH : 0)
        array.push(pts, chart.point.from_index(bar_index - k, math.max(zB, math.min(zT, rawY))))
    array.push(pts, chart.point.from_index(bar_index, midY))
    if not na(oldLine)
        polyline.delete(oldLine)
    [polyline.new(pts, closed=true, line_color=lCol, fill_color=fCol, line_width=1), cd]

    if not ov

        array.push(cvdDemBoxes, b)

        array.push(cvdSupBoxes, b)
        array.push(cvdSupTops,  zT)
        array.push(cvdSupBots,  zB)
        array.push(cvdSupLefts, bl)
        array.push(cvdSupLines, polyline.new(na))
    if array.size(cvdSupBoxes) > maxBoxesCVD
        cvdTrimSup()

// Demand mitigation & update
dsz = array.size(cvdDemBoxes)
if dsz > 0
    i = dsz - 1
    while i >= 0
        if i < array.size(cvdDemBoxes)
            zB = array.get(cvdDemBots, i), zT = array.get(cvdDemTops,  i)
            bl = array.get(cvdDemLefts,i), b  = array.get(cvdDemBoxes, i)

        i -= 1

// Supply mitigation & update

        i -= 1

if showCVDSig and isDemandBrk
