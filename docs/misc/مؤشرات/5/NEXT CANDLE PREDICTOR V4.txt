// ============================================================================
// CUSTOM SUPERTREND IMPLEMENTATION
// ============================================================================

trendAtr = ta.atr(stPeriod)
hl2Value = (high + low) / 2
upperBandRaw = hl2Value + (stFactor * trendAtr)
lowerBandRaw = hl2Value - (stFactor * trendAtr)

var float upperBand = na
var float lowerBand = na
var int trendDirection = 1

prevUpperBand = nz(upperBand[1], upperBandRaw)
prevLowerBand = nz(lowerBand[1], lowerBandRaw)
prevDirection = nz(trendDirection[1], 1)

lowerBand := close[1] > prevLowerBand ? math.max(lowerBandRaw, prevLowerBand) : lowerBandRaw
upperBand := close[1] < prevUpperBand ? math.min(upperBandRaw, prevUpperBand) : upperBandRaw

if prevDirection == -1
    trendDirection := close < lowerBand ? 1 : -1
else
    trendDirection := close > upperBand ? -1 : 1

trendLine = trendDirection == -1 ? lowerBand : upperBand

isUptrend = trendDirection == -1
isDowntrend = trendDirection == 1
trendChanged = trendDirection != trendDirection[1]

// ============================================================================
// VOLUME DELTA - LEADING INDICATOR
// ============================================================================

candleRange = high - low
buyVolume = candleRange > 0 ? volume * (close - low) / candleRange : volume * 0.5
sellVolume = candleRange > 0 ? volume * (high - close) / candleRange : volume * 0.5
volumeDelta = buyVolume - sellVolume

deltaEma = ta.ema(volumeDelta, 10)
deltaMomentum = volumeDelta > deltaEma

deltaBullish = volumeDelta > 0
deltaBearish = volumeDelta < 0

// ============================================================================
// STANDARD TECHNICAL INDICATORS
// ============================================================================

atr = ta.atr(atrLength)

ema8 = ta.ema(close, 8)
ema21 = ta.ema(close, 21)
ema50 = ta.ema(close, 50)

rsi = ta.rsi(close, 14)
rsiAbove50 = rsi > 50
rsiBelow50 = rsi < 50

[macdLine, signalLine, macdHist] = ta.macd(close, 12, 26, 9)

stochK = ta.stoch(close, high, low, 14)
stochD = ta.sma(stochK, 3)
stochAbove50 = stochK > 50
stochBelow50 = stochK < 50

volSma = ta.sma(volume, 20)
volRatio = nz(volume / volSma, 1.0)
avgBody = ta.sma(math.abs(close - open), 5)

adxLen = 14
[diPlus, diMinus, adxValue] = ta.dmi(adxLen, adxLen)

bbMid = ta.sma(close, 20)
bbStd = ta.stdev(close, 20)

// ============================================================================

macdScoreShort = macdLine < signalLine and macdHist < 0 ? 100 : macdLine < signalLine ? 70 : macdHist < 0 ? 50 : 20

rsiScoreLong = rsi < 30 ? 100 : rsi < 40 ? 85 : rsi < 50 ? 70 : rsi < 60 ? 50 : 25
rsiScoreShort = rsi > 70 ? 100 : rsi > 60 ? 85 : rsi > 50 ? 70 : rsi > 40 ? 50 : 25

stochScoreLong = stochK > stochD and stochK < 20 ? 100 : stochK > stochD and stochK < 50 ? 85 : stochK > stochD ? 65 : 25
stochScoreShort = stochK < stochD and stochK > 80 ? 100 : stochK < stochD and stochK > 50 ? 85 : stochK < stochD ? 65 : 25

volScore = volRatio > 2.0 ? 100 : volRatio > 1.5 ? 80 : volRatio > 1.0 ? 60 : volRatio > 0.8 ? 45 : 25

deltaScoreLong = volumeDelta > 0 and deltaMomentum ? 100 : volumeDelta > 0 ? 75 : volumeDelta > -math.abs(deltaEma) ? 40 : 20
deltaScoreShort = volumeDelta < 0 and not deltaMomentum ? 100 : volumeDelta < 0 ? 75 : volumeDelta < math.abs(deltaEma) ? 40 : 20

adxScore = adxValue > 35 ? 100 : adxValue > 30 ? 85 : adxValue > 25 ? 70 : adxValue > 20 ? 50 : 30

trendScoreLong = isUptrend and ema8 > ema21 and ema21 > ema50 ? 100 : isUptrend and ema8 > ema21 ? 80 : isUptrend ? 60 : 0
trendScoreShort = isDowntrend and ema8 < ema21 and ema21 < ema50 ? 100 : isDowntrend and ema8 < ema21 ? 80 : isDowntrend ? 60 : 0

longScoreRaw = (trendScoreLong * 0.23) + (macdScoreLong * 0.18) + (deltaScoreLong * 0.15) + (rsiScoreLong * 0.12) + (stochScoreLong * 0.12) + (adxScore * 0.10) + (volScore * 0.10)
shortScoreRaw = (trendScoreShort * 0.23) + (macdScoreShort * 0.18) + (deltaScoreShort * 0.15) + (rsiScoreShort * 0.12) + (stochScoreShort * 0.12) + (adxScore * 0.10) + (volScore * 0.10)

shortScore = math.round(math.min(100, math.max(0, shortScoreRaw * volMultiplier)))

totalRaw = longScore + shortScore
longPct = totalRaw > 0 ? math.round(longScore / totalRaw * 100) : 50

getAsciiBar(score) =>
    filled = math.round(score / 10)
    filled := math.max(0, math.min(10, filled))
    "[" + str.repeat("█", filled) + str.repeat("░", 10 - filled) + "]"

getMacdText(isLongParam) =>
    if isLongParam
        macdLine > signalLine and macdHist > 0 ? "Bullish ✓" : macdLine > signalLine ? "Bullish" : "Bearish"
    else
        macdLine < signalLine and macdHist < 0 ? "Bearish ✓" : macdLine < signalLine ? "Bearish" : "Bullish"

getRsiText() =>
    arrow = rsi > 50 ? "↑" : "↓"
    "(" + str.tostring(math.round(rsi)) + ") " + arrow

getStochText(isLongParam) =>
    arrow = stochK > 50 ? "↑" : "↓"
    if isLongParam
        stochK < 20 ? "Oversold " + arrow : stochK > stochD ? "Bullish " + arrow : "Bearish " + arrow
    else
        stochK > 80 ? "Overbought " + arrow : stochK < stochD ? "Bearish " + arrow : "Bullish " + arrow

getVolText() =>
    str.tostring(math.round(volRatio, 1)) + "x"

getDeltaText(isLongParam) =>
    if isLongParam
        volumeDelta > 0 and deltaMomentum ? "Strong Buy ⚡" : volumeDelta > 0 ? "Buy ↑" : "Sell ↓"
    else
        volumeDelta < 0 and not deltaMomentum ? "Strong Sell ⚡" : volumeDelta < 0 ? "Sell ↓" : "Buy ↑"

getTrendText(isLongParam) =>
    if isLongParam
        isUptrend and ema8 > ema21 ? "Strong Up ✓" : isUptrend ? "Up" : "Down"
    else
        isDowntrend and ema8 < ema21 ? "Strong Down ✓" : isDowntrend ? "Down" : "Up"

getAdxText() =>
    strength = adxValue > 30 ? "Strong" : adxValue > 25 ? "OK" : "Weak"
    str.tostring(math.round(adxValue)) + " " + strength
