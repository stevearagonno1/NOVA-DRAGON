// ============================================================================
// Anchored VWAP Trade Planner
// ============================================================================
//
// WHAT IT DOES:
//   Calculates an Anchored VWAP (Session / Weekly / Manual / Auto-detect pivot)
//   and projects an actionable Trade Plan (Entry / SL / TP1 / TP2 / TP3 option)
//   using Fibonacci extension multipliers of the close-to-AVWAP distance d.
//
// HOW IT WORKS (W-5 formulae, d = close - AVWAP, sign auto-handles Long/Short):
//   TP1 = AVWAP + d * 2.618    (Fib)
//   TP2 = AVWAP + d * 4.236    (Fib)
//   TP3 = AVWAP + d * 5.236    (Fib, option)
//   SL  = AVWAP + d * 0.382 + bufSign * ATR(14) * 0.25
//         (SL sits between AVWAP and close at Fib 0.618 retrace from close
//          toward AVWAP, with a small ATR safety buffer pushed away from Entry)
//   Real R:R (close-based, ATR buffer ignored):
//     RR1 = 1.618 / 0.618 = 2.618
//     RR2 = 3.236 / 0.618 = 5.236
//     RR3 = 4.236 / 0.618 = 6.854    (all Fibonacci)
//
// USAGE:
//   1. Pick an anchor type (Session is the default)
//   2. Wait for a signal (|Z-score| >= Min |Z|, with cooldown)
//   3. Trade Plan snapshot is fixed at the signal bar; lines do not redraw
//   4. Dashboard reports state (INIT / ENTERED / TP1_HIT / TP2_HIT / TP3_HIT
//      / EXITED) and recommends partial-close action
//
// NON-REPAINTING:
//   Signal snapshot (Entry/SL/TP) is captured on the signal bar close and
//   stored in var-floats; subsequent bars only update state machine and
//   redraw existing snapshot levels. This is the expected behaviour of a
//   real-time trade-planning tool.
//
// ============================================================================

// f_formatPrice: thousand separator + decimal-aware price formatter.
// - Decimal digits derived from syminfo.mintick (min 2 digits).
//   BTCUSD mintick=1   -> 2 dec ("81,068.24")
//   EURUSD mintick=1e-5 -> 5 dec ("1.02340")
//   USDJPY mintick=1e-3 -> 3 dec ("155.234")
// - Thousand separator inserted manually into integer part (Pine v6 format
//   spec does not officially support "#,##0.00"; safer to build by hand).
f_formatPrice(val) =>
    string out = ""
    if na(val)
        out := "NaN"
    else
        int nDec = syminfo.mintick < 0.01 ? math.max(2, math.round(math.log10(1.0 / syminfo.mintick))) : 2
        float absVal = math.abs(val)
        float scale  = math.pow(10.0, nDec)
        float scaled = math.round(absVal * scale)
        float intPartF = math.floor(scaled / scale)
        float decPartF = scaled - intPartF * scale
        // Integer part as raw digits (no decimal point thanks to format "#").
        string intRaw = str.tostring(intPartF, "#")
        int intLen = str.length(intRaw)
        string intStr = ""
        for i = 0 to intLen - 1
            int idxFromRight = intLen - 1 - i
            string ch = str.substring(intRaw, idxFromRight, idxFromRight + 1)
            string sep = (i > 0 and i % 3 == 0) ? "," : ""
            intStr := ch + sep + intStr
        // Decimal part zero-padded to nDec digits.
        string decRaw = str.tostring(decPartF, "#")
        int decLen = str.length(decRaw)
        string decStr = decLen < nDec ? str.repeat("0", nDec - decLen) + decRaw : decRaw
        string sign = val < 0 ? "-" : ""
        out := sign + intStr + "." + decStr
    out

f_anchorLetter(mode) =>

        "Manual"      => "M"
        "Auto-detect" => "A"
        => "S"

// ============================================================================
// Anchor Detection (4 modes)
// ============================================================================

isSessionStart = ta.change(time("D")) != 0
isWeeklyStart  = ta.change(time("W")) != 0
isManualStart  = (time >= i_manualTime) and (time[1] < i_manualTime)

phVal = ta.pivothigh(high, i_pivotLen, i_pivotLen)
plVal = ta.pivotlow(low,  i_pivotLen, i_pivotLen)
isAutoStart = (not na(phVal)) or (not na(plVal))

anchorCond = switch i_anchorMode
    "Session"     => isSessionStart
    "Weekly"      => isWeeklyStart
    "Manual"      => isManualStart
    "Auto-detect" => isAutoStart
    => false

// Anchor bar tracking (for pin)
var int   anchorBarIdx = na
var float anchorPrice  = na
if anchorCond
    anchorBarIdx := bar_index
    anchorPrice  := i_vwapSrc

// ============================================================================
// AVWAP Calculation
// ============================================================================

[vwapVal, sigmaUpper, _sigmaLower] = ta.vwap(i_vwapSrc, anchorCond, 1.0)
sigma = (not na(sigmaUpper) and not na(vwapVal)) ? sigmaUpper - vwapVal : na

// ============================================================================
// Z-score (rolling stdev normalization of (close - AVWAP))
// ============================================================================

diff    = (not na(vwapVal)) ? close - vwapVal : na
diffStd = ta.stdev(diff, i_zStdLen)
z       = (not na(diffStd) and diffStd > 0 and not na(vwapVal)) ? diff / diffStd : 0.0

// ============================================================================
// Signal Logic (raw + cooldown)
// ============================================================================

isLongRaw  = (not na(vwapVal)) and (close > vwapVal) and (z >= i_minAbsZ)
isShortRaw = (not na(vwapVal)) and (close < vwapVal) and (z <= -i_minAbsZ)

var int lastSigBar = na
cooldownOK = na(lastSigBar) or (bar_index - lastSigBar) >= i_cooldown

// Full non-repaint guard: trigger evaluated ONLY on confirmed bar close.
// During the forming (real-time) bar barstate.isconfirmed is false, so no
// signal fires. Trigger fires the instant the bar closes, snapshot is then
// frozen in var-floats, and from the next bar onward only redraw/state
// progression occurs. No intra-bar repaint possible.
longTrigger  = isLongRaw  and cooldownOK and barstate.isconfirmed
shortTrigger = isShortRaw and cooldownOK and barstate.isconfirmed

// ============================================================================
// State Machine + Snapshot (var-floats persist across bars)
// ============================================================================

var int    sigBarIdx   = na
var int    sigDir      = 0
var float  sigEntry    = na
var float  sigAVWAP    = na
var float  sigTP1      = na
var float  sigTP2      = na
var float  sigTP3      = na
var float  sigSL       = na
var string tradeState  = "INIT"
var bool   tp1Hit      = false
var bool   tp2Hit      = false
var bool   tp3Hit      = false
var bool   slHit       = false
var float  trailLevel  = na

atrVal14 = ta.atr(14)

// Snapshot fix on signal bar (W-5 unified formulae, sign auto-handles Long/Short)
if longTrigger or shortTrigger
    sigBarIdx := bar_index
    sigDir    := longTrigger ? 1 : -1
    sigEntry  := close
    sigAVWAP  := vwapVal
    lastSigBar := bar_index
    float d = close - vwapVal
    float bufSign = d >= 0 ? -1.0 : 1.0
    sigTP1 := vwapVal + d * i_tp1Mult
    sigTP2 := vwapVal + d * i_tp2Mult
    sigTP3 := i_showTP3 ? vwapVal + d * i_tp3Mult : na
    sigSL  := vwapVal + d * i_slFibMult + atrVal14 * i_atrBufMult * bufSign
    tradeState := "ENTERED"
    tp1Hit := false
    tp2Hit := false
    tp3Hit := false
    slHit  := false
    trailLevel := na

// TP/SL hit detection (state machine progression)
if sigDir == 1 and tradeState != "INIT" and tradeState != "EXITED"
    if not tp1Hit and high >= sigTP1
        tp1Hit := true
        tradeState := "TP1_HIT"
    if not tp2Hit and high >= sigTP2
        tp2Hit := true
        tradeState := "TP2_HIT"
    if not tp3Hit and not na(sigTP3) and high >= sigTP3
        tp3Hit := true
        tradeState := i_showTrail ? "TP3_HIT" : "EXITED"
    if not slHit and low <= sigSL
        slHit := true
        tradeState := "EXITED"

if sigDir == -1 and tradeState != "INIT" and tradeState != "EXITED"
    if not tp1Hit and low <= sigTP1
        tp1Hit := true
        tradeState := "TP1_HIT"
    if not tp2Hit and low <= sigTP2
        tp2Hit := true
        tradeState := "TP2_HIT"
    if not tp3Hit and not na(sigTP3) and low <= sigTP3
        tp3Hit := true
        tradeState := i_showTrail ? "TP3_HIT" : "EXITED"
    if not slHit and high >= sigSL
        slHit := true
        tradeState := "EXITED"

// Chandelier trail (option, after TP2)
if i_showTrail and (tradeState == "TP2_HIT" or tradeState == "TP3_HIT")
    atrCh = ta.atr(22)
    if sigDir == 1
        newTrail = high - atrCh * 3.0
        trailLevel := na(trailLevel) ? newTrail : math.max(trailLevel, newTrail)
        if low <= trailLevel
            tradeState := "EXITED"
    else if sigDir == -1
        newTrail = low + atrCh * 3.0
        trailLevel := na(trailLevel) ? newTrail : math.min(trailLevel, newTrail)
        if high >= trailLevel
            tradeState := "EXITED"

        line.delete(array.get(planLines, i))
    array.clear(planLines)

    for i = array.size(planLabels) - 1 to 0
        label.delete(array.get(planLabels, i))
    array.clear(planLabels)
if array.size(planBoxes) > 0
    for i = array.size(planBoxes) - 1 to 0
        box.delete(array.get(planBoxes, i))
    array.clear(planBoxes)

    for i = array.size(pinLines) - 1 to 0
        line.delete(array.get(pinLines, i))
    array.clear(pinLines)

    for i = array.size(pinLabels) - 1 to 0
        label.delete(array.get(pinLabels, i))
    array.clear(pinLabels)

statusText = switch tradeState
    "INIT"     => "Wait for entry signal"
    "ENTERED"  => "Hold for TP1 / Close 1/3 at TP1, move SL to Entry"
    "TP1_HIT"  => "Hold 2/3 for TP2 / Close 1/3 at TP2, move SL to TP1"
    "TP2_HIT"  => "Hold 1/3 (Trail) - Chandelier recommended"
    "TP3_HIT"  => "Hold last 1/3 with trail"
    "EXITED"   => "Exited. Wait for next signal"
    => "-"

// ============================================================================
// Sigma Band (option, single box, barstate.islast)
// ============================================================================

if i_showSigma and barstate.islast and not na(vwapVal) and not na(sigma) and sigma > 0
    int xLs = bar_index - 200
    int xRs = bar_index + i_planLookback
    color bandFill = color.new(i_colAVWAP, 88)
    bxSig = box.new(xLs, vwapVal + sigma, xRs, vwapVal - sigma,
         bgcolor=bandFill, border_color=color.new(i_colAVWAP, 100))
    array.push(planBoxes, bxSig)

// ============================================================================
// Plan Drawing (Entry / SL / TP1 / TP2 / TP3 / Trail) on barstate.islast
// ============================================================================

if barstate.islast and tradeState != "INIT" and not na(sigEntry) and not na(sigSL)
    int xL = sigBarIdx
    int xR = bar_index + i_planLookback
    int labelX = bar_index + 5

    // Real R:R (close-based, ATR buffer included) - W-5 unified RR
    float slDistLbl = math.abs(sigEntry - sigSL)
    float rr1RealLbl = slDistLbl > 0 ? math.abs(sigTP1 - sigEntry) / slDistLbl : 0.0
    float rr2RealLbl = slDistLbl > 0 ? math.abs(sigTP2 - sigEntry) / slDistLbl : 0.0
    float rr3RealLbl = (slDistLbl > 0 and not na(sigTP3)) ? math.abs(sigTP3 - sigEntry) / slDistLbl : 0.0

    // Entry (white solid)
    lnE = line.new(xL, sigEntry, xR, sigEntry, color=C_WHITE, width=2, style=line.style_solid)
    array.push(planLines, lnE)
    lblE = label.new(labelX, sigEntry, "Entry " + f_formatPrice(sigEntry),
         style=label.style_label_left, color=C_WHITE, textcolor=color.black, size=size.large)
    array.push(planLabels, lblE)

    // SL (red solid, thick) - "(Fib 0.618 retrace)"
    lnS = line.new(xL, sigSL, xR, sigSL, color=i_colSig2, width=3, style=line.style_solid)
    array.push(planLines, lnS)
    string slLblTxt = "SL " + f_formatPrice(sigSL) + "  (Fib 0.618 retrace)"
    lblS = label.new(labelX, sigSL, slLblTxt,
         style=label.style_label_left, color=i_colSig2, textcolor=C_WHITE, size=size.large)
    array.push(planLabels, lblS)

    // TP1 (green solid, bright)
    lnT1 = line.new(xL, sigTP1, xR, sigTP1, color=i_colSig, width=2, style=line.style_solid)
    array.push(planLines, lnT1)
    string tp1Suffix = tp1Hit ? "  HIT" : "  (Fib " + str.tostring(i_tp1Mult, "#.###") + ", RR " + str.tostring(rr1RealLbl, "#.##") + ")"
    lblT1 = label.new(labelX, sigTP1, "TP1 " + f_formatPrice(sigTP1) + tp1Suffix,
         style=label.style_label_left, color=i_colSig, textcolor=color.white, size=size.large)
    array.push(planLabels, lblT1)

    // TP2 (green solid, dimmer)
    lnT2 = line.new(xL, sigTP2, xR, sigTP2, color=color.new(i_colSig, 30), width=2, style=line.style_solid)
    array.push(planLines, lnT2)
    string tp2Suffix = tp2Hit ? "  HIT" : "  (Fib " + str.tostring(i_tp2Mult, "#.###") + ", RR " + str.tostring(rr2RealLbl, "#.##") + ")"
    lblT2 = label.new(labelX, sigTP2, "TP2 " + f_formatPrice(sigTP2) + tp2Suffix,
         style=label.style_label_left, color=i_colSig, textcolor=color.white, size=size.large)
    array.push(planLabels, lblT2)

    // TP3 (option, green dotted)
    if i_showTP3 and not na(sigTP3)
        lnT3 = line.new(xL, sigTP3, xR, sigTP3, color=color.new(i_colSig, 50), width=1, style=line.style_dotted)
        array.push(planLines, lnT3)
        string tp3Suffix = tp3Hit ? "  HIT" : "  (Fib " + str.tostring(i_tp3Mult, "#.###") + ", RR " + str.tostring(rr3RealLbl, "#.##") + ")"
        lblT3 = label.new(labelX, sigTP3, "TP3 " + f_formatPrice(sigTP3) + tp3Suffix,
             style=label.style_label_left, color=i_colSig, textcolor=color.white, size=size.large)
        array.push(planLabels, lblT3)

    // Chandelier trail (option ON + after TP2)
    if i_showTrail and not na(trailLevel) and (tradeState == "TP2_HIT" or tradeState == "TP3_HIT")
        lnTr = line.new(xL, trailLevel, xR, trailLevel, color=color.yellow, width=1, style=line.style_dashed)
        array.push(planLines, lnTr)
        lblTr = label.new(labelX, trailLevel, "Trail " + f_formatPrice(trailLevel),
             style=label.style_label_left, color=color.new(color.yellow, 70), textcolor=#000000, size=size.small)
        array.push(planLabels, lblTr)

// Anchor Pin (option, barstate.islast)
// ============================================================================

if i_showAnchorPin and not na(anchorBarIdx) and barstate.islast and not na(atrVal14)
    float pinTop = high + atrVal14 * 0.5
    float pinBot = low  - atrVal14 * 0.5
    lnPin = line.new(anchorBarIdx, pinBot, anchorBarIdx, pinTop, color=color.new(i_colAVWAP, 50), width=1)
    array.push(pinLines, lnPin)
    lblPin = label.new(anchorBarIdx, pinTop, f_anchorLetter(i_anchorMode),
         style=label.style_label_down, color=i_colAVWAP, textcolor=C_WHITE, size=size.large)
    array.push(pinLabels, lblPin)

     bgcolor=C_PANEL_BG,
     frame_color=C_FRAME, frame_width=1,
     border_color=C_FRAME, border_width=1)

if i_showDash and barstate.islast
    dashSz = f_dashSize(i_dashSize)
    if i_dashMin
        // Minimized: 1 row only
        string minText = "◆ AVWAP Plan"
        if tradeState != "INIT"
            string dirCh = sigDir == 1 ? "LONG" : "SHORT"
            string stCh  = tp2Hit ? "TP2" : tp1Hit ? "TP1" : "OPEN"
            minText := "◆ AVWAP " + dirCh + " " + stCh
        table.cell(dashTable, 0, 0, minText, text_color=C_WHITE, text_size=dashSz, bgcolor=C_PANEL_BG)
        // Clear remaining cells
        table.cell(dashTable, 1, 0, "", bgcolor=C_PANEL_BG)
        for r = 1 to 6
            table.cell(dashTable, 0, r, "", bgcolor=C_PANEL_BG)
            table.cell(dashTable, 1, r, "", bgcolor=C_PANEL_BG)
    else
