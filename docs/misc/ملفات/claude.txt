# خطة شاملة لموازنة البوت وتحسين أدائه

## 1️⃣ **تشخيص المشكلة الأساسية**

مشكلتك الحالية هي **المعضلة الكلاسيكية للتداول الخوارزمي**:
- **تشديد الشروط** → إشارات نادرة جداً (Precision عالية، Recall منخفض)
- **تخفيف الشروط** → إشارات كثيرة فاشلة (Recall عالي، Precision منخفض)

```python
class FilterPerformanceTracker:
    def __init__(self):
        self.filter_stats = {
            'mss_strong': {'trades': 0, 'wins': 0},
            'fvg_wide': {'trades': 0, 'wins': 0},
            'imbalance_high': {'trades': 0, 'wins': 0}
        }
    
    def record_trade(self, filters_passed, win):
        for filter_name in filters_passed:
            self.filter_stats[filter_name]['trades'] += 1
            if win:
                self.filter_stats[filter_name]['wins'] += 1
    
    def get_best_filters(self):
        # ابحث عن المرشحات ذات أعلى نسبة نجاح
        return sorted(
            self.filter_stats.items(),
            key=lambda x: x[1]['wins'] / max(x[1]['trades'], 1),
            reverse=True
        )
```


```
2x ATR للخسارة، 4x للربح
لا يتكيف مع تقلبات السوق المتغيرة
```

### **الحل 2: نظام وقف خسارة ديناميكي (Dynamic ATR)**

```python
class DynamicRiskManager:
    """إدارة مخاطر متكيفة مع التقلبات"""
    
    def calculate_adaptive_stops(self, df_5m, entry_price):
        """حساب وقف الخسارة والهدف بناءً على التقلب الفعلي"""
        
        # حساب ATR على 3 فترات زمنية
        atr_fast = self.calculate_atr(df_5m, period=7)   # حساس
        atr_medium = self.calculate_atr(df_5m, period=14) # متوسط
        atr_slow = self.calculate_atr(df_5m, period=28)   # بطيء
        
        # حساب معامل التقلب
        volatility_ratio = atr_fast / atr_slow if atr_slow > 0 else 1.0
        
        # تحديد المضاعف الديناميكي
        if volatility_ratio > 1.5:
            # تقلب شديد - توسيع الوقف
            sl_multiplier = 3.0
            tp_multiplier = 6.0
            regime = "HIGH_VOLATILITY"
        elif volatility_ratio > 1.2:
            # تقلب متوسط
            sl_multiplier = 2.5
            tp_multiplier = 5.0
            regime = "MEDIUM_VOLATILITY"
        else:
            # تقلب منخفض - تشديد الوقف
            sl_multiplier = 1.5
            tp_multiplier = 3.5
            regime = "LOW_VOLATILITY"
        
        # حساب المستويات الفعلية
        stop_loss = entry_price - (atr_medium * sl_multiplier)
        take_profit = entry_price + (atr_medium * tp_multiplier)
        
        # ضمان نسبة المخاطرة/العائد لا تقل عن 1:2
        risk = entry_price - stop_loss
        reward = take_profit - entry_price
        
        if reward / risk < 2.0:
            take_profit = entry_price + (risk * 2.0)
        
        return {
            'stop_loss': round(stop_loss, 8),
            'take_profit': round(take_profit, 8),
            'atr_used': atr_medium,
            'regime': regime,
            'risk_reward': round(reward / risk, 2)
        }
    
    @staticmethod
    def calculate_atr(df, period=14):
        """حساب ATR الصحيح"""
        high = df['high']
        low = df['low']
        close = df['close']
        
        tr1 = high - low
        tr2 = abs(high - close.shift())
        tr3 = abs(low - close.shift())
        
        tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
        atr = tr.ewm(span=period, adjust=False).mean()
        
        return atr.iloc[-1]
```

```python
class SmartOrderBookFilter:
    """فلتر ذكي لدفتر الأوامر يتجنب الإشارات الكاذبة"""
    
    def __init__(self):
        self.orderbook_history = {}  # تاريخ دفتر الأوامر
        self.window_size = 10        # عدد القراءات
    
    def evaluate_orderbook(self, symbol, bid_qty, ask_qty):
        """تقييم دفتر الأوامر بناءً على الاتجاه المستمر"""
        
        # حفظ القراءة الحالية
        if symbol not in self.orderbook_history:
            self.orderbook_history[symbol] = []
        
        ratio = bid_qty / ask_qty if ask_qty > 0 else 0
        self.orderbook_history[symbol].append({
            'ratio': ratio,
            'timestamp': time.time()
        })
        
        # الاحتفاظ بآخر 10 قراءات فقط
        self.orderbook_history[symbol] = self.orderbook_history[symbol][-self.window_size:]
        
        # حساب المتوسط المتحرك للنسبة
        recent_ratios = [x['ratio'] for x in self.orderbook_history[symbol]]
        avg_ratio = np.mean(recent_ratios)
        
        # التأكيد: النسبة الحالية والمتوسط يجب أن يكونا في اتجاه الشراء
        if ratio > 1.3 and avg_ratio > 1.15:
            return True, "STRONG_BUY_PRESSURE"
        elif ratio > 1.1 and avg_ratio > 1.05:
            return True, "MODERATE_BUY_PRESSURE"
        else:
            return False, "INSUFFICIENT_PRESSURE"
```

```python
class SignalScorer:
    """نظام تقييم الإشارات بدلاً من الرفض/القبول الثنائي"""
    
    def score_signal(self, symbol, df_5m, df_15m, orderbook_data):
        """إعطاء نقاط للإشارة من 0-100"""
        
        score = 0
        reasons = []
        
        # 1. تقييم الاتجاه العام (25 نقطة)
        current_price = df_5m['close'].iloc[-1]
        ema_200_15m = df_15m['close'].ewm(span=200).mean().iloc[-1]
        ema_50_15m = df_15m['close'].ewm(span=50).mean().iloc[-1]
        
        if current_price > ema_200_15m:
            score += 25
            reasons.append("+25: Above EMA200")
        elif current_price > ema_50_15m:
            score += 15
            reasons.append("+15: Above EMA50 only")
        else:
            score += 5
            reasons.append("+5: Below major EMAs (risky)")
        
        # 2. قوة كسر الهيكل (25 نقطة)
        mss_strength = self.calculate_mss_strength(df_5m)
        if mss_strength > 0.3:
            score += 25
            reasons.append(f"+25: Strong MSS ({mss_strength:.2%})")
        elif mss_strength > 0.2:
            score += 15
            reasons.append(f"+15: Moderate MSS ({mss_strength:.2%})")
        elif mss_strength > 0.1:
            score += 8
            reasons.append(f"+8: Weak MSS ({mss_strength:.2%})")
        
        # 3. جودة FVG (20 نقطة)
        fvg_quality = self.evaluate_fvg(df_5m)
        score += min(fvg_quality * 100, 20)
        reasons.append(f"+{min(fvg_quality * 100, 20):.0f}: FVG quality")
        
        # 4. دفتر الأوامر (15 نقطة)
        ob_score = self.score_orderbook(orderbook_data)
        score += ob_score
        reasons.append(f"+{ob_score}: Orderbook pressure")
        
        # 5. حجم التداول (15 نقطة)
        volume_score = self.score_volume(df_5m)
        score += volume_score
        reasons.append(f"+{volume_score}: Volume analysis")
        
        return {
            'total_score': score,
            'reasons': reasons,
            'should_trade': score >= 60  # عتبة الدخول 60/100
        }
```

```python
class AggresivePositionManager:
    """إدارة صفقات عدوانية مع حماية صارمة"""
    
    def __init__(self):
        self.max_concurrent_positions = 5  # بدلاً من 3
        self.cooldown_period = 120         # بدلاً من 300 ثانية
        self.min_score_threshold = 55      # عتبة منخفضة
        self.high_score_threshold = 75     # عتبة عالية
        
    def should_enter_position(self, symbol, score, current_positions):
        """قرار الدخول بناءً على النقاط والظروف"""
        
        # عدد المراكز الحالية على هذا الرمز
        symbol_positions = len([p for p in current_positions if p['symbol'] == symbol])
        
        # سياسة مرنة:
        if score >= self.high_score_threshold:
            # إشارة قوية جداً - دخول مضمون
            max_allowed = 3
        elif score >= self.min_score_threshold:
            # إشارة متوسطة - حذر
            max_allowed = 2
        else:
            # إشارة ضعيفة - رفض
            return False, "Score too low"
        
        if symbol_positions >= max_allowed:
            return False, f"Max positions reached for {symbol}"
        
        if len(current_positions) >= self.max_concurrent_positions:
            return False, "Max total positions reached"
        
        return True, "Entry approved"
    
    def calculate_position_size(self, score, account_balance):
        """حجم الصفقة يعتمد على قوة الإشارة"""
        
        base_size = 20  # دولار
        
        if score >= 85:
            return base_size * 1.5  # 30 دولار للإشارات القوية جداً
        elif score >= 75:
            return base_size * 1.2  # 24 دولار
        elif score >= 65:
            return base_size        # 20 دولار
        else:
            return base_size * 0.8  # 16 دولار للإشارات الضعيفة
```

### **تعديل دالة check_entry_signal الحالية:**

```python
async def check_entry_signal(self, symbol: str):
    """منطق دخول محسّن"""
    
    df_5m = self.get_candles(symbol, '5m')
    df_15m = self.get_candles(symbol, '15m')
    
    if df_5m is None or df_15m is None:
        return
    
    # 1. تحديد حالة السوق
    regime = self.adaptive_filters.detect_regime(df_5m, df_15m)
    entry_reqs = self.adaptive_filters.get_entry_requirements(regime)
    
    if entry_reqs.get('skip_trading'):
        return  # تخطي التداول في الترند الهابط
    
    # 2. حساب نقاط الإشارة
    signal_eval = self.signal_scorer.score_signal(
        symbol, df_5m, df_15m, 
        self.latest_orderbook.get(symbol)
    )
    
    # 3. فحص إمكانية الدخول
    can_enter, reason = self.position_manager.should_enter_position(
        symbol, 
        signal_eval['total_score'],
        self.active_positions
    )
    
    if not can_enter:
        logger.debug(f"{symbol}: Entry rejected - {reason}")
        return
    
    # 4. حساب حجم الصفقة الديناميكي
    position_size_usd = self.position_manager.calculate_position_size(
        signal_eval['total_score'],
        self.account_balance
    )
    
    # 5. حساب مستويات الخروج الديناميكية
    entry_price = self.latest_price.get(symbol)
    exit_levels = self.risk_manager.calculate_adaptive_stops(
        df_5m, entry_price
    )
    
    # 6. تنفيذ الأمر
    logger.info(f"""
    🎯 ENTRY SIGNAL: {symbol}
    Score: {signal_eval['total_score']}/100
    Regime: {regime.value}
    Entry: {entry_price}
    SL: {exit_levels['stop_loss']} (-{((entry_price - exit_levels['stop_loss'])/entry_price)*100:.2f}%)
    TP: {exit_levels['take_profit']} (+{((exit_levels['take_profit'] - entry_price)/entry_price)*100:.2f}%)
    R:R: 1:{exit_levels['risk_reward']}
    Size: ${position_size_usd}
    Volatility: {exit_levels['regime']}
    """)
    
    await self.execute_entry(symbol, position_size_usd, exit_levels)
```
