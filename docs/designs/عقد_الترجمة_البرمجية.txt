from typing import Tuple
import numpy as np
import pandas as pd

# [F-XXX] الاسم — النوع
# المصدر: (المصدر من الكتلة)
SWEEP_PARAMS = {          # كل معامل من "المعاملات_القابلة_للسويب" هنا
    "rsi2_os": [5, 10, 15],
    "sma_trend": [100, 200],
}

def make_signals(df: pd.DataFrame, **params) -> pd.Series:
    """
    df أعمدته: open high low close volume (فهرس datetime UTC) + المؤشرات الجاهزة
    إن احتجت مؤشراً غير موجود: احسبه داخلياً بـpandas/numpy فقط.
    ترجع: Series بولياني (True = شمعة إشارة دخول صالحة).
    الإشارة تُقرأ عند إغلاق الشمعة، والتنفيذ على افتتاح الشمعة التالية (تتولاه آلة الاختبار).
    """
    rsi_os = params.get("rsi2_os", 10)
    sma_trend = params.get("sma_trend", 200)

    # ==== الحسابات (pandas/numpy فقط — مكشوفة موجهة، ممنوع حلقات for على الصفوف) ====
    sma = df["close"].rolling(sma_trend, min_periods=sma_trend//2).mean()
    delta = df["close"].diff()
    gain = delta.clip(lower=0).ewm(alpha=1/rsi_os, adjust=False).mean()
    loss = (-delta.clip(upper=0)).ewm(alpha=1/rsi_os, adjust=False).mean()
    rsi2 = 100 - 100/(1 + gain/loss.replace(0, np.nan))

    # ==== الشرط الآلي حرفياً من الكتلة ====
    sig = (df["close"] > sma) & (rsi2 < rsi_os)

    return sig.fillna(False)

def self_test():
    """اختبار ذاتي إلزامي — يجب أن يمر قبل التسليم"""
    rng = np.random.default_rng(42)
    n = 3000
    idx = pd.date_range("2024-01-01", periods=n, freq="5min", tz="UTC")
    df = pd.DataFrame({
        "open": 100 + rng.normal(0, 1, n).cumsum(),
        "high": 0, "low": 0, "volume": rng.uniform(1, 100, n),
    }, index=idx)
    df["close"] = df["open"] + rng.normal(0, 0.5, n)
    df["high"] = df[["open", "close"]].max(axis=1) + abs(rng.normal(0, 0.3, n))
    df["low"] = df[["open", "close"]].min(axis=1) - abs(rng.normal(0, 0.3, n))

    sig = make_signals(df)
    assert isinstance(sig, pd.Series), "المخرج Series"
    assert sig.dtype == bool, "المخرج بولياني"
    assert len(sig) == len(df), "نفس الطول"
    assert sig.iloc[:sma_trend].sum() == 0 or True, "بداية بلا إشارات مقبولة"
    # استدعاء بمعاملات مختلفة يجب أن يعمل
    for p in SWEEP_PARAMS["rsi2_os"]:
        make_signals(df, rsi2_os=p)
    print(f"[F-XXX] self_test OK — إشارات: {sig.sum()}")

if __name__ == "__main__":
    self_test()

القواعد الصارمة:
1. ممنوع حلقات for/while على صفوف df — كل الحسابات موجهة (rolling/ewm/shift/np.where)
2. الدالة نقية: لا طباعة لا ملفات لا حالة عالمية
3. كل معامل من السويب يُقرأ params.get مع قيمة افتراضية من الكتلة
4. استخدم أعمدة df الجاهزة أولاً (close/high/low/volume) قبل إعادة حساب أي مؤشر
5. NaN يُعامل آمناً (fillna(False) في النهاية) — إشارة ناقصة ليست إشارة
6. self_test إلزامي ويمر قبل التسليم — بيانات عشوائية seed ثابت 42
7. إن كان الشرط يحتاج مؤشراً من إطار أعلى (4h/1d): أعد-shifting صريح واحسبه من df نفسها
   بدالة resample داخلياً، ووثّق ذلك في تعليق — ممنوع افتراض أعمدة خارجية
8. شرط "الشمعة السابقة" = shift(1) دائماً — لا lookahead أبداً
9. أنواع الحمولة: دخول=Series بولياني | فلتر=Series بولياني (يُضرب AND مع الدخول)
   | خروج=Series بولياني (يخرج عند True) | إدارة_حصة=Series قيم 0-1 (نسبة الحجم)
