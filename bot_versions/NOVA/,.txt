# المؤشرات الرياضية
def calc_ema(prices, period):
    if len(prices) < period: return prices[-1]
    k = 2 / (period + 1)
    ema = sum(prices[:period]) / period
    for price in prices[period:]: ema = (price - ema) * k + ema
    return ema

def calc_rsi(prices, period=14):
    if len(prices) < period + 1: return 50
    gains = [max(0, prices[i] - prices[i-1]) for i in range(1, period + 1)]
    losses = [max(0, prices[i-1] - prices[i]) for i in range(1, period + 1)]
    avg_gain = sum(gains) / period
    avg_loss = sum(losses) / period
    for i in range(period + 1, len(prices)):
        change = prices[i] - prices[i-1]
        avg_gain = (avg_gain * 13 + max(0, change)) / 14
        avg_loss = (avg_loss * 13 + max(0, -change)) / 14
    if avg_loss == 0: return 100
    return 100 - (100 / (1 + (avg_gain / avg_loss)))

def calc_macd(prices):
    return calc_ema(prices, 12) - calc_ema(prices, 26)

def get_binance_data(symbol, interval):
    url = f"https://api.binance.com/api/v3/klines?symbol={symbol}&interval={interval}&limit=100"
    res = requests.get(url, timeout=5).json()
    closes = [float(c[4]) for c in res]
    volumes = [float(c[5]) for c in res]
    return closes[-1], calc_rsi(closes, 14), calc_macd(closes), volumes[-1], (sum(volumes[-20:]) / 20 if len(volumes) >= 20 else sum(volumes) / len(volumes))
