//@version=5
indicator("Alpha Hunter System [MTF + Risk Manager]", overlay=true, max_lines_count=100, max_labels_count=100)

// ==========================================
// 1. SETTINGS
// ==========================================
// --- Main Channel ---
grp_main = "Channel Settings"
emaLen   = input.int(20, title="EMA Length", group=grp_main)
sens     = input.float(1.0, title="Sensitivity (ATR)", minval=0.0, step=0.1, group=grp_main)

// --- Risk Management ---
grp_risk = "Risk Management (Targets)"
rr1      = input.float(1, title="TP 1 Ratio", step=0.5, group=grp_risk)
rr2      = input.float(1.5, title="TP 2 Ratio", step=0.5, group=grp_risk)
rr3      = input.float(2.0, title="TP 3 Ratio", step=0.5, group=grp_risk)

// --- Trailing Stop ---
grp_trail = "Trailing Stop (Profit Locker)"
useTrail  = input.bool(true, title="Show Trailing Stop Line?", group=grp_trail)
trailMult = input.float(2.0, title="Trailing ATR Multiplier", step=0.1, group=grp_trail)

// --- Filters ---
grp_flt  = "Filters (Volume & ADX)"
useVol   = input.bool(true, title="Filter Low Volume?", group=grp_flt)
volLen   = input.int(20, title="Volume MA Length", group=grp_flt)
useFlt   = input.bool(true, title="Filter Sideways (ADX)?", group=grp_flt)
adxThresh= input.int(20, title="ADX Threshold", minval=1, group=grp_flt)

// --- Visuals ---
grp_vis  = "Visuals"
colHigh  = input.color(#089981, title="Bullish Color", group=grp_vis)
colLow   = input.color(#f23645, title="Bearish Color", group=grp_vis)
colSide  = input.color(color.gray, title="Sideways Color", group=grp_vis)
colTrail = input.color(#9c27b0, title="Trailing Stop Color", group=grp_vis)

// ==========================================
// 2. CALCULATIONS
// ==========================================
// --- EMAs & ATR ---
emaH = ta.ema(high, emaLen)
emaL = ta.ema(low, emaLen)
atrVal = ta.atr(14)

// Sensitivity Bands
filterHigh = emaH + (atrVal * sens)
filterLow  = emaL - (atrVal * sens)

// --- Volume Filter ---
volAvg = ta.sma(volume, volLen)
volValid = useVol ? (volume > volAvg * 1.0) : true 

// --- Sideways Filter (ADX) ---
dmp = ta.change(high)
dmn = -ta.change(low)
plus = fixnan(100 * ta.rma(dmp > dmn and dmp > 0 ? dmp : 0, 14) / ta.atr(14))
minus = fixnan(100 * ta.rma(dmn > dmp and dmn > 0 ? dmn : 0, 14) / ta.atr(14))
sum = plus + minus
adxValue = 100 * ta.rma(math.abs(plus - minus) / (sum == 0 ? 1 : sum), 14)
isSideways = useFlt ? (adxValue < adxThresh) : false

// ==========================================
// 3. SIGNAL GENERATION
// ==========================================
var int trendState = 0 
crossBuy  = ta.crossover(close, filterHigh)
crossSell = ta.crossunder(close, filterLow)

// Combined Logic
validBuy  = crossBuy and not isSideways and volValid
validSell = crossSell and not isSideways and volValid

if validBuy
    trendState := 1
if validSell
    trendState := -1

signalBuy  = (trendState == 1) and (trendState[1] != 1)
signalSell = (trendState == -1) and (trendState[1] != -1)

// ==========================================
// 4. TRAILING STOP LOGIC
// ==========================================
var float trailPrice = na
trailATR = ta.atr(14) * trailMult

if signalBuy
    trailPrice := low - trailATR
else if signalSell
    trailPrice := high + trailATR
else if trendState == 1
    trailPrice := math.max(trailPrice, close - trailATR)
else if trendState == -1
    trailPrice := math.min(trailPrice, close + trailATR)

// ==========================================
// 5. TRADE & TP TRACKING LOGIC
// ==========================================
// Arrays to store history
var string[] tradeTypes = array.new_string(0) // "LONG" or "SHORT"
var bool[] resTP1 = array.new_bool(0)
var bool[] resTP2 = array.new_bool(0)
var bool[] resTP3 = array.new_bool(0)

// Active Trade Variables
var bool inTrade = false
var string activeType = na
var bool activeTP1Hit = false
var bool activeTP2Hit = false
var bool activeTP3Hit = false

// TP Prices for current trade
var float curTP1 = na
var float curTP2 = na
var float curTP3 = na

// --- DETECT NEW ENTRY ---
if signalBuy or signalSell
    // If a trade was already active, save its result before starting new one
    if inTrade
        array.unshift(tradeTypes, activeType)
        array.unshift(resTP1, activeTP1Hit)
        array.unshift(resTP2, activeTP2Hit)
        array.unshift(resTP3, activeTP3Hit)
        // Trim Arrays to 10
        if array.size(tradeTypes) > 10
            array.pop(tradeTypes), array.pop(resTP1), array.pop(resTP2), array.pop(resTP3)

    // Start New Trade
    inTrade := true
    activeType := signalBuy ? "LONG" : "SHORT"
    activeTP1Hit := false
    activeTP2Hit := false
    activeTP3Hit := false
    
    // Calculate Targets
    staticSL = signalBuy ? emaL : emaH
    risk = math.abs(close - staticSL)
    curTP1 := close + (signalBuy ? risk*rr1 : -risk*rr1)
    curTP2 := close + (signalBuy ? risk*rr2 : -risk*rr2)
    curTP3 := close + (signalBuy ? risk*rr3 : -risk*rr3)

// --- CHECK TP HITS DURING TRADE ---
if inTrade
    if activeType == "LONG"
        if high >= curTP1 
            activeTP1Hit := true
        if high >= curTP2 
            activeTP2Hit := true
        if high >= curTP3 
            activeTP3Hit := true
    else 
        if low <= curTP1 
            activeTP1Hit := true
        if low <= curTP2 
            activeTP2Hit := true
        if low <= curTP3 
            activeTP3Hit := true

// ==========================================
// 6. DRAWING & VISUALS
// ==========================================
color finalColor = isSideways ? color.new(colSide, 70) : trendState == 1 ? color.new(colHigh, 80) : color.new(colLow, 80)
p1 = plot(emaH, "High Band", isSideways ? color.new(colSide, 50) : colHigh)
p2 = plot(emaL, "Low Band", isSideways ? color.new(colSide, 50) : colLow)
fill(p1, p2, finalColor)

// Trailing Line
bool showTrail = useTrail and not isSideways
plot(showTrail ? trailPrice : na, "Trailing Stop", color=colTrail, style=plot.style_linebr, linewidth=2)

// Trailing Label
var label lb_trail = na
label.delete(lb_trail) 
if showTrail and not na(trailPrice) and barstate.islast
    lb_trail := label.new(bar_index + 3, trailPrice, "Trail SL: " + str.tostring(trailPrice, format.mintick), color=colTrail, style=label.style_label_left, textcolor=color.white, size=size.small)

// Signal Labels
plotshape(signalBuy, "Buy", shape.labelup, location.belowbar, colHigh, 0, " BUY ", color.white, size=size.small)
plotshape(signalSell, "Sell", shape.labeldown, location.abovebar, colLow, 0, " SELL ", color.white, size=size.small)

// TP Lines Drawing
var line l_entry = na
var line l_tp1 = na
var line l_tp2 = na
var line l_tp3 = na
var label lb_entry = na
var label lb_tp1 = na
var label lb_tp2 = na
var label lb_tp3 = na

drawLevel(price, col, style, txt) =>
    l = line.new(bar_index, price, bar_index + 10, price, color=col, style=style)
    lb = label.new(bar_index + 10, price, txt + str.tostring(price, format.mintick), color=col, style=label.style_label_left, textcolor=color.white, size=size.small)
    [l, lb]

if signalBuy or signalSell
    line.delete(l_entry), line.delete(l_tp1), line.delete(l_tp2), line.delete(l_tp3)
    label.delete(lb_entry), label.delete(lb_tp1), label.delete(lb_tp2), label.delete(lb_tp3)
    
    [le, lbe] = drawLevel(close, color.blue, line.style_solid, "ENTRY: ")
    l_entry := le, lb_entry := lbe
    
    [lt1, lbt1] = drawLevel(curTP1, colHigh, line.style_dashed, "TP1: ")
    l_tp1 := lt1, lb_tp1 := lbt1
    
    [lt2, lbt2] = drawLevel(curTP2, colHigh, line.style_dashed, "TP2: ")
    l_tp2 := lt2, lb_tp2 := lbt2
    
    [lt3, lbt3] = drawLevel(curTP3, colHigh, line.style_dashed, "TP3: ")
    l_tp3 := lt3, lb_tp3 := lbt3

if not (signalBuy or signalSell) and not na(l_entry)
    line.set_x2(l_entry, bar_index + 5), label.set_x(lb_entry, bar_index + 5)
    line.set_x2(l_tp1, bar_index + 5), label.set_x(lb_tp1, bar_index + 5)
    line.set_x2(l_tp2, bar_index + 5), label.set_x(lb_tp2, bar_index + 5)
    line.set_x2(l_tp3, bar_index + 5), label.set_x(lb_tp3, bar_index + 5)

// ==========================================
// 7. RESULTS TABLE (TRANSPARENT + BLACK HEADER)
// ==========================================
var table resTable = table.new(position.bottom_right, 5, 11, border_width=0)

if barstate.islast
    // Header (Transparent Background, Black Text)
    headerColor = color.new(color.gray, 95) 
    textColor   = color.black // Changed to Black
    
    table.cell(resTable, 0, 0, "#", bgcolor=headerColor, text_color=textColor)
    table.cell(resTable, 1, 0, "Type", bgcolor=headerColor, text_color=textColor)
    table.cell(resTable, 2, 0, "TP 1", bgcolor=headerColor, text_color=textColor)
    table.cell(resTable, 3, 0, "TP 2", bgcolor=headerColor, text_color=textColor)
    table.cell(resTable, 4, 0, "TP 3", bgcolor=headerColor, text_color=textColor)
    
    // Rows (Transparent Background)
    transpBg = color.new(color.black, 100)
    
    for i = 0 to array.size(tradeTypes) - 1
        t_type = array.get(tradeTypes, i)
        t_tp1  = array.get(resTP1, i)
        t_tp2  = array.get(resTP2, i)
        t_tp3  = array.get(resTP3, i)
        
        // Dynamic Text Color for LONG/SHORT
        typeCol = t_type == "LONG" ? colHigh : colLow
        
        table.cell(resTable, 0, i+1, str.tostring(i+1), bgcolor=transpBg, text_color=color.gray, text_size=size.small)
        table.cell(resTable, 1, i+1, t_type, bgcolor=transpBg, text_color=typeCol, text_size=size.small)
        
        // TP Columns
        table.cell(resTable, 2, i+1, t_tp1 ? "✅" : "❌", bgcolor=transpBg, text_color=color.white, text_size=size.small)
        table.cell(resTable, 3, i+1, t_tp2 ? "✅" : "❌", bgcolor=transpBg, text_color=color.white, text_size=size.small)
        table.cell(resTable, 4, i+1, t_tp3 ? "✅" : "❌", bgcolor=transpBg, text_color=color.white, text_size=size.small)

// ==========================================
// 8. ALERTS
// ==========================================
alertcondition(signalBuy, "Buy Alert", "Valid BUY Signal")
alertcondition(signalSell, "Sell Alert", "Valid SELL Signal")
alertcondition(inTrade and activeTP1Hit, "TP1 Hit Alert", "Take Profit 1 Hit!")
alertcondition(inTrade and activeTP2Hit, "TP2 Hit Alert", "Take Profit 2 Hit!")
alertcondition(inTrade and activeTP3Hit, "TP3 Hit Alert", "Take Profit 3 Hit!")

// ==========================================================================================

var table myTable = table.new(position.top_center, 1, 1, border_width=1, frame_color=color.black, bgcolor=color.white)

// Add Telegram Message to Dashboard
table.cell(myTable, 0, 0, "Join Telegram @mrexpert_ai", bgcolor=color.blue, text_color=color.white, text_size=size.normal)
