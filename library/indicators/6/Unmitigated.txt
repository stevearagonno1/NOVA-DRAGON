// Convert line style string to constant
f_get_line_style(style_str) =>
    switch style_str
        "Solid" => line.style_solid
        "Dashed" => line.style_dashed
        "Dotted" => line.style_dotted
        => line.style_solid

// Get age-based color for high levels
f_get_high_color(age) =>
    age >= 7 ? color_ancient_high : age >= 4 ? color_old_high : age >= 2 ? color_med_high : color_new_high

// Get age-based color for low levels
f_get_low_color(age) =>age >= 7 ? color_ancient_low : age >= 4 ? color_old_low : age >= 2 ? color_med_low : color_new_low

// Timeframe formatter
f_format_timeframe(tf) =>
    string result = ""
    if tf == "1"
        result := "1 min"
    else if tf == "3"
        result := "3 mins"
    else if tf == "5"
        result := "5 mins"
    else if tf == "15"
        result := "15 mins"
    else if tf == "30"
        result := "30 mins"
    else if tf == "45"
        result := "45 mins"
    else if tf == "60"
        result := "1 Hour"
    else if tf == "120"
        result := "2 Hours"
    else if tf == "180"
        result := "3 Hours"
    else if tf == "240"
        result := "4 Hours"
    else if tf == "D" or tf == "1D"
        result := "1 Day"
    else if tf == "W" or tf == "1W"
        result := "1 Week"
    else if tf == "M" or tf == "1M"
        result := "1 Month"
    else
        result := tf
    result

// Table position helper
f_get_position(str) =>
    switch str
        "Top Left" => position.top_left
        "Top Right" => position.top_right
        "Bottom Left" => position.bottom_left
        "Bottom Right" => position.bottom_right
        => position.top_right

// Table size helper
f_get_size(str) =>
    switch str
        "Tiny" => size.tiny
        "Small" => size.small
        "Normal" => size.normal
        "Large" => size.large
        => size.normal

// Format level display
f_format_level(lvl, fmt_price) => 
    na(lvl) ? "N/A" : str.tostring(lvl, fmt_price)

// Format delta with sign
f_format_delta(delta, is_high) => 
    if na(delta)
        "N/A"
    else
        string prefix = is_high ? "+" : "-"
        string abs_delta_str = str.tostring(math.abs(delta), "#.00")
        prefix + abs_delta_str

// Format percentage distance
f_format_percentage(lvl, current_price) =>
    if na(lvl)
        "N/A"
    else
        float pct = ((lvl - current_price) / current_price) * 100
        string sign = pct > 0 ? "+" : ""
        sign + str.tostring(pct, "#.##") + "%"

// ═══════════════════════════════════════════════════════════════════════════

f_calculate_strat() =>
    string strat_pattern = "1"
    bool broke_high = TF_High_curr > TF_High_prev
    bool broke_low = TF_Low_curr < TF_Low_prev
    
    if broke_high and broke_low
        strat_pattern := "3"
    else if broke_high and not broke_low
        strat_pattern := "2U"
    else if broke_low and not broke_high
        strat_pattern := "2D"
    else
        strat_pattern := "1"
    strat_pattern

f_cleanup_old_levels(levels_arr, active_arr, start_time_arr, session_arr, max_size) =>
    int current_size = array.size(levels_arr)
    if current_size > max_size
        int to_remove = current_size - max_size
        for i = 0 to to_remove - 1
            array.shift(levels_arr)
            array.shift(active_arr)
            array.shift(start_time_arr)
            array.shift(session_arr)

// ═══════════════════════════════════════════════════════════════════════════

f_get_active_levels(levels_arr, active_arr, start_time_arr, session_arr) =>
    int n = array.size(levels_arr)
    var float[] active_lvls = array.new_float(5, na)
    var int[] active_times = array.new_int(5, na)
    var int[] active_sessions = array.new_int(5, na)
    
    if n > 0
        int active_count = 0
        for i = n - 1 to 0
            if array.get(active_arr, i) and active_count < 5
                array.set(active_lvls, active_count, array.get(levels_arr, i))
                array.set(active_times, active_count, array.get(start_time_arr, i))
                array.set(active_sessions, active_count, array.get(session_arr, i))
                active_count := active_count + 1
    
    [active_lvls, active_times, active_sessions]

// Count total active levels
f_count_active_levels(active_arr) =>
    int n = array.size(active_arr)
    int count = 0
    if n > 0
        for i = 0 to n - 1
            if array.get(active_arr, i)
                count := count + 1
    count

f_add_enhanced_table(table_id) =>
    if not na(table_id)
        // Extract levels
        float last_hi_lvl = array.get(hi_lvls_arr, 0)
        float second_last_hi_lvl = array.get(hi_lvls_arr, 1)
        float third_last_hi_lvl = array.get(hi_lvls_arr, 2)
        float fourth_last_hi_lvl = array.get(hi_lvls_arr, 3)
        float fifth_last_hi_lvl = array.get(hi_lvls_arr, 4)

        float delta_1h = not na(last_hi_lvl) ? (last_hi_lvl - close) : na
        float delta_2h = not na(second_last_hi_lvl) ? (second_last_hi_lvl - close) : na
        float delta_3h = not na(third_last_hi_lvl) ? (third_last_hi_lvl - close) : na
        float delta_4h = not na(fourth_last_hi_lvl) ? (fourth_last_hi_lvl - close) : na
        float delta_5h = not na(fifth_last_hi_lvl) ? (fifth_last_hi_lvl - close) : na

        float last_lo_lvl = array.get(lo_lvls_arr, 0)
        float second_last_lo_lvl = array.get(lo_lvls_arr, 1)
        float third_last_lo_lvl = array.get(lo_lvls_arr, 2)
        float fourth_last_lo_lvl = array.get(lo_lvls_arr, 3)
        float fifth_last_lo_lvl = array.get(lo_lvls_arr, 4)

        float delta_1l = not na(last_lo_lvl) ? (close - last_lo_lvl) : na
        float delta_2l = not na(second_last_lo_lvl) ? (close - second_last_lo_lvl) : na
        float delta_3l = not na(third_last_lo_lvl) ? (close - third_last_lo_lvl) : na
        float delta_4l = not na(fourth_last_lo_lvl) ? (close - fourth_last_lo_lvl) : na
        float delta_5l = not na(fifth_last_lo_lvl) ? (close - fifth_last_lo_lvl) : na
        
        // Net delta
        float net_delta_1h_1l = (not na(last_hi_lvl) and not na(last_lo_lvl)) ? (last_hi_lvl - last_lo_lvl) : na
        string net_delta_str = na(net_delta_1h_1l) ? "N/A" : ("Δ " + str.tostring(net_delta_1h_1l, "#.00"))
        
        // Age calculations
        int last_hi_session_index = array.get(hi_sess_arr, 0)
        int second_last_hi_session_index = array.get(hi_sess_arr, 1)
        int third_last_hi_session_index = array.get(hi_sess_arr, 2)
        int fourth_last_hi_session_index = array.get(hi_sess_arr, 3)
        int fifth_last_hi_session_index = array.get(hi_sess_arr, 4)

        int age_1h = not na(last_hi_session_index) ? f_session_age(last_hi_session_index) : na
        int age_2h = not na(second_last_hi_session_index) ? f_session_age(second_last_hi_session_index) : na
        int age_3h = not na(third_last_hi_session_index) ? f_session_age(third_last_hi_session_index) : na
        int age_4h = not na(fourth_last_hi_session_index) ? f_session_age(fourth_last_hi_session_index) : na
        int age_5h = not na(fifth_last_hi_session_index) ? f_session_age(fifth_last_hi_session_index) : na

        int last_lo_session_index = array.get(lo_sess_arr, 0)
        int second_last_lo_session_index = array.get(lo_sess_arr, 1)
        int third_last_lo_session_index = array.get(lo_sess_arr, 2)
        int fourth_last_lo_session_index = array.get(lo_sess_arr, 3)
        int fifth_last_lo_session_index = array.get(lo_sess_arr, 4)

        int age_1l = not na(last_lo_session_index) ? f_session_age(last_lo_session_index) : na
        int age_2l = not na(second_last_lo_session_index) ? f_session_age(second_last_lo_session_index) : na
        int age_3l = not na(third_last_lo_session_index) ? f_session_age(third_last_lo_session_index) : na
        int age_4l = not na(fourth_last_lo_session_index) ? f_session_age(fourth_last_lo_session_index) : na
        int age_5l = not na(fifth_last_lo_session_index) ? f_session_age(fifth_last_lo_session_index) : na
        
        // Status indicators (proximity warnings)
        string status_1h = (enable_proximity_alerts and not na(delta_1h) and math.abs(delta_1h) <= proximity_threshold) ? "⚠️" : ""
        string status_2h = (enable_proximity_alerts and not na(delta_2h) and math.abs(delta_2h) <= proximity_threshold) ? "⚠️" : ""
        string status_3h = (enable_proximity_alerts and not na(delta_3h) and math.abs(delta_3h) <= proximity_threshold) ? "⚠️" : ""
        string status_4h = (enable_proximity_alerts and not na(delta_4h) and math.abs(delta_4h) <= proximity_threshold) ? "⚠️" : ""
        string status_5h = (enable_proximity_alerts and not na(delta_5h) and math.abs(delta_5h) <= proximity_threshold) ? "⚠️" : ""

        string status_1l = (enable_proximity_alerts and not na(delta_1l) and math.abs(delta_1l) <= proximity_threshold) ? "⚠️" : ""
        string status_2l = (enable_proximity_alerts and not na(delta_2l) and math.abs(delta_2l) <= proximity_threshold) ? "⚠️" : ""
        string status_3l = (enable_proximity_alerts and not na(delta_3l) and math.abs(delta_3l) <= proximity_threshold) ? "⚠️" : ""
        string status_4l = (enable_proximity_alerts and not na(delta_4l) and math.abs(delta_4l) <= proximity_threshold) ? "⚠️" : ""
        string status_5l = (enable_proximity_alerts and not na(delta_5l) and math.abs(delta_5l) <= proximity_threshold) ? "⚠️" : ""
        
        // Direction and Strat
        bool is_bullish = TF_Close_curr > TF_Close_prev
        string direction_indicator = is_bullish ? "🟢" : "🔴"
        string current_strat = f_calculate_strat()
        color strat_color = current_strat == "2U" ? color.lime : current_strat == "2D" ? color.red : current_strat == "3" ? color.yellow : color.gray
        
        // Statistics
        int total_hi_active = f_count_active_levels(hi_active)
        int total_lo_active = f_count_active_levels(lo_active)
        
        string timeframe_display = f_format_timeframe(interval)
        string fmt_price = syminfo.mintick <= 0.00001 ? "#.######" : "#.##"
        
        table.set_position(table_id, f_get_position(table_location))
        text_size_var = f_get_size(table_text_size)
        default_text_clr = table_text_color
        
        // Headers
        int col_count = show_percentage ? 5 : 4
        table.cell(table_id, 0, 0, "Level", text_size = text_size_var, bgcolor = color.gray, text_color = default_text_clr)
        table.cell(table_id, 1, 0, "$", text_size = text_size_var, bgcolor = color.gray, text_color = default_text_clr)
        table.cell(table_id, 2, 0, "Pts", text_size = text_size_var, bgcolor = color.gray, text_color = default_text_clr)
        table.cell(table_id, 3, 0, "Age", text_size = text_size_var, bgcolor = color.gray, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 0, "%", text_size = text_size_var, bgcolor = color.gray, text_color = default_text_clr)

        // HIGH LEVELS (Rows 1-5)
        table.cell(table_id, 0, 1, "↑↑↑↑↑" + status_5h, text_size = text_size_var, bgcolor = f_get_high_color(age_5h), text_color = color.white)
        table.cell(table_id, 1, 1, f_format_level(fifth_last_hi_lvl, fmt_price), text_size = text_size_var, text_color = default_text_clr)
        table.cell(table_id, 2, 1, f_format_delta(delta_5h, true), text_size = text_size_var, text_color = color.red)
        table.cell(table_id, 3, 1, na(age_5h) ? "N/A" : str.tostring(age_5h), text_size = text_size_var, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 1, f_format_percentage(fifth_last_hi_lvl, close), text_size = text_size_var, text_color = color.red)

        table.cell(table_id, 0, 2, "↑↑↑↑" + status_4h, text_size = text_size_var, bgcolor = f_get_high_color(age_4h), text_color = color.white)
        table.cell(table_id, 1, 2, f_format_level(fourth_last_hi_lvl, fmt_price), text_size = text_size_var, text_color = default_text_clr)
        table.cell(table_id, 2, 2, f_format_delta(delta_4h, true), text_size = text_size_var, text_color = color.red)
        table.cell(table_id, 3, 2, na(age_4h) ? "N/A" : str.tostring(age_4h), text_size = text_size_var, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 2, f_format_percentage(fourth_last_hi_lvl, close), text_size = text_size_var, text_color = color.red)
        
        table.cell(table_id, 0, 3, "↑↑↑" + status_3h, text_size = text_size_var, bgcolor = f_get_high_color(age_3h), text_color = color.white)
        table.cell(table_id, 1, 3, f_format_level(third_last_hi_lvl, fmt_price), text_size = text_size_var, text_color = default_text_clr)
        table.cell(table_id, 2, 3, f_format_delta(delta_3h, true), text_size = text_size_var, text_color = color.red)
        table.cell(table_id, 3, 3, na(age_3h) ? "N/A" : str.tostring(age_3h), text_size = text_size_var, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 3, f_format_percentage(third_last_hi_lvl, close), text_size = text_size_var, text_color = color.red)

        table.cell(table_id, 0, 4, "↑↑" + status_2h, text_size = text_size_var, bgcolor = f_get_high_color(age_2h), text_color = color.white)
        table.cell(table_id, 1, 4, f_format_level(second_last_hi_lvl, fmt_price), text_size = text_size_var, text_color = default_text_clr)
        table.cell(table_id, 2, 4, f_format_delta(delta_2h, true), text_size = text_size_var, text_color = color.red)
        table.cell(table_id, 3, 4, na(age_2h) ? "N/A" : str.tostring(age_2h), text_size = text_size_var, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 4, f_format_percentage(second_last_hi_lvl, close), text_size = text_size_var, text_color = color.red)

        table.cell(table_id, 0, 5, "↑" + status_1h, text_size = text_size_var, bgcolor = f_get_high_color(age_1h), text_color = color.white)
        table.cell(table_id, 1, 5, f_format_level(last_hi_lvl, fmt_price), text_size = text_size_var, text_color = default_text_clr)
        table.cell(table_id, 2, 5, f_format_delta(delta_1h, true), text_size = text_size_var, text_color = color.red)
        table.cell(table_id, 3, 5, na(age_1h) ? "N/A" : str.tostring(age_1h), text_size = text_size_var, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 5, f_format_percentage(last_hi_lvl, close), text_size = text_size_var, text_color = color.red)
        
        // Separator Row
        color tunnel_bg_color = color.rgb(120, 123, 134, 67)
        table.cell(table_id, 0, 6, timeframe_display, text_size = text_size_var, bgcolor = tunnel_bg_color, text_color = default_text_clr)
        table.cell(table_id, 1, 6, direction_indicator, text_size = text_size_var, bgcolor = tunnel_bg_color)
        table.cell(table_id, 2, 6, net_delta_str, text_size = text_size_var, bgcolor = tunnel_bg_color, text_color = color.yellow)
        table.cell(table_id, 3, 6, current_strat, text_size = text_size_var, bgcolor = tunnel_bg_color, text_color = strat_color)
        if show_percentage
            table.cell(table_id, 4, 6, "", text_size = text_size_var, bgcolor = tunnel_bg_color)

        // LOW LEVELS (Rows 7-11)
        table.cell(table_id, 0, 7, "↓" + status_1l, text_size = text_size_var, bgcolor = f_get_low_color(age_1l), text_color = color.white)
        table.cell(table_id, 1, 7, f_format_level(last_lo_lvl, fmt_price), text_size = text_size_var, text_color = default_text_clr)
        table.cell(table_id, 2, 7, f_format_delta(delta_1l, false), text_size = text_size_var, text_color = color.green)
        table.cell(table_id, 3, 7, na(age_1l) ? "N/A" : str.tostring(age_1l), text_size = text_size_var, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 7, f_format_percentage(last_lo_lvl, close), text_size = text_size_var, text_color = color.green)
        
        table.cell(table_id, 0, 8, "↓↓" + status_2l, text_size = text_size_var, bgcolor = f_get_low_color(age_2l), text_color = color.white)
        table.cell(table_id, 1, 8, f_format_level(second_last_lo_lvl, fmt_price), text_size = text_size_var, text_color = default_text_clr)
        table.cell(table_id, 2, 8, f_format_delta(delta_2l, false), text_size = text_size_var, text_color = color.green)
        table.cell(table_id, 3, 8, na(age_2l) ? "N/A" : str.tostring(age_2l), text_size = text_size_var, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 8, f_format_percentage(second_last_lo_lvl, close), text_size = text_size_var, text_color = color.green)

        table.cell(table_id, 0, 9, "↓↓↓" + status_3l, text_size = text_size_var, bgcolor = f_get_low_color(age_3l), text_color = color.white)
        table.cell(table_id, 1, 9, f_format_level(third_last_lo_lvl, fmt_price), text_size = text_size_var, text_color = default_text_clr)
        table.cell(table_id, 2, 9, f_format_delta(delta_3l, false), text_size = text_size_var, text_color = color.green)
        table.cell(table_id, 3, 9, na(age_3l) ? "N/A" : str.tostring(age_3l), text_size = text_size_var, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 9, f_format_percentage(third_last_lo_lvl, close), text_size = text_size_var, text_color = color.green)

        table.cell(table_id, 0, 10, "↓↓↓↓" + status_4l, text_size = text_size_var, bgcolor = f_get_low_color(age_4l), text_color = color.white)
        table.cell(table_id, 1, 10, f_format_level(fourth_last_lo_lvl, fmt_price), text_size = text_size_var, text_color = default_text_clr)
        table.cell(table_id, 2, 10, f_format_delta(delta_4l, false), text_size = text_size_var, text_color = color.green)
        table.cell(table_id, 3, 10, na(age_4l) ? "N/A" : str.tostring(age_4l), text_size = text_size_var, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 10, f_format_percentage(fourth_last_lo_lvl, close), text_size = text_size_var, text_color = color.green)

        table.cell(table_id, 0, 11, "↓↓↓↓↓" + status_5l, text_size = text_size_var, bgcolor = f_get_low_color(age_5l), text_color = color.white)
        table.cell(table_id, 1, 11, f_format_level(fifth_last_lo_lvl, fmt_price), text_size = text_size_var, text_color = default_text_clr)
        table.cell(table_id, 2, 11, f_format_delta(delta_5l, false), text_size = text_size_var, text_color = color.green)
        table.cell(table_id, 3, 11, na(age_5l) ? "N/A" : str.tostring(age_5l), text_size = text_size_var, text_color = default_text_clr)
        if show_percentage
            table.cell(table_id, 4, 11, f_format_percentage(fifth_last_lo_lvl, close), text_size = text_size_var, text_color = color.green)
    
        // Statistics Row (optional)
        if show_stats_row
            string stats_text = "📊 " + str.tostring(total_hi_active) + "↑ / " + str.tostring(total_lo_active) + "↓"
            table.cell(table_id, 0, 12, stats_text, text_size = text_size_var, bgcolor = color.rgb(30, 30, 30, 50), text_color = color.yellow)
            if show_percentage
                table.merge_cells(table_id, 0, 12, 4, 12)
            else
                table.merge_cells(table_id, 0, 12, 3, 12)
