//@version=5
indicator("AI SWING Algo (Non-Repaint Edition) with SL/TP", overlay = true, max_labels_count = 500, max_lines_count = 500, max_boxes_count = 500)

//------------------------------------------------------------------------------
// INPUTS
//------------------------------------------------------------------------------
groupMode = "Mode"
atr_p = input.int(53, "ATR period", minval=1, group = groupMode)
mult = input.float(2.8,"ATR Multiplier", minval=0.1, step=0.1, group = groupMode)
mode = input.string("Type A", "Signal mode", options = ["Type A", "Type B"], group = groupMode)

groupSrc = "Source"
use_ema_smooth= input.string("No", "Smooth source with EMA?", options = ["Yes","No"], group = groupSrc)
src_ema_period= input.int(3, "EMA Smoother period", minval=1, group = groupSrc)

groupSig = "Signal’s Addon"
signals_view = input.string("All", "Signals to show", options = ["All","Buy/Sell","Strong","None"], group = groupSig)
signals_shape = input.string("Labels","Signal's shape", options = ["Labels","Arrows"], group = groupSig)
buy_col = input.color(#045766, "Buy colour", inline="BS", group = groupSig)
sell_col = input.color(color.new(#7e0978, 0), "Sell colour", inline="BS", group = groupSig)
color_bars = input.bool(true, "Color bars?", group = groupSig)

groupT3 = "T3 Levels (Non-repaint)"
t3hot = input.float(0.7, "K4 Hot", group=groupT3)
t3type = input.string("K4 New","K4 Type", options=["K4 New","K4 Original"], group=groupT3)

//------------------------------------------------------------------------------
// SOURCE
//------------------------------------------------------------------------------
src_raw = close
src = use_ema_smooth == "Yes" ? ta.ema(src_raw, src_ema_period) : src_raw

//------------------------------------------------------------------------------
// NON-REPAINT M/LINES (epsilon bands)
//------------------------------------------------------------------------------
p = 200
h = ta.highest(src, p)
l = ta.lowest(src, p)
d = h - l
var float m = na
atr_prev = ta.atr(atr_p)[1]
epsilon = mult * atr_prev
m_prev = nz(m[1], (h + l)/2)
upper_prev = m_prev + epsilon
lower_prev = m_prev - epsilon

change_up_raw = mode == "Type B" ? ta.cross(src, upper_prev) : ta.crossover(src, upper_prev)
change_down_raw = mode == "Type B" ? ta.cross(src, lower_prev) : ta.crossunder(src, lower_prev)
change_up = change_up_raw and barstate.isconfirmed
change_down = change_down_raw and barstate.isconfirmed
m := change_up ? m_prev + epsilon : change_down ? m_prev - epsilon : m_prev

//------------------------------------------------------------------------------
// LAST SIGNAL TRACKER (to enforce alternate buy/sell)
//------------------------------------------------------------------------------
var int lastSignal = 0 // 1 = Buy, -1 = Sell, 0 = None

//------------------------------------------------------------------------------
// T3 (non-repaint)
//------------------------------------------------------------------------------
greencolor = color.rgb(150, 223, 132, 29)
redcolor = color.rgb(241, 174, 187, 25)

_iT3(src1, per, hot, clean)=>
    a = hot
    alpha = clean == "K4 New" ? 2.0 / (2.0 + (per - 1.0) / 2.0) : 2.0 / (1.0 + per)
    _t30 = 0.0, _t31 = 0.0, _t32 = 0.0, _t33 = 0.0, _t34 = 0.0, _t35 = 0.0
    _t30 := nz(_t30[1]) + alpha * (src1 - nz(_t30[1]))
    _t31 := nz(_t31[1]) + alpha * (_t30 - nz(_t31[1]))
    _t32 := nz(_t32[1]) + alpha * (_t31 - nz(_t32[1]))
    _t33 := nz(_t33[1]) + alpha * (_t32 - nz(_t33[1]))
    _t34 := nz(_t34[1]) + alpha * (_t33 - nz(_t34[1]))
    _t35 := nz(_t35[1]) + alpha * (_t34 - nz(_t35[1]))
    [ _t30, _t35 ]

perT3 = 14
[lev0, lev5] = _iT3(src, perT3, t3hot, t3type)
colorout = lev0 > lev5 ? greencolor : redcolor
pl0 = plot(lev0, "level 0", color=color.rgb(70, 65, 65, 10), linewidth=2)
pl5 = plot(lev5, "level 5", color=color.rgb(83, 90, 90, 7), linewidth=2)
fill(pl0,pl5,color=color.new(colorout,80))
barcolor(color_bars ? colorout : na)

//------------------------------------------------------------------------------
// BUY/SELL SIGNALS (alternate enforced)
//------------------------------------------------------------------------------
goLong = ta.crossover(lev0, lev5) and barstate.isconfirmed
goShort = ta.crossunder(lev0, lev5) and barstate.isconfirmed

validBuy = goLong and lastSignal != 1
validSell = goShort and lastSignal != -1

if validBuy
    lastSignal := 1
if validSell
    lastSignal := -1

//------------------------------------------------------------------------------
// ENTRY, SL, TP CALCULATIONS
//------------------------------------------------------------------------------
var float entryPrice = na
var float slPrice = na
var float tp1 = na
var float tp2 = na
var float tp3 = na
var float tp4 = na
var float tp5 = na

tpMult1 = 1.5
tpMult2 = 3
tpMult3 = 4.5
tpMult4 = 6
tpMult5 = 7.5

if validBuy
    entryPrice := close
    slPrice := entryPrice - atr_prev * tpMult1 * 1.5
    tp1 := entryPrice + atr_prev * tpMult1
    tp2 := entryPrice + atr_prev * tpMult2
    tp3 := entryPrice + atr_prev * tpMult3
    tp4 := entryPrice + atr_prev * tpMult4
    tp5 := entryPrice + atr_prev * tpMult5

if validSell
    entryPrice := close
    slPrice := entryPrice + atr_prev * tpMult1 * 1.5
    tp1 := entryPrice - atr_prev * tpMult1
    tp2 := entryPrice - atr_prev * tpMult2
    tp3 := entryPrice - atr_prev * tpMult3
    tp4 := entryPrice - atr_prev * tpMult4
    tp5 := entryPrice - atr_prev * tpMult5

//------------------------------------------------------------------------------
// PLOTS
//------------------------------------------------------------------------------
plotshape(signals_shape == "Labels" and validBuy , "BUY", color=buy_col, style=shape.labelup, location=location.belowbar, size=size.small, text="BUY", textcolor=color.white)
plotshape(signals_shape == "Labels" and validSell, "SELL", color=sell_col, style=shape.labeldown, size=size.small, text="SELL", textcolor=color.white)
plotshape(signals_shape == "Arrows" and validBuy , "Buy", color=buy_col, style=shape.triangleup, location=location.belowbar, size=size.normal)
plotshape(signals_shape == "Arrows" and validSell, "Sell", color=sell_col, style=shape.triangledown, size=size.normal)

plot(entryPrice, title="Entry", color=color.blue, linewidth=3)
plot(slPrice, title="Stop Loss", color=color.red, linewidth=3)
plot(tp1, title="TP1", color=color.green, linewidth=1)
plot(tp2, title="TP2", color=color.green, linewidth=1)
plot(tp3, title="TP3", color=color.green, linewidth=1)
plot(tp4, title="TP4", color=color.green, linewidth=1)
plot(tp5, title="TP5", color=color.green, linewidth=1)

//------------------------------------------------------------------------------
// ALERTS
//------------------------------------------------------------------------------
alertcondition(validBuy, "SWING KILLER BUY (NR)", "Confirmed non-repaint BUY (alternate)")
alertcondition(validSell, "SWING KILLER SELL (NR)", "Confirmed non-repaint SELL (alternate)")
