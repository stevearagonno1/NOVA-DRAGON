strategy('Pearson SL/TP', overlay=true, max_lines_count=500, max_labels_count=500)

// ═══════════════════════════════════════════════════════════════════

group1 = '⚙️ Pearson Settings'
pearsonLength = input.int(20, 'Pearson Length', minval=5, maxval=500, group=group1, tooltip='Number of bars to calculate Pearson correlation')
priceSource = input.source(close, 'Price Source', group=group1)
smoothing = input.int(1, 'Smoothing', minval=1, maxval=50, group=group1, tooltip='SMA smoothing of Pearson values')

// Signal thresholds
upperThreshold = input.float(0.7, 'Overbought Threshold', minval=0.1, maxval=1.0, step=0.05, group=group1, tooltip='Threshold for SHORT signals')
lowerThreshold = input.float(-0.7, 'Oversold Threshold', minval=-1.0, maxval=-0.1, step=0.05, group=group1, tooltip='Threshold for LONG signals')

// ═══════════════════════════════════════════════════════════════════
// 💰 RISK MANAGEMENT
// ═══════════════════════════════════════════════════════════════════

group2 = '💰 Risk Management'
atrLength = input.int(14, 'ATR Length', minval=5, maxval=100, group=group2)
slMultiplier = input.float(1.5, 'Stop Loss Multiplier (x ATR)', minval=0.5, maxval=5.0, step=0.1, group=group2)
tpMultiplier1 = input.float(2.0, 'Take Profit 1 Multiplier (x ATR)', minval=0.5, maxval=10.0, step=0.5, group=group2)
tpMultiplier2 = input.float(3.5, 'Take Profit 2 Multiplier (x ATR)', minval=0.5, maxval=10.0, step=0.5, group=group2)
riskRewardMin = input.float(1.5, 'Minimum Risk/Reward', minval=1.0, maxval=5.0, step=0.1, group=group2)

// ═══════════════════════════════════════════════════════════════════
// 🎨 VISUAL SETTINGS
// ═══════════════════════════════════════════════════════════════════

group3 = '🎨 Display Settings'
showDashboard = input.bool(true, 'Show Dashboard', group=group3)
showZones = input.bool(true, 'Show Pearson Zones', group=group3)
showLevels = input.bool(true, 'Show SL/TP Levels', group=group3)
showLabels = input.bool(true, 'Show Price Labels', group=group3)

// Colors
colorLong = input.color(color.new(color.green, 0), 'LONG Color', group=group3)
colorShort = input.color(color.new(color.red, 0), 'SHORT Color', group=group3)
colorSL = input.color(color.new(color.red, 0), 'Stop Loss Color', group=group3)
colorTP1 = input.color(color.new(color.blue, 0), 'Take Profit 1 Color', group=group3)
colorTP2 = input.color(color.new(color.orange, 0), 'Take Profit 2 Color', group=group3)

// ═══════════════════════════════════════════════════════════════════
// 🔢 TECHNICAL CALCULATIONS
// ═══════════════════════════════════════════════════════════════════

// Pearson calculation function
calculatePearson(src, length) =>
    float sumX = 0.0
    float sumY = 0.0
    float sumXX = 0.0
    float sumYY = 0.0
    float sumXY = 0.0
    
    for i = 0 to length - 1
        price = src[i]
        timeValue = float(i + 1)
        sumX += timeValue
        sumY += price
        sumXX += timeValue * timeValue
        sumYY += price * price
        sumXY += timeValue * price
    
    n = float(length)
    num = n * sumXY - sumX * sumY
    den = math.sqrt((n * sumXX - sumX * sumX) * (n * sumYY - sumY * sumY))
    
    den != 0 ? num / den : 0

// Pearson calculations
pearsonRaw = calculatePearson(priceSource, pearsonLength)
pearsonSmooth = ta.sma(pearsonRaw, smoothing)

// ATR for dynamic SL/TP
atr = ta.atr(atrLength)

// ═══════════════════════════════════════════════════════════════════
// 📈 SIGNAL STRENGTH CALCULATION
// ═══════════════════════════════════════════════════════════════════

// Signal strength criteria
volCriteria = volume > ta.sma(volume, 20) ? 1 : 0
trendCriteria = ta.ema(close, 50) > ta.ema(close, 200) ? 1 : -1
momentumCriteria = math.abs(pearsonSmooth) > 0.8 ? 1 : 0
distanceCriteria = math.abs(pearsonSmooth - pearsonSmooth[1]) > 0.05 ? 1 : 0

// Total strength (0-4)
signalStrength = volCriteria + momentumCriteria + distanceCriteria + math.abs(trendCriteria)

// Signal quality
getSignalQuality(strength) =>
    switch strength
        4 => '🔥 VERY STRONG'
        3 => '⚡ STRONG'
        2 => '📊 MEDIUM'
        1 => '⚠️ WEAK'
        => '❌ INVALID'

// ═══════════════════════════════════════════════════════════════════
// 🎯 SIGNAL DETECTION
// ═══════════════════════════════════════════════════════════════════

// Base signals
longSignal = ta.crossunder(pearsonSmooth, lowerThreshold)
shortSignal = ta.crossover(pearsonSmooth, upperThreshold)

// Divergence confirmation
bullishDiv = pearsonSmooth > pearsonSmooth[5] and close < close[5]
bearishDiv = pearsonSmooth < pearsonSmooth[5] and close > close[5]

// Confirmed signals
confirmedLong = longSignal and signalStrength >= 2
confirmedShort = shortSignal and signalStrength >= 2

// ═══════════════════════════════════════════════════════════════════
// 💵 TRADING LEVEL CALCULATIONS
// ═══════════════════════════════════════════════════════════════════

var float entryPrice = na
var float stopLoss = na
var float takeProfit1 = na
var float takeProfit2 = na
var string currentDirection = na
var int signalStrengthValue = na
var float riskReward1 = na
var float riskReward2 = na

if confirmedLong
    entryPrice := close
    stopLoss := close - (atr * slMultiplier)
    takeProfit1 := close + (atr * tpMultiplier1)
    takeProfit2 := close + (atr * tpMultiplier2)
    currentDirection := 'LONG'
    signalStrengthValue := signalStrength
    riskReward1 := (takeProfit1 - entryPrice) / (entryPrice - stopLoss)
    riskReward2 := (takeProfit2 - entryPrice) / (entryPrice - stopLoss)

if confirmedShort
    entryPrice := close
    stopLoss := close + (atr * slMultiplier)
    takeProfit1 := close - (atr * tpMultiplier1)
    takeProfit2 := close - (atr * tpMultiplier2)
    currentDirection := 'SHORT'
    signalStrengthValue := signalStrength
    riskReward1 := (entryPrice - takeProfit1) / (stopLoss - entryPrice)
    riskReward2 := (entryPrice - takeProfit2) / (stopLoss - entryPrice)

// ═══════════════════════════════════════════════════════════════════

bgcolor(showZones and pearsonSmooth < lowerThreshold ? color.new(colorLong, 98) : na, title='Oversold Zone')

// Note: hline threshold lines removed to avoid cluttering the chart
// Background zones are sufficient to identify overbought/oversold areas

// Signal arrows (without text)
plotshape(confirmedLong, style=shape.triangleup, location=location.belowbar,

plotshape(confirmedShort, style=shape.triangledown, location=location.abovebar, 
          color=colorShort, size=size.tiny, title='SHORT Signal')

// ═══════════════════════════════════════════════════════════════════
// 🤖 TRADE EXECUTION
// ═══════════════════════════════════════════════════════════════════

if confirmedLong
    if strategy.position_size < 0
        strategy.close_all(comment='Reverse to Long')
    strategy.entry(id='Long', direction=strategy.long, comment='Pearson Long')

if confirmedShort
    if strategy.position_size > 0
        strategy.close_all(comment='Reverse to Short')
    strategy.entry(id='Short', direction=strategy.short, comment='Pearson Short')

longActive = strategy.position_size > 0
shortActive = strategy.position_size < 0

if longActive and not na(stopLoss) and not na(takeProfit1) and not na(takeProfit2)
    strategy.exit(id='Long TP1', from_entry='Long', stop=stopLoss, limit=takeProfit1, qty_percent=50, comment='Long TP1')
    strategy.exit(id='Long TP2', from_entry='Long', stop=stopLoss, limit=takeProfit2, qty_percent=50, comment='Long TP2')

if shortActive and not na(stopLoss) and not na(takeProfit1) and not na(takeProfit2)
    strategy.exit(id='Short TP1', from_entry='Short', stop=stopLoss, limit=takeProfit1, qty_percent=50, comment='Short TP1')
    strategy.exit(id='Short TP2', from_entry='Short', stop=stopLoss, limit=takeProfit2, qty_percent=50, comment='Short TP2')
