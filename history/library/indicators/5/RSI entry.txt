//@version=6
indicator("RSI Entry Engine", shorttitle="RSI Entry Engine", overlay=false, max_labels_count=500, max_boxes_count=500, max_lines_count=500)

// ============================================================================
// OVERVIEW
// This script combines:
// 1) A smoothed RSI engine
// 2) Extreme-zone reclaim signals
// 3) Optional trade projection boxes on the main chart
// 4) A compact status panel
//
// Notes:
// - Signals are confirmed only on bar close.
// - If TP and SL are both touched on the same bar, the selected priority is used.
// - This script is an analytical tool, not an automated execution system.
// ============================================================================

// ============================================================================
// INPUTS
// ============================================================================
groupRsi = "RSI Engine"
rsiSource   = input.source(close, "RSI Source", group=groupRsi)
rsiLength   = input.int(14, "RSI Length", minval=1, group=groupRsi)
smoothMode  = input.string("EMA", "RSI Smoothing", options=["None", "EMA", "SMA", "RMA"], group=groupRsi)
smoothLen   = input.int(5, "Smoothing Length", minval=1, group=groupRsi)
signalLen   = input.int(9, "Signal Length", minval=1, group=groupRsi)
signalMode  = input.string("EMA", "Signal Smoothing", options=["EMA", "SMA", "RMA"], group=groupRsi)

groupLevels = "RSI Levels"
overbought  = input.int(70, "Overbought", minval=50, maxval=100, group=groupLevels)
oversold    = input.int(30, "Oversold", minval=0, maxval=50, group=groupLevels)
extremeHigh = input.int(80, "Extreme High", minval=60, maxval=100, group=groupLevels)
extremeLow  = input.int(20, "Extreme Low", minval=0, maxval=40, group=groupLevels)

groupTrade = "Trade Projection"
showTradeBoxes  = input.bool(true, "Show TP / SL Boxes On Main Chart", group=groupTrade)
entryMode       = input.string("Close", "Entry Price", options=["Close", "Open", "HLC3"], group=groupTrade)
slMode          = input.string("ATR", "Stop Loss Mode", options=["Signal Candle", "ATR", "Percent"], group=groupTrade)
rrTarget        = input.float(5.0, "Risk / Reward", minval=0.1, step=0.1, group=groupTrade)
atrLen          = input.int(14, "ATR Length", minval=1, group=groupTrade)
atrMult         = input.float(1.0, "ATR Multiplier", minval=0.1, step=0.1, group=groupTrade)
percentSL       = input.float(0.50, "Percent Stop Loss", minval=0.01, step=0.01, group=groupTrade)
sameBarPriority = input.string("SL", "Same-Bar TP / SL Priority", options=["SL", "TP"], group=groupTrade)
maxStoredTrades = input.int(20, "Max Stored Trade Projections", minval=1, maxval=100, group=groupTrade)

groupVisual = "Visual Settings"
showBgExtreme = input.bool(true, "Highlight Extreme Background", group=groupVisual)
showPanel     = input.bool(true, "Show Status Panel", group=groupVisual)
panelPosition = input.string("Top Right", "Panel Position", options=["Top Right", "Top Left", "Bottom Right", "Bottom Left"], group=groupVisual)
panelTextSize = input.string("Small", "Panel Text Size", options=["Tiny", "Small", "Normal"], group=groupVisual)

groupAlerts = "Alerts"
enableCrossAlerts  = input.bool(true, "Enable RSI / Signal Cross Alerts", group=groupAlerts)
enableZoneAlerts   = input.bool(true, "Enable Overbought / Oversold Reclaim Alerts", group=groupAlerts)
enableCenterAlerts = input.bool(true, "Enable Centerline Alerts", group=groupAlerts)
enableSignalAlerts = input.bool(true, "Enable Entry Signal Alerts", group=groupAlerts)
enableTradeAlerts  = input.bool(true, "Enable TP / SL Hit Alerts", group=groupAlerts)

// ============================================================================
// COLORS
// ============================================================================
clrBg       = color.rgb(13, 17, 23)
clrHeader   = color.rgb(31, 41, 55)
clrBorder   = color.rgb(48, 54, 61)
clrText     = color.rgb(230, 237, 243)
clrMuted    = color.rgb(139, 148, 158)
clrBlue     = color.rgb(88, 166, 255)
clrGreen    = color.rgb(34, 197, 94)
clrRed      = color.rgb(239, 68, 68)
clrCardA    = color.rgb(22, 27, 34)
clrCardB    = color.rgb(18, 22, 28)
clrLineDark = color.rgb(20, 20, 20)

// ============================================================================
// HELPERS
// ============================================================================
getTablePosition(_pos) =>
    switch _pos
        "Top Left"     => position.top_left
        "Bottom Right" => position.bottom_right
        "Bottom Left"  => position.bottom_left
        => position.top_right

getTableTextSize(_size) =>
    switch _size
        "Tiny"   => size.tiny
        "Normal" => size.normal
        => size.small

ma(_src, _len, _mode) =>
    _mode == "EMA" ? ta.ema(_src, _len) :
     _mode == "SMA" ? ta.sma(_src, _len) :
     _mode == "RMA" ? ta.rma(_src, _len) :
     _src

getEntryPrice(_mode) =>
    _mode == "Open" ? open : _mode == "HLC3" ? hlc3 : close

// ============================================================================
// RSI ENGINE
// ============================================================================
rsiRaw    = ta.rsi(rsiSource, rsiLength)
rsiSmooth = smoothMode == "None" ? rsiRaw : ma(rsiRaw, smoothLen, smoothMode)
signal    = ma(rsiSmooth, signalLen, signalMode)
spread    = rsiSmooth - signal
delta     = rsiSmooth - rsiSmooth[1]
atrValue  = ta.atr(atrLen)

isExtremeHigh = rsiSmooth >= extremeHigh
isOverbought  = rsiSmooth >= overbought
isBullish     = rsiSmooth > 50 and rsiSmooth < overbought
isNeutral     = rsiSmooth >= 45 and rsiSmooth <= 55
isBearish     = rsiSmooth < 50 and rsiSmooth > oversold
isOversold    = rsiSmooth <= oversold
isExtremeLow  = rsiSmooth <= extremeLow

rsiColor = isExtremeHigh ? clrRed : isExtremeLow ? clrGreen : clrBlue

// ============================================================================
// LEVELS
// ============================================================================
h100 = hline(100, "100", color=color.new(clrMuted, 100))
h0   = hline(0, "0", color=color.new(clrMuted, 100))
h50  = hline(50, "Center", color=color.new(clrMuted, 60), linestyle=hline.style_dotted)
hOB  = hline(overbought, "Overbought", color=color.new(clrRed, 30), linestyle=hline.style_dashed)
hOS  = hline(oversold, "Oversold", color=color.new(clrGreen, 30), linestyle=hline.style_dashed)
hEH  = hline(extremeHigh, "Extreme High", color=color.new(clrRed, 55), linestyle=hline.style_dotted)
hEL  = hline(extremeLow, "Extreme Low", color=color.new(clrGreen, 55), linestyle=hline.style_dotted)

fill(hOB, h100, color=color.new(clrRed, 70), title="Overbought Zone")
fill(hOS, h0, color=color.new(clrGreen, 70), title="Oversold Zone")

bgcolor(showBgExtreme and isExtremeHigh ? color.new(clrRed, 80) : na)
bgcolor(showBgExtreme and isExtremeLow ? color.new(clrGreen, 80) : na)

// ============================================================================
// RSI PLOT
// ============================================================================
plot(rsiSmooth, "RSI", color=rsiColor, linewidth=3)
plot(signal, "Signal", color=color.new(clrText, 35), linewidth=2)

// ============================================================================
// SIGNAL LOGIC
// Main entry model:
// - Long signal when RSI reclaims up through Extreme Low
// - Short signal when RSI rejects down through Extreme High
// - Signals are confirmed only at bar close
// ============================================================================
upSignal   = ta.crossover(rsiSmooth, extremeLow) and barstate.isconfirmed
downSignal = ta.crossunder(rsiSmooth, extremeHigh) and barstate.isconfirmed

if upSignal
    label.new(
         x=bar_index,
         y=rsiSmooth,
         text="UP",
         xloc=xloc.bar_index,
         yloc=yloc.price,
         style=label.style_label_up,
         color=color.new(clrGreen, 0),
         textcolor=color.black,
         size=size.normal
     )

if downSignal
    label.new(
         x=bar_index,
         y=rsiSmooth,
         text="DOWN",
         xloc=xloc.bar_index,
         yloc=yloc.price,
         style=label.style_label_down,
         color=color.new(clrRed, 0),
         textcolor=color.white,
         size=size.normal
     )

// ============================================================================
// TRADE STORAGE
// ============================================================================
var box[]   tpBoxes    = array.new_box()
var box[]   slBoxes    = array.new_box()
var line[]  entryLines = array.new_line()
var float[] entries    = array.new_float()
var float[] tps        = array.new_float()
var float[] sls        = array.new_float()
var int[]   dirs       = array.new_int()   // 1 = long, -1 = short
var int[]   startBars  = array.new_int()
var bool[]  actives    = array.new_bool()

// ============================================================================
// TRADE CREATION
// ============================================================================
entryBase = getEntryPrice(entryMode)

if showTradeBoxes and upSignal
    longEntry = entryBase
    longSlRaw = slMode == "Signal Candle" ? low : slMode == "ATR" ? longEntry - atrValue * atrMult : longEntry * (1.0 - percentSL / 100.0)
    longSl    = math.min(longSlRaw, longEntry - syminfo.mintick)
    longRisk  = longEntry - longSl

    if longRisk > 0
        longTp = longEntry + longRisk * rrTarget

        tpBox = box.new(
             left=bar_index,
             top=longTp,
             right=bar_index + 1,
             bottom=longEntry,
             xloc=xloc.bar_index,
             border_color=color.new(clrGreen, 0),
             border_width=1,
             bgcolor=color.new(clrGreen, 65),
             force_overlay=true
         )

        slBox = box.new(
             left=bar_index,
             top=longEntry,
             right=bar_index + 1,
             bottom=longSl,
             xloc=xloc.bar_index,
             border_color=color.new(clrRed, 0),
             border_width=1,
             bgcolor=color.new(clrRed, 55),
             force_overlay=true
         )

        eLine = line.new(
             x1=bar_index,
             y1=longEntry,
             x2=bar_index + 1,
             y2=longEntry,
             xloc=xloc.bar_index,
             color=clrLineDark,
             width=2,
             force_overlay=true
         )

        label.new(
             x=bar_index,
             y=longEntry,
             text="BUY",
             xloc=xloc.bar_index,
             yloc=yloc.price,
             style=label.style_label_up,
             color=color.new(clrGreen, 0),
             textcolor=color.black,
             size=size.normal,
             force_overlay=true
         )

        array.push(tpBoxes, tpBox)
        array.push(slBoxes, slBox)
        array.push(entryLines, eLine)
        array.push(entries, longEntry)
        array.push(tps, longTp)
        array.push(sls, longSl)
        array.push(dirs, 1)
        array.push(startBars, bar_index)
        array.push(actives, true)

if showTradeBoxes and downSignal
    shortEntry = entryBase
    shortSlRaw = slMode == "Signal Candle" ? high : slMode == "ATR" ? shortEntry + atrValue * atrMult : shortEntry * (1.0 + percentSL / 100.0)
    shortSl    = math.max(shortSlRaw, shortEntry + syminfo.mintick)
    shortRisk  = shortSl - shortEntry

    if shortRisk > 0
        shortTp = shortEntry - shortRisk * rrTarget

        tpBox = box.new(
             left=bar_index,
             top=shortEntry,
             right=bar_index + 1,
             bottom=shortTp,
             xloc=xloc.bar_index,
             border_color=color.new(clrGreen, 0),
             border_width=1,
             bgcolor=color.new(clrGreen, 65),
             force_overlay=true
         )

        slBox = box.new(
             left=bar_index,
             top=shortSl,
             right=bar_index + 1,
             bottom=shortEntry,
             xloc=xloc.bar_index,
             border_color=color.new(clrRed, 0),
             border_width=1,
             bgcolor=color.new(clrRed, 55),
             force_overlay=true
         )

        eLine = line.new(
             x1=bar_index,
             y1=shortEntry,
             x2=bar_index + 1,
             y2=shortEntry,
             xloc=xloc.bar_index,
             color=clrLineDark,
             width=2,
             force_overlay=true
         )

        label.new(
             x=bar_index,
             y=shortEntry,
             text="SELL",
             xloc=xloc.bar_index,
             yloc=yloc.price,
             style=label.style_label_down,
             color=color.new(clrRed, 0),
             textcolor=color.white,
             size=size.normal,
             force_overlay=true
         )

        array.push(tpBoxes, tpBox)
        array.push(slBoxes, slBox)
        array.push(entryLines, eLine)
        array.push(entries, shortEntry)
        array.push(tps, shortTp)
        array.push(sls, shortSl)
        array.push(dirs, -1)
        array.push(startBars, bar_index)
        array.push(actives, true)

// Limit stored trade objects
while array.size(tpBoxes) > maxStoredTrades
    oldTpBox = array.shift(tpBoxes)
    oldSlBox = array.shift(slBoxes)
    oldELine = array.shift(entryLines)
    array.shift(entries)
    array.shift(tps)
    array.shift(sls)
    array.shift(dirs)
    array.shift(startBars)
    array.shift(actives)
    box.delete(oldTpBox)
    box.delete(oldSlBox)
    line.delete(oldELine)

// ============================================================================
// ACTIVE TRADE UPDATE
// ============================================================================
tradeCount = array.size(tpBoxes)

bool longTpHitBar  = false
bool longSlHitBar  = false
bool shortTpHitBar = false
bool shortSlHitBar = false

if tradeCount > 0
    for i = 0 to tradeCount - 1
        isActive = array.get(actives, i)

        if isActive
            tpBox    = array.get(tpBoxes, i)
            slBox    = array.get(slBoxes, i)
            eLine    = array.get(entryLines, i)
            tpPx     = array.get(tps, i)
            slPx     = array.get(sls, i)
            dir      = array.get(dirs, i)
            startBar = array.get(startBars, i)

            box.set_right(tpBox, bar_index)
            box.set_right(slBox, bar_index)
            line.set_x2(eLine, bar_index)

            tpHit = false
            slHit = false
            canCheck = bar_index > startBar

            if canCheck
                if dir == 1
                    tpHit := high >= tpPx
                    slHit := low <= slPx
                else
                    tpHit := low <= tpPx
                    slHit := high >= slPx

            if tpHit and slHit
                if sameBarPriority == "SL"
                    tpHit := false
                else
                    slHit := false

            if tpHit or slHit
                array.set(actives, i, false)

                if tpHit
                    if dir == 1
                        longTpHitBar := true
                    else
                        shortTpHitBar := true

                    label.new(
                         x=bar_index,
                         y=tpPx,
                         text="TP HIT",
                         xloc=xloc.bar_index,
                         yloc=yloc.price,
                         style=label.style_label_down,
                         color=color.new(clrGreen, 0),
                         textcolor=color.black,
                         size=size.small,
                         force_overlay=true
                     )
                else
                    if dir == 1
                        longSlHitBar := true
                    else
                        shortSlHitBar := true

                    label.new(
                         x=bar_index,
                         y=slPx,
                         text="SL HIT",
                         xloc=xloc.bar_index,
                         yloc=yloc.price,
                         style=label.style_label_up,
                         color=color.new(clrRed, 0),
                         textcolor=color.white,
                         size=size.small,
                         force_overlay=true
                     )
// ============================================================================
// BACKGROUND WATERMARK
// ============================================================================
var table bgWatermark = table.new(position.middle_center, 1, 1)
if barstate.islast
    table.cell(bgWatermark, 0, 0, "join telegram :@mrexpert_ai", text_color=color.new(clrBlue, 1), text_size=size.huge)
// ============================================================================
// PANEL STATUS
// ============================================================================
statusText =
     isExtremeHigh ? "Extreme High" :
     isOverbought  ? "Overbought" :
     isBullish     ? "Bullish" :
     isNeutral     ? "Neutral" :
     isBearish     ? "Bearish" :
     isOversold    ? "Oversold" :
     "Extreme Low"

statusColor =
     isExtremeHigh ? clrRed :
     isOverbought  ? clrRed :
     isBullish     ? clrBlue :
     isNeutral     ? clrText :
     isBearish     ? clrBlue :
     isOversold    ? clrGreen :
     clrGreen

momentumText =
     delta >= 1.5  ? "Strong Up" :
     delta > 0     ? "Up" :
     delta <= -1.5 ? "Strong Down" :
     delta < 0     ? "Down" :
     "Flat"

momentumColor =
     delta >= 1.5  ? clrGreen :
     delta > 0     ? clrBlue :
     delta <= -1.5 ? clrRed :
     delta < 0     ? clrRed :
     clrMuted

crossState = spread > 0 ? "RSI > Signal" : spread < 0 ? "RSI < Signal" : "At Signal"
crossColor = spread > 0 ? clrGreen : spread < 0 ? clrRed : clrMuted

// ============================================================================
// STATUS PANEL
// ============================================================================
panelPos  = getTablePosition(panelPosition)
panelSize = getTableTextSize(panelTextSize)

var table panel = table.new(panelPos, 2, 6, border_width=1, frame_color=clrBorder, border_color=clrBorder)

if barstate.islast
    if showPanel
        table.cell(panel, 0, 0, "RSI PANEL", text_color=clrText, text_size=panelSize, bgcolor=clrHeader)
        table.cell(panel, 1, 0, "VALUE", text_color=clrText, text_size=panelSize, bgcolor=clrHeader)

        table.cell(panel, 0, 1, "RSI", text_color=clrMuted, text_size=panelSize, bgcolor=clrCardA)
        table.cell(panel, 1, 1, str.tostring(rsiSmooth, "#.##"), text_color=rsiColor, text_size=panelSize, bgcolor=clrCardA)

        table.cell(panel, 0, 2, "Signal", text_color=clrMuted, text_size=panelSize, bgcolor=clrCardB)
        table.cell(panel, 1, 2, str.tostring(signal, "#.##"), text_color=clrText, text_size=panelSize, bgcolor=clrCardB)

        table.cell(panel, 0, 3, "Status", text_color=clrMuted, text_size=panelSize, bgcolor=clrCardA)
        table.cell(panel, 1, 3, statusText, text_color=statusColor, text_size=panelSize, bgcolor=clrCardA)

        table.cell(panel, 0, 4, "Momentum", text_color=clrMuted, text_size=panelSize, bgcolor=clrCardB)
        table.cell(panel, 1, 4, momentumText, text_color=momentumColor, text_size=panelSize, bgcolor=clrCardB)

        table.cell(panel, 0, 5, "Structure", text_color=clrMuted, text_size=panelSize, bgcolor=clrCardA)
        table.cell(panel, 1, 5, crossState, text_color=crossColor, text_size=panelSize, bgcolor=clrCardA)
    else
        for row = 0 to 5
            table.cell(panel, 0, row, "", bgcolor=color.new(clrBg, 100))
            table.cell(panel, 1, row, "", bgcolor=color.new(clrBg, 100))

// ============================================================================
// ALERTS
// ============================================================================
bullCross   = ta.crossover(rsiSmooth, signal)
bearCross   = ta.crossunder(rsiSmooth, signal)
bullReclaim = ta.crossover(rsiSmooth, oversold)
bearReject  = ta.crossunder(rsiSmooth, overbought)
centerBull  = ta.crossover(rsiSmooth, 50)
centerBear  = ta.crossunder(rsiSmooth, 50)

alertcondition(enableCrossAlerts and bullCross, title="RSI Bullish Cross", message="RSI crossed above Signal")
alertcondition(enableCrossAlerts and bearCross, title="RSI Bearish Cross", message="RSI crossed below Signal")
alertcondition(enableZoneAlerts and bullReclaim, title="RSI Oversold Reclaim", message="RSI reclaimed above Oversold")
alertcondition(enableZoneAlerts and bearReject, title="RSI Overbought Reject", message="RSI crossed below Overbought")
alertcondition(enableCenterAlerts and centerBull, title="RSI Centerline Bullish", message="RSI crossed above 50")
alertcondition(enableCenterAlerts and centerBear, title="RSI Centerline Bearish", message="RSI crossed below 50")

alertcondition(enableSignalAlerts and upSignal, title="Long Entry Signal", message="Long entry signal confirmed")
alertcondition(enableSignalAlerts and downSignal, title="Short Entry Signal", message="Short entry signal confirmed")

alertcondition(enableTradeAlerts and longTpHitBar, title="Long TP Hit", message="Long projected take-profit reached")
alertcondition(enableTradeAlerts and longSlHitBar, title="Long SL Hit", message="Long projected stop-loss reached")
alertcondition(enableTradeAlerts and shortTpHitBar, title="Short TP Hit", message="Short projected take-profit reached")
alertcondition(enableTradeAlerts and shortSlHitBar, title="Short SL Hit", message="Short projected stop-loss reached")
