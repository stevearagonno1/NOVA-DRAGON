//@version=6
indicator("Volatility Covenant Ribbon [JOAT]", shorttitle="VCR [JOAT]", overlay=true, max_lines_count=160, max_boxes_count=140, max_labels_count=40)

//------------------------------------------------------------------------------
// Volatility Covenant Ribbon [JOAT]
// Measures volatility expansion, compression, and directional persistence using
// ATR percentiles, directional range separation, layered ribbons, state boxes,
// and an optional informational TP/SL scaffold when expansion confirms.
//------------------------------------------------------------------------------

//------------------------------------------------------------------------------
// Inputs
//------------------------------------------------------------------------------
groupEngine = "Volatility Engine"
atrLen = input.int(14, "ATR Length", minval=2, group=groupEngine)
percentileLookback = input.int(160, "Percentile Lookback", minval=30, maxval=400, group=groupEngine)
ribbonBasisLen = input.int(55, "Ribbon Basis Length", minval=5, group=groupEngine)
directionLen = input.int(21, "Directional Range Length", minval=5, group=groupEngine)
innerMult = input.float(1.0, "Inner Multiplier", minval=0.2, step=0.1, group=groupEngine)
outerMult = input.float(1.9, "Outer Multiplier", minval=0.4, step=0.1, group=groupEngine)
stateThreshold = input.float(12.0, "State Threshold", minval=4.0, step=1.0, group=groupEngine)

groupZones = "State Zones"
showExpansionBoxes = input.bool(true, "Show Expansion Boxes", group=groupZones)
showCompressionBoxes = input.bool(true, "Show Compression Boxes", group=groupZones)
storedStateBoxes = input.int(8, "Stored State Boxes", minval=2, maxval=20, group=groupZones)
zoneTransparency = input.int(88, "Zone Transparency", minval=50, maxval=98, group=groupZones)

groupRisk = "Volatility Scaffold"
showScaffold = input.bool(true, "Show Expansion Scaffold", group=groupRisk)
scaffoldAtrMult = input.float(1.6, "Stop ATR Multiplier", minval=0.3, step=0.1, group=groupRisk)
tpOneR = input.float(1.0, "TP1 R", minval=0.4, step=0.1, group=groupRisk)
tpTwoR = input.float(2.0, "TP2 R", minval=0.6, step=0.1, group=groupRisk)
tpThreeR = input.float(3.0, "TP3 R", minval=1.0, step=0.1, group=groupRisk)

groupDisplay = "Visual Design"
showRibbon = input.bool(true, "Show Volatility Ribbon", group=groupDisplay)
showBasis = input.bool(true, "Show Ribbon Basis", group=groupDisplay)
showDashboard = input.bool(true, "Show Dashboard", group=groupDisplay)
showBarBias = input.bool(true, "Color Candles", group=groupDisplay)
showBackdrop = input.bool(true, "Show Volatility Backdrop", group=groupDisplay)
showInnerGradient = input.bool(true, "Show Inner Gradient", group=groupDisplay)

groupPalette = "Dark Palette"
bullCore = input.color(color.rgb(0, 201, 166), "Bull Core", group=groupPalette)
bearCore = input.color(color.rgb(255, 92, 118), "Bear Core", group=groupPalette)
neutralCore = input.color(color.rgb(148, 163, 184), "Neutral Core", group=groupPalette)
accentCore = input.color(color.rgb(255, 204, 92), "Ribbon Accent", group=groupPalette)
panelBg = input.color(color.rgb(11, 18, 32), "Panel Background", group=groupPalette)
panelFrame = input.color(color.rgb(58, 76, 109), "Panel Frame", group=groupPalette)

//------------------------------------------------------------------------------
// Helpers
//------------------------------------------------------------------------------
f_clamp(float value, float minValue, float maxValue) =>
    math.max(minValue, math.min(maxValue, value))

f_percent_rank(series float value, int length) =>
    float count = 0.0
    for i = 0 to length - 1
        count += value >= value[i] ? 1.0 : 0.0
    100.0 * count / length

f_state_color(float score) =>
    score >= stateThreshold ? bullCore :
     score <= -stateThreshold ? bearCore : neutralCore

f_state_text(float score, float percentile) =>
    percentile >= 80 and score >= stateThreshold ? "Expansion Up" :
     percentile >= 80 and score <= -stateThreshold ? "Expansion Down" :
     percentile <= 25 ? "Compression" :
     score >= stateThreshold ? "Firm" :
     score <= -stateThreshold ? "Heavy" : "Balanced"

f_delete_line(line lineRef) =>
    if not na(lineRef)
        line.delete(lineRef)

f_delete_label(label labelRef) =>
    if not na(labelRef)
        label.delete(labelRef)

f_delete_fill(linefill fillRef) =>
    if not na(fillRef)
        linefill.delete(fillRef)

f_trim_boxes(array<box> refArray, int keepCount) =>
    while array.size(refArray) > keepCount
        box stale = array.shift(refArray)
        box.delete(stale)

//------------------------------------------------------------------------------
// Core engine
//------------------------------------------------------------------------------
atrValue = ta.atr(atrLen)
atrPercentile = bar_index >= percentileLookback ? f_percent_rank(atrValue, percentileLookback) : na

basis = ta.ema(close, ribbonBasisLen)
dirBullRange = ta.ema(close > open ? high - low : 0.0, directionLen)
dirBearRange = ta.ema(close < open ? high - low : 0.0, directionLen)
dirImbalance = (dirBullRange - dirBearRange) / math.max(atrValue, syminfo.mintick)
trendBasis = ta.ema(close, 21) - ta.ema(close, 55)
locationScore = (close - basis) / math.max(atrValue, syminfo.mintick) * 16.0
percentileScore = not na(atrPercentile) ? atrPercentile - 50.0 : 0.0
volatilityScore = not na(atrPercentile) ? f_clamp(percentileScore + dirImbalance * 20.0 + trendBasis / math.max(atrValue, syminfo.mintick) * 18.0 + locationScore, -100.0, 100.0) : 0.0

innerUpper = basis + atrValue * innerMult
innerLower = basis - atrValue * innerMult
outerUpper = basis + atrValue * outerMult
outerLower = basis - atrValue * outerMult
midUpper = basis + atrValue * ((innerMult + outerMult) * 0.5)
midLower = basis - atrValue * ((innerMult + outerMult) * 0.5)

expansionUp = barstate.isconfirmed and not na(atrPercentile) and atrPercentile >= 80 and volatilityScore >= stateThreshold
expansionDown = barstate.isconfirmed and not na(atrPercentile) and atrPercentile >= 80 and volatilityScore <= -stateThreshold
compressionState = barstate.isconfirmed and not na(atrPercentile) and atrPercentile <= 25
newExpansionUp = expansionUp and not expansionUp[1]
newExpansionDown = expansionDown and not expansionDown[1]
newCompression = compressionState and not compressionState[1]

//------------------------------------------------------------------------------
// State boxes
//------------------------------------------------------------------------------
var array<box> expansionBoxes = array.new_box()
var array<box> compressionBoxes = array.new_box()

if newExpansionUp and showExpansionBoxes
    box expansionBox = box.new(bar_index, high, bar_index + 12, low, xloc=xloc.bar_index, bgcolor=color.new(bullCore, zoneTransparency), border_color=color.new(bullCore, 0), border_width=2, extend=extend.none)
    array.push(expansionBoxes, expansionBox)
    f_trim_boxes(expansionBoxes, storedStateBoxes)

if newExpansionDown and showExpansionBoxes
    box expansionBox = box.new(bar_index, high, bar_index + 12, low, xloc=xloc.bar_index, bgcolor=color.new(bearCore, zoneTransparency), border_color=color.new(bearCore, 0), border_width=2, extend=extend.none)
    array.push(expansionBoxes, expansionBox)
    f_trim_boxes(expansionBoxes, storedStateBoxes)

if newCompression and showCompressionBoxes
    box compressionBox = box.new(bar_index, outerUpper, bar_index + 14, outerLower, xloc=xloc.bar_index, bgcolor=color.new(accentCore, 92), border_color=color.new(accentCore, 0), border_width=1, extend=extend.none)
    array.push(compressionBoxes, compressionBox)
    f_trim_boxes(compressionBoxes, storedStateBoxes)

//------------------------------------------------------------------------------
// Expansion scaffold
//------------------------------------------------------------------------------
var line entryLine = na
var line slLine = na
var line tp1Line = na
var line tp2Line = na
var line tp3Line = na
var line extremeLine = na
var label entryLabel = na
var label slLabel = na
var label tp1Label = na
var label tp2Label = na
var label tp3Label = na
var linefill riskFill = na
var linefill rewardFill = na
var int activeBias = 0

if (newExpansionUp or newExpansionDown) and showScaffold
    f_delete_line(entryLine)
    f_delete_line(slLine)
    f_delete_line(tp1Line)
    f_delete_line(tp2Line)
    f_delete_line(tp3Line)
    f_delete_line(extremeLine)
    f_delete_label(entryLabel)
    f_delete_label(slLabel)
    f_delete_label(tp1Label)
    f_delete_label(tp2Label)
    f_delete_label(tp3Label)
    f_delete_fill(riskFill)
    f_delete_fill(rewardFill)

    activeBias := newExpansionUp ? 1 : -1
    float entryPrice = close
    float slPrice = activeBias == 1 ? close - atrValue * scaffoldAtrMult : close + atrValue * scaffoldAtrMult
    float riskDistance = activeBias == 1 ? math.max(close - slPrice, syminfo.mintick * 5) : math.max(slPrice - close, syminfo.mintick * 5)
    float tp1Price = activeBias == 1 ? close + riskDistance * tpOneR : close - riskDistance * tpOneR
    float tp2Price = activeBias == 1 ? close + riskDistance * tpTwoR : close - riskDistance * tpTwoR
    float tp3Price = activeBias == 1 ? close + riskDistance * tpThreeR : close - riskDistance * tpThreeR

    color entryColor = activeBias == 1 ? bullCore : bearCore
    color stopColor = color.new(activeBias == 1 ? bearCore : bullCore, 30)
    color targetColor = color.new(activeBias == 1 ? bullCore : bearCore, 28)

    entryLine := line.new(bar_index, entryPrice, bar_index + 1, entryPrice, color=color.new(entryColor, 0), width=2, extend=extend.right)
    slLine := line.new(bar_index, slPrice, bar_index + 1, slPrice, color=stopColor, width=1, style=line.style_dashed, extend=extend.right)
    tp1Line := line.new(bar_index, tp1Price, bar_index + 1, tp1Price, color=targetColor, width=1, extend=extend.right)
    tp2Line := line.new(bar_index, tp2Price, bar_index + 1, tp2Price, color=targetColor, width=1, extend=extend.right)
    tp3Line := line.new(bar_index, tp3Price, bar_index + 1, tp3Price, color=targetColor, width=1, style=line.style_dotted, extend=extend.right)
    extremeLine := line.new(bar_index, tp3Price, bar_index + 1, tp3Price, color=na, extend=extend.right)

    entryLabel := label.new(bar_index, entryPrice, activeBias == 1 ? "Vol Long" : "Vol Short", style=label.style_label_left, color=color.new(entryColor, 0), textcolor=color.white, size=size.small)
    slLabel := label.new(bar_index, slPrice, "SL " + str.tostring(slPrice, format.mintick), style=label.style_label_left, color=stopColor, textcolor=color.white, size=size.small)
    tp1Label := label.new(bar_index, tp1Price, "TP1 " + str.tostring(tp1Price, format.mintick), style=label.style_label_left, color=targetColor, textcolor=color.white, size=size.small)
    tp2Label := label.new(bar_index, tp2Price, "TP2 " + str.tostring(tp2Price, format.mintick), style=label.style_label_left, color=targetColor, textcolor=color.white, size=size.small)
    tp3Label := label.new(bar_index, tp3Price, "TP3 " + str.tostring(tp3Price, format.mintick), style=label.style_label_left, color=targetColor, textcolor=color.white, size=size.small)

    riskFill := linefill.new(entryLine, slLine, color.new(activeBias == 1 ? bearCore : bullCore, 72))
    rewardFill := linefill.new(entryLine, extremeLine, color.new(activeBias == 1 ? bullCore : bearCore, 78))

if activeBias == 1 and barstate.isconfirmed and volatilityScore < 0
    activeBias := 0
    line.set_extend(entryLine, extend.none)
    line.set_extend(slLine, extend.none)
    line.set_extend(tp1Line, extend.none)
    line.set_extend(tp2Line, extend.none)
    line.set_extend(tp3Line, extend.none)
    line.set_x2(entryLine, bar_index)
    line.set_x2(slLine, bar_index)
    line.set_x2(tp1Line, bar_index)
    line.set_x2(tp2Line, bar_index)
    line.set_x2(tp3Line, bar_index)

if activeBias == -1 and barstate.isconfirmed and volatilityScore > 0
    activeBias := 0
    line.set_extend(entryLine, extend.none)
    line.set_extend(slLine, extend.none)
    line.set_extend(tp1Line, extend.none)
    line.set_extend(tp2Line, extend.none)
    line.set_extend(tp3Line, extend.none)
    line.set_x2(entryLine, bar_index)
    line.set_x2(slLine, bar_index)
    line.set_x2(tp1Line, bar_index)
    line.set_x2(tp2Line, bar_index)
    line.set_x2(tp3Line, bar_index)

//------------------------------------------------------------------------------
// Visual layer
//------------------------------------------------------------------------------
pOuterUpper = plot(showRibbon ? outerUpper : na, "Outer Upper", color=color.new(bearCore, 40), linewidth=1)
pMidUpper = plot(showRibbon ? midUpper : na, "Mid Upper", color=color.new(bearCore, 15), linewidth=1)
pInnerUpper = plot(showRibbon ? innerUpper : na, "Inner Upper", color=color.new(bearCore, 0), linewidth=1)
pBasis = plot(showBasis ? basis : na, "Basis", color=color.new(accentCore, 0), linewidth=2)
pInnerLower = plot(showRibbon ? innerLower : na, "Inner Lower", color=color.new(bullCore, 0), linewidth=1)
pMidLower = plot(showRibbon ? midLower : na, "Mid Lower", color=color.new(bullCore, 15), linewidth=1)
pOuterLower = plot(showRibbon ? outerLower : na, "Outer Lower", color=color.new(bullCore, 40), linewidth=1)

fill(pOuterUpper, pMidUpper, color=showRibbon ? color.new(bearCore, 92) : na, title="Upper Outer Ribbon")
fill(pMidUpper, pInnerUpper, color=showRibbon ? color.new(bearCore, 95) : na, title="Upper Mid Ribbon")
fill(pInnerUpper, pBasis, color=showInnerGradient and showRibbon ? color.new(bearCore, 97) : na, title="Upper Inner Gradient")
fill(pBasis, pInnerLower, color=showInnerGradient and showRibbon ? color.new(bullCore, 97) : na, title="Lower Inner Gradient")
fill(pInnerLower, pMidLower, color=showRibbon ? color.new(bullCore, 95) : na, title="Lower Mid Ribbon")
fill(pMidLower, pOuterLower, color=showRibbon ? color.new(bullCore, 92) : na, title="Lower Outer Ribbon")

barcolor(showBarBias ? color.new(f_state_color(volatilityScore), math.abs(volatilityScore) < stateThreshold ? 72 : 18) : na)
bgcolor(showBackdrop and expansionUp ? color.new(bullCore, 92) : showBackdrop and expansionDown ? color.new(bearCore, 92) : showBackdrop and compressionState ? color.new(accentCore, 94) : na, title="Volatility Backdrop")

plot(atrPercentile, "ATR Percentile", color=color.new(accentCore, 80), linewidth=1, display=display.none)
plot(volatilityScore, "Volatility Score", color=color.new(f_state_color(volatilityScore), 80), linewidth=1, display=display.none)
plot(dirImbalance, "Directional Imbalance", color=color.new(accentCore, 80), linewidth=1, display=display.none)

//------------------------------------------------------------------------------
// Dashboard
//------------------------------------------------------------------------------
var table dashboard = table.new(position.top_right, 2, 8, bgcolor=color.new(panelBg, 0), border_width=1, border_color=panelFrame, frame_width=1, frame_color=panelFrame)

if barstate.islast and showDashboard
    string basisText = close > basis ? "Above Basis" : "Below Basis"
    string imbalanceText = dirImbalance >= 0 ? "Bull Dominant" : "Bear Dominant"
    string scaffoldText = activeBias == 1 ? "Long Active" : activeBias == -1 ? "Short Active" : "Idle"

    table.cell(dashboard, 0, 0, "Ribbon", text_color=neutralCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 1, 0, f_state_text(volatilityScore, nz(atrPercentile, 50.0)), text_color=f_state_color(volatilityScore), bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 0, 1, "ATR %", text_color=neutralCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 1, 1, not na(atrPercentile) ? str.tostring(atrPercentile, "#.0") : "Waiting", text_color=accentCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 0, 2, "Volatility", text_color=neutralCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 1, 2, not na(atrPercentile) and atrPercentile >= 75 ? "Expansion" : not na(atrPercentile) and atrPercentile <= 30 ? "Compression" : "Neutral", text_color=not na(atrPercentile) and atrPercentile >= 75 ? accentCore : neutralCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 0, 3, "Basis", text_color=neutralCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 1, 3, basisText, text_color=close >= basis ? bullCore : bearCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 0, 4, "Direction", text_color=neutralCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 1, 4, imbalanceText, text_color=dirImbalance >= 0 ? bullCore : bearCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 0, 5, "Score", text_color=neutralCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 1, 5, str.tostring(volatilityScore, "#.0"), text_color=f_state_color(volatilityScore), bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 0, 6, "ATR", text_color=neutralCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 1, 6, str.tostring(atrValue, format.mintick), text_color=accentCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 0, 7, "Scaffold", text_color=neutralCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
    table.cell(dashboard, 1, 7, scaffoldText, text_color=activeBias == 1 ? bullCore : activeBias == -1 ? bearCore : neutralCore, bgcolor=color.new(panelBg, 0), text_size=size.normal)
else if barstate.islast
    for col = 0 to 1
        for row = 0 to 7
            table.cell(dashboard, col, row, "", text_color=color.new(neutralCore, 100), bgcolor=color.new(panelBg, 100), text_size=size.normal)

alertcondition(expansionUp, "Volatility Expansion Up", "Volatility Covenant Ribbon [JOAT]: confirmed volatility expansion up")
alertcondition(expansionDown, "Volatility Expansion Down", "Volatility Covenant Ribbon [JOAT]: confirmed volatility expansion down")
alertcondition(compressionState, "Volatility Compression", "Volatility Covenant Ribbon [JOAT]: confirmed volatility compression")
alertcondition(newExpansionUp, "New Volatility Expansion Up", "Volatility Covenant Ribbon [JOAT]: new upside expansion cycle started")
alertcondition(newExpansionDown, "New Volatility Expansion Down", "Volatility Covenant Ribbon [JOAT]: new downside expansion cycle started")
alertcondition(newCompression, "New Volatility Compression", "Volatility Covenant Ribbon [JOAT]: new compression cycle started")
