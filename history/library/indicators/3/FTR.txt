Calculate_BoS_ChoCh(_open1, _close1, _low1, _high1, _floats, _ints, _bools, _strings, _lines, _labels, _colors, TimeStamp1, TimeStamp2, TimeStamp3) =>
        
    BoS = array.get(_floats, _BoS)
    ChoCh = array.get(_floats, _ChoCh)
    Highest_High = array.get(_floats, _Highest_High)
    Lowest_Low = array.get(_floats, _Lowest_Low)
    Old_BoS = array.get(_floats, _Old_BoS)
    Old_ChoCh = array.get(_floats, _Old_ChoCh)
    BoS_Down_High = array.get(_floats, _BoS_Down_High)
    BoS_Up_Low = array.get(_floats, _BoS_Up_Low)

    Last_BoS = array.get(_ints, _Last_BoS)
    Last_ChoCh = array.get(_ints, _Last_ChoCh)
    Last_ChoCh_Candidate = array.get(_ints, _Last_ChoCh_Candidate)

    First_Valid_Candle_Read = array.get(_bools, _First_Valid_Candle_Read)
    Low_ChoCh_Candidate = array.get(_bools, _Low_ChoCh_Candidate)
    High_ChoCh_Candidate = array.get(_bools, _High_ChoCh_Candidate)
    Is_Current = array.get(_bools, _Is_Current)

    Trend = array.get(_strings, _Trend)
    BoS_Break_Type = array.get(_strings, _BoS_Break_Type)
    Prev_BoS_Break_Type = array.get(_strings, _Prev_BoS_Break_Type)
    ChoCh_Break_Type = array.get(_strings, _ChoCh_Break_Type)
    Prev_ChoCh_Break_Type = array.get(_strings, _Prev_ChoCh_Break_Type)

    BoS_Line = array.get(_lines, _BoS_Line)
    ChoCh_Line = array.get(_lines, _ChoCh_Line)

    BoS_Line_Label = array.get(_labels, _BoS_Line_Label)
    ChoCh_Line_Label = array.get(_labels, _ChoCh_Line_Label)

    BoS_Line_Color_fx = array.get(_colors, _BoS_Line_Color)
    ChoCh_Line_Color_fx = array.get(_colors, _ChoCh_Line_Color)

    Chart_Text_fx = array.get(_strings, _Chart_Text)

    if ((_open1 != _close1) and (First_Valid_Candle_Read == false))
        First_Valid_Candle_Read := true

        Lowest_Low := _low1
        Highest_High := _high1

        if _open1 < _close1
            BoS := _high1
            ChoCh := _low1
            Trend := "Up"

        else if _open1 > _close1            
            BoS := _low1
            ChoCh := _high1
            Trend := "Down"
        
        Last_BoS := TimeStamp1
        Last_ChoCh := TimeStamp1
        
    else if (First_Valid_Candle_Read)
        
        Old_BoS := BoS
        Old_ChoCh := ChoCh
        
        if Trend == "Up"
            if (_low1 < Lowest_Low)
                Lowest_Low := _low1
                Last_ChoCh_Candidate := TimeStamp1
                if ((TimeStamp1 == (Last_BoS + (TimeStamp1 - TimeStamp2))) and (Lowest_Low < BoS_Up_Low))  // Choch candidate if 1 candle away from last BoS and has lower low
                    Low_ChoCh_Candidate := true
                else if ((TimeStamp1 >= (Last_BoS + (TimeStamp1 - TimeStamp3))) and (_high1 > BoS))  // ChoCh candidate if 2 or more candles away, BoS break   
                    if (_close1 > BoS)  // BoS body break and up candle
                        Low_ChoCh_Candidate := true
                    // down candle or BoS wick break will not be a ChoCh candidate    
                else if (TimeStamp1 > (Last_BoS + (TimeStamp1 -TimeStamp2)))  // Last BoS at least 2 candles away
                    Low_ChoCh_Candidate := true

            if (_high1 > BoS)  // BoS up break
                if ((_close1 >= ChoCh) or (_open1 < _close1))  // set BoS if not getting ChoCh body break in same candle or candle is up candle
                    if (_close1 > BoS)
                        BoS_Break_Type := "BB"
                    else 
                        BoS_Break_Type := "BW"
                    BoS := _high1
                    if (_high1 > Highest_High)
                        Highest_High := _high1
                    if ((Low_ChoCh_Candidate) and (TimeStamp1 > (Last_BoS + (TimeStamp1 - TimeStamp2))))  // make sure Last_BoS is at least 2 candles away
                        ChoCh := Lowest_Low
                        Last_ChoCh := TimeStamp1
                    Low_ChoCh_Candidate := false
                    Lowest_Low := _high1
                    BoS_Up_Low := _low1
                    Last_BoS := TimeStamp1
            
            if (_close1 < ChoCh)  // ChoCh down body break
                ChoCh := BoS
                Last_ChoCh := TimeStamp1
                BoS := _low1
                Last_BoS := TimeStamp1
                Highest_High := _low1
                BoS_Down_High := _high1
                
                Low_ChoCh_Candidate := false
                BoS_Break_Type := "CB"
                ChoCh_Break_Type := ""
                if (Prev_BoS_Break_Type != "CW")
                    Prev_ChoCh_Break_Type := ""
                Trend := "Down"
                
            else if (_low1 < ChoCh)
                ChoCh := _low1
                ChoCh_Break_Type := "CW"
                Last_ChoCh := TimeStamp1
                
        else if Trend == "Down"
            if (_high1 > Highest_High)
                Highest_High := _high1
                Last_ChoCh_Candidate := TimeStamp1
                if ((TimeStamp1 == (Last_BoS + (TimeStamp1 - TimeStamp2))) and (Highest_High > BoS_Down_High))  // Choch candidate if 1 candle away from last BoS and has higher high
                    High_ChoCh_Candidate := true
                else if ((TimeStamp1 > (Last_BoS + (TimeStamp1 - TimeStamp3))) and (_low1 < BoS))  // ChoCh candidate if 2 or more candles away, BoS body break and down candle
                    if (_close1 < BoS)  // BoS body break and down candle
                        High_ChoCh_Candidate := true
                // up candle or BoS wick break will not be a ChoCh candidate
                else if (TimeStamp1 > (Last_BoS + (TimeStamp1 -TimeStamp2)))  // Last BoS at least 2 candles away
                    High_ChoCh_Candidate := true

            if (_low1 < BoS)  // BoS down break
                if ((_close1 <= ChoCh) or (_open1 > _close1))  // set BoS if not getting ChoCh body break in same candle or candle is down candle
                    if (_close1 < BoS)
                        BoS_Break_Type := "BB"
                    else    
                        BoS_Break_Type := "BW"
                    BoS := _low1
                    if (_low1 < Lowest_Low)
                        Lowest_Low := _low1
                    if ((High_ChoCh_Candidate) and (TimeStamp1 > (Last_BoS + (TimeStamp1 - TimeStamp2))))  // make sure Last_BoS is at least 2 candles away
                        ChoCh := Highest_High
                        Last_ChoCh := TimeStamp1
                    High_ChoCh_Candidate := false
                    Highest_High := _low1
                    BoS_Down_High := _high1
                    Last_BoS := TimeStamp1
            
            if (_close1 > ChoCh)  // ChoCh up body break
                ChoCh := BoS
                Last_ChoCh := TimeStamp1
                BoS := _high1
                Last_BoS := TimeStamp1
                Lowest_Low := _high1
                BoS_Up_Low := _low1
                High_ChoCh_Candidate := false
                BoS_Break_Type := "CB"
                ChoCh_Break_Type := ""
                if (Prev_ChoCh_Break_Type != "CW")
                    Prev_ChoCh_Break_Type := ""
                Trend := "Up"     
            else if (_high1 > ChoCh)
                ChoCh := _high1
                ChoCh_Break_Type := "CW"
                Last_ChoCh := TimeStamp1

        if (BoS != Old_BoS)    
            if ((Hide_BoS_History) or not(Is_Current))
                line.delete(BoS_Line)
                label.delete(BoS_Line_Label)
            else    
                line.set_extend(BoS_Line, extend.none)
                line.set_x2(BoS_Line, TimeStamp1)
            if ((Label_BoS_ChoCh_History) and (Is_Current))
                label.set_text(BoS_Line_Label, text = Prev_BoS_Break_Type)
            else
                label.delete(BoS_Line_Label)
            
            BoS_Line := line.new(Last_BoS,BoS,time,BoS,xloc=xloc.bar_time, color = BoS_Line_Color_fx, extend = extend.right)

            BoS_Line_Label := label.new(Last_BoS, BoS, text= Chart_Text_fx + "BoS", xloc=xloc.bar_time, color = BoS_Line_Color_fx, textcolor = BoS_Line_Color_fx)
            label.set_style(BoS_Line_Label,label.style_none)
        
            Prev_BoS_Break_Type := BoS_Break_Type
            
        if (ChoCh != Old_ChoCh)     
            if ((Hide_ChoCh_History) or not(Is_Current))
                line.delete(ChoCh_Line)
                label.delete(ChoCh_Line_Label)
            else
                line.set_extend(ChoCh_Line, extend.none)
                line.set_x2(ChoCh_Line, TimeStamp1)
            if ((Label_BoS_ChoCh_History) and (Is_Current))
                label.set_text(ChoCh_Line_Label, Prev_ChoCh_Break_Type)
            else
                label.delete(ChoCh_Line_Label)
            
            ChoCh_Line := line.new(Last_ChoCh_Candidate,ChoCh,time,ChoCh,xloc=xloc.bar_time, color = ChoCh_Line_Color_fx, extend = extend.right)

            ChoCh_Line_Label := label.new(Last_ChoCh_Candidate, ChoCh, text= Chart_Text_fx + "ChoCh", xloc=xloc.bar_time, color = ChoCh_Line_Color_fx, textcolor = ChoCh_Line_Color_fx)
            label.set_style(ChoCh_Line_Label, label.style_none)
            
            Prev_ChoCh_Break_Type := ChoCh_Break_Type

        if (BoS > ChoCh)
            label.set_yloc(BoS_Line_Label, yloc.abovebar)
            label.set_yloc(ChoCh_Line_Label, yloc.belowbar)
        else
            label.set_yloc(BoS_Line_Label, yloc.belowbar)
            label.set_yloc(ChoCh_Line_Label, yloc.abovebar)

        if (barstate.islast)
            label.set_x(BoS_Line_Label, time)
            label.set_x(ChoCh_Line_Label, time) 
            label.set_yloc(BoS_Line_Label, yloc.price)
            label.set_yloc(ChoCh_Line_Label, yloc.price)

    array.set(_floats, _BoS, BoS)
    array.set(_floats, _ChoCh, ChoCh)
    array.set(_floats, _Highest_High, Highest_High)
    array.set(_floats, _Lowest_Low, Lowest_Low)
    array.set(_floats, _Old_BoS, Old_BoS)
    array.set(_floats, _Old_ChoCh, Old_ChoCh)
    array.set(_floats, _BoS_Down_High, BoS_Down_High)
    array.set(_floats, _BoS_Up_Low, BoS_Up_Low)

    array.set(_ints, _Last_BoS, Last_BoS)
    array.set(_ints, _Last_ChoCh, Last_ChoCh)
    array.set(_ints, _Last_ChoCh_Candidate, Last_ChoCh_Candidate)

    array.set(_bools, _First_Valid_Candle_Read, First_Valid_Candle_Read)
    array.set(_bools, _Low_ChoCh_Candidate, Low_ChoCh_Candidate)
    array.set(_bools, _High_ChoCh_Candidate, High_ChoCh_Candidate)

    array.set(_strings, _Trend, Trend)
    array.set(_strings, _BoS_Break_Type, BoS_Break_Type)
    array.set(_strings, _Prev_BoS_Break_Type, Prev_BoS_Break_Type)
    array.set(_strings, _ChoCh_Break_Type, ChoCh_Break_Type)
    array.set(_strings, _Prev_ChoCh_Break_Type, Prev_ChoCh_Break_Type)

    array.set(_lines, _BoS_Line, BoS_Line)
    array.set(_lines, _ChoCh_Line, ChoCh_Line)

    array.set(_labels, _BoS_Line_Label, BoS_Line_Label)
    array.set(_labels, _ChoCh_Line_Label, ChoCh_Line_Label)

    //result = 1  //needed to return result to expression in request.security function

    //if ((time > (timestamp("13 Feb 2024 01:00:00 GMT-5"))) and (time < (timestamp("14 Feb 2024 01:00:00 GMT-5"))))
    //    log.info("\nLow1: {0,number,#.#####} Lowest_Low: {1,number,#.#####} BoS_Down_High: {2,number,#.######} Trend: {3} \nBoS: {4,number,#.#####}  ChoCh: {5,number,#.#####} \nHigh_ChoCh_Candidate: {6}", low[1], Lowest_Low, BoS_Down_High, Trend, BoS, ChoCh, High_ChoCh_Candidate)

TimeStamp1 = time[1]
TimeStamp2 = time[2]
TimeStamp3 = time[3]

var float BoS = 0
var float ChoCh = 0

Calculate_BoS_ChoCh(open[1], close[1], low[1], high[1], current_floats, current_ints, current_bools, current_strings, current_lines, current_labels, current_colors, TimeStamp1, TimeStamp2, TimeStamp3)

BoS := array.get(current_floats, _BoS)
ChoCh := array.get(current_floats, _ChoCh)

// added these lines only to show BoS and ChoCh values in status line and at end of line on chart.  Set transparency for colors to 100% so plot is not visible on chart

if ((Include_htf == false) or (htf == ""))

_controlBox_OB(_boxes, _lines25, _lines50, _lines75, _high, _low, TimeStamp1, _isCurrent) =>
    if array.size(_boxes) > 0
        for i = array.size(_boxes) - 1 to 0 by 1
            _box = array.get(_boxes, i)
            _boxLow = box.get_bottom(_box)
            _boxHigh = box.get_top(_box)
            _boxRight = box.get_right(_box)
            _line25 = array.get(_lines25, i)
            _line25y1 = line.get_y1(_line25)
            _line50 = array.get(_lines50, i)
            _line50y1 = line.get_y1(_line50)
            _line75 = array.get(_lines75, i)
            _line75y1 = line.get_y1(_line75)
             
            if (timeframe.isseconds or (timeframe.isminutes and timeframe.multiplier <= 15))
                if ((_high > _line25y1 and _line25y1 < _line50y1) or (_low < _line25y1 and _line25y1 > _line50y1))
                    if (filterMitOB) and (_isCurrent)   
                        box.set_bgcolor(_box, mitOBColor)
                        box.set_border_color(_box, mitOBColor)
                        line.set_color(_line25, mitOBColor)
                        line.set_color(_line50, mitOBColor)
                        line.set_color(_line75, mitOBColor)
                    else if not (_isCurrent)
                        box.delete(_box)
                        line.delete(_line25)
                        line.delete(_line50)
                        line.delete(_line75)
                else if (_boxRight == TimeStamp1)
                    box.set_right(_box, time)
                    line.set_x2(_line50, time)
                    line.set_x2(_line25, time)
                    line.set_x2(_line75, time)
            else if ((_high > _line50y1 and _line25y1 < _line50y1) or (_low < _line50y1 and _line25y1 > _line50y1))
                if (filterMitOB) and (_isCurrent)
                    box.set_bgcolor(_box, mitOBColor)
                    box.set_border_color(_box, mitOBColor)
                    line.set_color(_line25, mitOBColor)
                    line.set_color(_line50, mitOBColor)
                    line.set_color(_line75, mitOBColor) 
                else if not (_isCurrent)
                    box.delete(_box)
                    line.delete(_line25)
                    line.delete(_line50)
                    line.delete(_line75)
            else if (_boxRight == TimeStamp1)
                box.set_right(_box, time)
                line.set_x2(_line50, time)
                line.set_x2(_line25, time)
                line.set_x2(_line75, time)    
                //log.info("time: {0} ", str.format_time(time + (time - time[1]), "dd HH:mm z", "America/New_York"))
                    
_controlBox_FVG(_boxes, _lines50, _high, _low, TimeStamp1, _isCurrent) =>
    if array.size(_boxes) > 0
        for i = array.size(_boxes) - 1 to 0 by 1
            _box = array.get(_boxes, i)
            _boxLow = box.get_bottom(_box)
            _boxHigh = box.get_top(_box)
            _boxRight = box.get_right(_box)
            _line50 = array.get(_lines50, i)
            _line50y1 = line.get_y1(_line50)
        
            if na or (_boxRight == TimeStamp1 and not((_high > _boxLow and _low < _boxLow) or (_high > _boxHigh and _low < _boxHigh)))
                box.set_right(_box, time)
                line.set_x2(_line50, time)
            else if (_boxRight == TimeStamp1)
                if (_high > _boxHigh and _low < _boxHigh)
                    if (_low > _boxLow)
                        box.set_top(_box, _low)
                        box.set_right(_box, time)
                        line.set_y1(_line50, (_low + _boxLow)/2)
                        line.set_y2(_line50, (_low + _boxLow)/2)
                        line.set_x2(_line50, time)

                        box.set_bgcolor(_box, mitFVGColor)

                        line.set_color(_line50, mitFVGColor)
                    else if not (_isCurrent)
                        box.delete(_box)
                        line.delete(_line50)
                if (_high > _boxLow and _low < _boxLow)
                    if (_high < _boxHigh)    
                        box.set_bottom(_box, _high)
                        box.set_right(_box, time)
                        line.set_y1(_line50, (_high + _boxHigh)/2)
                        line.set_y2(_line50, (_high + _boxHigh)/2)

                    else if (filterMitFVG) and (_isCurrent)

//Bullish OB Box Plotting
_createBullBox_OB(_open1, _open2, _close1, _close2, _low1, _low2, _high2, bullBoxesOB, bullLinesOB25, bullLinesOB50, bullLinesOB75, TimeStamp1, TimeStamp2, _isCurrent) =>
    if (_close2 < _open2 and _close1 > _open1 and _close1 > _high2)
        line_50 = math.avg(_high2, math.min(_low2,_low1))
        line_25 = (line_50 + _high2) / 2
        line_75 = (line_50 + (math.min(_low2, _low1))) / 2

        BullColor = _isCurrent ? obBullColor : obBullColor_htf
        obLabel = _isCurrent ? _obLabel : htf_POI_Text + _obLabel

        bullboxOB = box.new(left = TimeStamp2, top=_high2, right = time, bottom=math.min(_low2, _low1), xloc = xloc.bar_time, border_color=color.new(BullColor, obBorderTransparency), border_style=obBoxBorder, border_width=1, bgcolor=BullColor, 
         text=plotLabelOB ? obLabel  + _plus : _empty, text_halign=text.align_right, text_valign=text.align_bottom, text_size=obLabelSize, text_color=obLabelColor)
        bulllineOB25 = line.new(TimeStamp2, line_25, time, line_25, xloc = xloc.bar_time, style = line.style_solid, color = color.new(BullColor, 60))
        bulllineOB50 = line.new(TimeStamp2, line_50, time, line_50, xloc = xloc.bar_time, style = line.style_solid, color = color.new(BullColor, 20))
        bulllineOB75 = line.new(TimeStamp2, line_75, time, line_75, xloc = xloc.bar_time, style = line.style_solid, color = color.new(BullColor, 60))
        if array.size(bullBoxesOB) > obMaxBoxSet
            box.delete(array.shift(bullBoxesOB))
            line.delete(array.shift(bullLinesOB25))
            line.delete(array.shift(bullLinesOB50))
            line.delete(array.shift(bullLinesOB75))
        array.push(bullBoxesOB, bullboxOB)
        array.push(bullLinesOB25, bulllineOB25)
        array.push(bullLinesOB50, bulllineOB50)
        array.push(bullLinesOB75, bulllineOB75)
