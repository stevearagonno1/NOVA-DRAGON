//@version=5
indicator("IQFXpro", overlay=true)

// === الإعدادات ===
a = input.int(2, title = 'Sawiqfx')
c = input.int(25, title = 'Signal (1-30 s)')
h = input.bool(false, title = 'Signals from Heikin Ashi Candles')
target_multiplier_1 = input.float(0.5, title = "Target Multiplier 1 (TP1)", step = 0.1)
target_multiplier_2 = input.float(1.0, title = "Target Multiplier 2 (TP2)", step = 0.1)
target_multiplier_3 = input.float(1.5, title = "Target Multiplier 3 (TP3)", step = 0.1)


// === حساب ATR والمصدر ===
xATR = ta.atr(c)
nLoss = a * xATR
src = h ? request.security(ticker.heikinashi(syminfo.tickerid), timeframe.period, close) : close

// === حساب ستوب متحرك ATR Trailing Stop ===
var float xATRTrailingStop = na
iff_1 = src > nz(xATRTrailingStop[1]) ? src - nLoss : src + nLoss
iff_2 = src < nz(xATRTrailingStop[1]) and src[1] < nz(xATRTrailingStop[1]) ? math.min(nz(xATRTrailingStop[1]), src + nLoss) : iff_1
xATRTrailingStop := src > nz(xATRTrailingStop[1]) and src[1] > nz(xATRTrailingStop[1]) ? math.max(nz(xATRTrailingStop[1]), src - nLoss) : iff_2

// === متغيرات الصفقة ===
var int pos = 0
var float entry_price = na
var float tp1 = na
var float tp2 = na
var float tp3 = na
var float sl = na

// === إشارات الدخول ===
new_buy_signal = src[1] < nz(xATRTrailingStop[1]) and src > nz(xATRTrailingStop[1])
new_sell_signal = src[1] > nz(xATRTrailingStop[1]) and src < nz(xATRTrailingStop[1])

if new_buy_signal
    pos := 1
    entry_price := src
    tp1 := entry_price + target_multiplier_1 * xATR
    tp2 := entry_price + target_multiplier_2 * xATR
    tp3 := entry_price + target_multiplier_3 * xATR
    sl := entry_price - nLoss

if new_sell_signal
    pos := -1
    entry_price := src
    tp1 := entry_price - target_multiplier_1 * xATR
    tp2 := entry_price - target_multiplier_2 * xATR
    tp3 := entry_price - target_multiplier_3 * xATR
    sl := entry_price + nLoss

// === إشارات التعزيز ===
buy_reinforcement = pos == 1 and close > tp3
sell_reinforcement = pos == -1 and close < tp3

if buy_reinforcement
    entry_price := src
    tp1 := entry_price + target_multiplier_1 * xATR
    tp2 := entry_price + target_multiplier_2 * xATR
    tp3 := entry_price + target_multiplier_3 * xATR
    sl := entry_price - nLoss

if sell_reinforcement
    entry_price := src
    tp1 := entry_price - target_multiplier_1 * xATR
    tp2 := entry_price - target_multiplier_2 * xATR
    tp3 := entry_price - target_multiplier_3 * xATR
    sl := entry_price + nLoss

// === رسم الخطوط ===
plot(pos != 0 and not na(tp1) ? tp1 : na, title="TP1", color=#19f520, linewidth=2, style=plot.style_cross)
plot(pos != 0 and not na(tp2) ? tp2 : na, title="TP2", color=#19f520, linewidth=2, style=plot.style_cross)
plot(pos != 0 and not na(tp3) ? tp3 : na, title="TP3", color=#19f520, linewidth=2, style=plot.style_cross)
plot(pos != 0 and not na(sl) ? sl : na, title="Stop Loss", color=color.red, linewidth=2, style=plot.style_cross)









plotshape(series=new_buy_signal, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.labelup, text="Buy", textcolor=color.white)
plotshape(series=new_sell_signal, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.labeldown, text="Sell", textcolor=color.white)


plotshape(series=buy_reinforcement, title="Buy Signal", location=location.belowbar, color=color.rgb(1, 5, 243), style=shape.labelup, text="Re-Buy", textcolor=color.white)
plotshape(series=sell_reinforcement, title="Sell Signal", location=location.abovebar, color=color.rgb(226, 128, 16), style=shape.labeldown, text="Re-Sell", textcolor=color.white)





// === تنبيهات ===
alertcondition(new_buy_signal, title="New Buy Alert", message="New Buy Signal")
alertcondition(new_sell_signal, title="New Sell Alert", message="New Sell Signal")
alertcondition(buy_reinforcement, title="Buy Reinforcement", message="Buy Reinforcement Signal")
alertcondition(sell_reinforcement, title="Sell Reinforcement", message="Sell Reinforcement Signal")
