// KAMA — Kaufman Adaptive Moving Average
kamaF(simple int kLen) =>
    float diff = math.abs(ta.change(close, kLen))
    float vola = math.sum(math.abs(ta.change(close)), kLen)
    float er   = vola != 0.0 ? diff / vola : 0.0
    float sc   = math.pow(er * (2.0 / 3.0 - 2.0 / 31.0) + 2.0 / 31.0, 2.0)
    var float k = na
    k := nz(k[1], close) + sc * (close - nz(k[1], close))
    k

// FER — Fractal Efficiency Ratio
fer(simple int len) =>
    float net  = math.abs(close - close[len])
    float path = 0.0
    for i = 0 to len - 2
        path += math.abs(close[i] - close[i + 1])
    path > 0.0 ? net / path : 0.0

// HMA ribbon helper
rh(simple int len) =>

    iHH := ph > nz(lPH, 0.0)
    iLH := ph < nz(lPH, 999999.0)
    lPH := ph
if not na(pl)
    iHL := pl > nz(lPL, 0.0)
    iLL := pl < nz(lPL, 999999.0)
    lPL := pl

float bStr = (iHH or iHL) and close > nz(lPL, close) ? float(wStruct) : 0.0
float eStr = (iLH or iLL) and close < nz(lPH, close) ? float(wStruct) : 0.0

// ◈ PILLAR 2 — Volume (OBV Slope)
// ============================================================

float bVol   = obvSlp > 0.0 ? float(wVolume) : 0.0
float eVol   = obvSlp < 0.0 ? float(wVolume) : 0.0

// ============================================================
// ◈ PILLAR 3 — Momentum (KAMA + RSI + WPR)
// ============================================================

float bMomR = (close > kVal   ? 1.0 : 0.0) +
              (rsiVal > 50.0   ? 1.0 : 0.0) +
              (wprVal > -40.0  ? 1.0 : 0.0)
float eMomR = (close < kVal   ? 1.0 : 0.0) +
              (rsiVal < 50.0   ? 1.0 : 0.0) +
              (wprVal < -60.0  ? 1.0 : 0.0)
float bMom  = (bMomR / 3.0) * float(wMomnt)
float eMom  = (eMomR / 3.0) * float(wMomnt)

// ◈ PILLAR 4 — Liquidity (Sweep Detection)
// ============================================================

float eSwp = not na(lPH) and high > lPH and close < lPH ? float(wLiquid) : 0.0

float eVlt   = (atrRel > 0.8 and atrRel < 1.6 and eMomR >= 2.0) ? float(wVolatl) : 0.0

// ◈ PILLAR 6 — Session (Daily Range Midpoint)
// ============================================================

    sH := high
    sL := low

    sH := math.max(high, nz(sH, high))
    sL := math.min(low,  nz(sL, low))

float sR  = nz(sH, high) - nz(sL, low)

float bSes = dM < 0.0 ?
     math.min(float(wSessn), math.abs(dM) * (float(wSessn) / 50.0)) :
     float(wSessn) * 0.4

     math.min(float(wSessn), math.abs(dM) * (float(wSessn) / 50.0)) :
     float(wSessn) * 0.4

// ============================================================
// ◈ PILLAR 8 — Delta Pressure
// ============================================================

var float cumDelta = 0.0

float deltaMA    = ta.sma(cumDelta, 14)
float deltaRatio = volume > 0.0 ? rawDelta / volume : 0.0
float bodyComp   = (high - low) > 0.0 ?
     math.min(math.abs(close - open) / (high - low), 1.0) : 0.0
float volMA14    = ta.ema(volume, 14)
float volComp    = volMA14 > 0.0 ?
     math.min(math.max(volume / volMA14 - 1.0, 0.0) / 1.5, 1.0) : 0.0
float pressScore = (math.abs(deltaRatio) * 0.40 +
     volComp * 0.35 + bodyComp * 0.25) * 100.0
bool  deltaXup   = ta.crossover(cumDelta,  deltaMA)
bool  deltaXdn   = ta.crossunder(cumDelta, deltaMA)
float bDelta = pressScore > 50.0 and deltaXup ? float(wDelta) : 0.0
float eDelta = pressScore > 50.0 and deltaXdn ? float(wDelta) : 0.0

// ◈ STOP DISTANCE FUNCTION
// ============================================================
stopDist() =>
    float raw = math.abs(close - stVal)
    float result = switch stopMode
        "ATR Cap" => math.min(raw, atrVal * stopAtrM)
        "Fixed %" => close * stopFixPct / 100.0
        "Pivot"   => confirmedLong ?
             math.abs(close - nz(lPL, close - atrVal)) :
             math.abs(close - nz(lPH, close + atrVal))
        => raw
    result

    bool btLoss = tDir == 1 ? low  <= tSL : high >= tSL
    if btWin or btLoss
        tTrades += 1

            tWins  += 1
            tPnlR  += rr2
