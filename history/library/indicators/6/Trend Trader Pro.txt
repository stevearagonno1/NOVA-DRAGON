indicator("Trend Trader Pro - Dynamic Volume & Trend v1.01", overlay=true, shorttitle = "Trend Trader Pro")
// --- 1. Inputs ---
emaFastLen   = input.int(9, title="Fast EMA (Trend Filter)", minval=1, maxval=50, step=1)
emaSlowLen   = input.int(21, title="Slow EMA Length", minval=10, maxval=200, step=1)
emaFilterLen = input.int(50, title="Filter EMA Length", tooltip = "Well known as 50 EMA", minval=20, maxval=500, step=1)
chopThresh   = input.int(60, title="Chop Threshold", tooltip = "Bigger number for room to move. Lower number for more restrict. Range 10 to 100", minval=10, maxval=100, step=1)

grp_cmf     = "Confirmation Candles"
CandleConfirmation = input.int(1, title="Confirmation candle Before Entry",minval=0, maxval=3, step=1, group=grp_cmf)

grp_mtf = "Multi-Timeframe Settings"
useMTF      = input.bool(true, "Use MTF for EMA 50 & Chop?", group=grp_mtf)
mtfTime     = input.timeframe("3", "Higher Timeframe for Smoothing", group=grp_mtf)
fastRev     = input.bool(true, "Fast Reversal Mode? (Prioritizes 1m Cross)", group=grp_mtf)

grp_vol      = "Volume & Trend Continuity"
rvolLen      = input.int(20, "Volume Avg Lookback", group=grp_vol, minval=1, maxval=30, step=1)
rvolThresh   = input.float(1.2, "RVOL Multiplier for Next TP", group=grp_vol, minval=1, maxval=3, step=0.1)
OnlyStrongVol = input.bool(true,"Only Strong Volume to show Signal", group=grp_vol)

atrLen = input.int(14, "ATR Length.", minval=1, maxval=20, step=1)
atrMult = input.float(1.5, "ATR Multiplier.", minval=1, maxval=3, step=0.5)
labelPersist = input.int(2, "Label Visibility (Bars).", minval=1, maxval=5, step=1, group="Style", tooltip = "Labels to kept show after SL/Exit/Trend Reversed")

// --- 1. Inputs ---
grp_filters = "Filter Toggles"
useMACD      = input.bool(true, "Use MACD Momentum Filter?", group=grp_filters)
useTrend     = input.bool(false, "Strict 50 EMA Filter? (Keep OFF for Downtrend Sells)", group=grp_filters)
useChop      = input.bool(true, "Use Choppiness Filter? (Turn OFF for >15m charts)", group=grp_filters)

grp_session  = "Session Settings"
useSession   = input.bool(true, "Restrict Trading to US Market Hours Session?", group=grp_session, tooltip = "Off this Market Hours Restriction, if you are trading BTC.")
sessionInput = input.session("0930-1600", "Market Hours (HHMM-HHMM)", group=grp_session, tooltip = "Adjust Market Hours to include Pre-Market Data.")
timezone     = "America/New_York" //input.string("UTC-5", "Timezone", options=["UTC-5", "UTC+0", "UTC+8", "UTC+7"], group=grp_session)

// Define the 4:00 PM Sharp Trigger (New York Time)
is_close_soon = (hour(time, "America/New_York") == 15 and minute(time, "America/New_York") == 59)

// Define the "After Hours" block
is_market_closed = hour(time, "America/New_York") >= 16

// --- 2. Session & Time Logic ---
// Logic: Data calculates always, but is_rth triggers the "release" at 09:30AM New Yoek Time
t_rth = time(timeframe.period, sessionInput, timezone)
is_rth = not useSession or not na(t_rth)

// --- Color Customization ---
grp_style = "Label Text Colors"
buyTextColor  = input.color(color.white, "Buy Text Color", group=grp_style)
sellTextColor = input.color(color.white, "Sell Text Color", group=grp_style)
exitTextColor = input.color(color.white, "Exit/SL Text Color", group=grp_style)
slColor       = input.color(color.red, "Stop Loss Color", group=grp_style)
tp1Color      = input.color(color.green, "TP1 Color", group=grp_style)
tp2Color      = input.color(color.green, "TP2 Color", group=grp_style)
tp3Color      = input.color(color.green, "TP3 Color", group=grp_style)
tp4Color      = input.color(color.green, "TP4 Color", group=grp_style)
tp5Color      = input.color(color.green, "TP5 Color", group=grp_style)
tp6Color      = input.color(color.green, "TP6 Color", group=grp_style)
hitColor      = input.color(color.orange, "TP Hit Label Color", group=grp_style)

grp_risk = "Risk Management"
IsShowSignals     = input.bool(true, "Show Entry/Exit/TP levels signals", group=grp_risk)
slPctVal   = input.float(0.2, "Default SL (%)", group=grp_risk, minval=0.1, maxval=10, step=0.1)
tp1PctVal  = input.float(0.2, "Default TP1 (%)", group=grp_risk, minval=0.1, maxval=10, step=0.1)
tp2PctVal  = input.float(0.4, "Default TP2 (%)", group=grp_risk, minval=0.1, maxval=10, step=0.1)
tp3PctVal  = input.float(0.6, "Default TP3 (%)", group=grp_risk, minval=0.1, maxval=10, step=0.1)
tp4PctVal  = input.float(0.8, "Default TP4 (%)", group=grp_risk, minval=0.1, maxval=10, step=0.1)
tp5PctVal  = input.float(1.0, "Default TP5 (%)", group=grp_risk, minval=0.1, maxval=10, step=0.1)
tp6PctVal  = input.float(1.2, "Default TP6 (%)", group=grp_risk, minval=0.1, maxval=10, step=0.1)

slPct  = slPctVal / 100
tp1Pct = tp1PctVal / 100
tp2Pct = tp2PctVal / 100
tp3Pct = tp3PctVal / 100
tp4Pct = tp4PctVal / 100
tp5Pct = tp5PctVal / 100
tp6Pct = tp6PctVal / 100

// --- 2. Calculations ---
emaFast   = ta.ema(close, emaFastLen)
emaSlow   = ta.ema(close, emaSlowLen)
// === Dashboard with Telegram Link ===
var table myTable = table.new(position.bottom_right, 1, 1, border_width=1, frame_color=color.black, bgcolor=color.white)

// Add Telegram Message to Dashboard
table.cell(myTable, 0, 0, "Join Telegram @mrexpert_ai", bgcolor=color.blue, text_color=color.white, text_size=size.normal)
// Fetching the 50 EMA from the Higher Timeframe
// Adaptive Filter: If fastRev is on, it uses local 50 EMA for faster reaction
emaFilterRaw = ta.ema(close, emaFilterLen)
emaFilter = (useMTF and not fastRev) ? request.security(syminfo.tickerid, mtfTime, emaFilterRaw, gaps=barmerge.gaps_off) : emaFilterRaw
vwapVal   = ta.vwap(close)
atrVal = ta.atr(atrLen)
avgVol      = ta.sma(volume, rvolLen)
isStrongVol = volume > (avgVol * rvolThresh)
trendStrength = math.abs(emaFast - emaSlow)
[mLine, sLine, hist] = ta.macd(close, 12, 26, 9)

// Fetching Choppiness from Higher Timeframe to keep signals consistent across 1m/3m/5m
ciRaw = 100 * math.log10(math.sum(ta.tr, atrLen) / (ta.highest(high, atrLen) - ta.lowest(low, atrLen))) / math.log10(atrLen)
ci = useMTF ? request.security(syminfo.tickerid, mtfTime, ciRaw, gaps=barmerge.gaps_off) : ciRaw

// 2. Define "Safe to Trade" 
// Gatekeeper: Market Hours + No Chop
canTrade = is_rth and (not useChop or ci < chopThresh)

// --- Market Hours Filter ---
inSession = not useSession or not na(time(timeframe.period, sessionInput, timezone))

// --- 3. State Management ---
var int dir = 0
var bool trendIsLong = false
var bool trendIsShort = false
var bool stoppedOut = false
var label lblAction = na
var line  lnAction  = na
var label lblEntrySig = na
var line  lnEntrySig  = na
var bool longCondition = false
var bool shortCondition = false
var float pEntry = 0.0, var float pSL = 0.0
var float pTP1 = 0.0, var float pTP2 = 0.0, var float pTP3 = 0.0, var float pTP4 = 0.0, var float pTP5 = 0.0, var float pTP6 = 0.0
var bool hit1 = false, var bool hit2 = false, var bool hit3 = false, var bool hit4 = false, var bool hit5 = false, var bool hit6 = false 

var line lEntry = na, var line lSL = na, var line lTP1 = na, var line lTP2 = na, var line lTP3 = na, var line lTP4 = na, var line lTP5 = na, var line lTP6 = na 
var label lblE = na, var label lblSL = na, var label lblTP1 = na, var label lblTP2 = na, var label lblTP3 = na, var label lblTP4 = na, var label lblTP5 = na, var label lblTP6 = na 
// Lists to keep track of multiple TP hits
var tpLabels = array.new_label(0)
var tpLines  = array.new_line(0)
var int tradeEndBar = 0

offset      = ta.atr(atrLen) * 1.5
LabelOffSet = ta.atr(atrLen) * 3

cleanVisuals() =>
    // 1. Delete Trade Levels
    line.delete(lEntry), line.delete(lSL), line.delete(lTP1), line.delete(lTP2), line.delete(lTP3), line.delete(lTP4), line.delete(lTP5), line.delete(lTP6)
    label.delete(lblE), label.delete(lblSL), label.delete(lblTP1), label.delete(lblTP2), label.delete(lblTP3), label.delete(lblTP4), label.delete(lblTP5), label.delete(lblTP6)
    
    // 2. Delete Entry Signals
    label.delete(lblEntrySig), line.delete(lnEntrySig)
    
    // 3. Delete Exit/SL Markers
    line.delete(lnAction), label.delete(lblAction)

    // 4. Clear TP Hit Array
    if array.size(tpLabels) > 0
        for i = 0 to array.size(tpLabels) - 1
            label.delete(array.get(tpLabels, i))
        array.clear(tpLabels)
        
    if array.size(tpLines) > 0
        for i = 0 to array.size(tpLines) - 1
            line.delete(array.get(tpLines, i))
        array.clear(tpLines)

// Anchor Function: Fixes zoom drift by using specific bar indexing and Y-levels
createStatusLabel(txt, col, isLong) =>
    labelY = isLong ? high + offset : low - offset
    candleY= isLong ? high  : low 
    lbl =   label.new(bar_index, isLong ? high + offset : low - offset, txt, yloc=isLong ? yloc.price : yloc.price, color=col,text_formatting = text.format_bold, textcolor=color.white, style=isLong ? label.style_label_down : label.style_label_up, size=size.small)
    ln = line.new(bar_index, candleY, bar_index, labelY, color=col, style=line.style_dotted, width=1)

    array.push(tpLabels, lbl)
    array.push(tpLines, ln)
    
createExitLabel(txt, col, isLong) =>
    labelY = isLong ? high + offset : low - offset
    candleY= isLong ? high  : low 
    label.new(bar_index, isLong ? high + LabelOffSet : low - LabelOffSet, txt, yloc=isLong ? yloc.price : yloc.price, color=col,text_formatting = text.format_bold, textcolor=color.white, style=isLong ? label.style_label_down : label.style_label_up, size=size.small)
    line.new(bar_index, candleY, bar_index, labelY, 
         color=col, 
         style=line.style_dotted, 
         width=1)

createBuySellLabel(txt, col, isLong) =>
    label.new(bar_index, isLong ? high + offset : low - offset, txt, yloc=isLong ? yloc.price : yloc.price, color=col,text_formatting = text.format_bold, textcolor=color.white, style=isLong ? label.style_label_down : label.style_label_up, size=size.small)

// --- 4. Logic ---
ReverseDown = ta.crossunder(emaFast, emaSlow)
ReverseUp = ta.crossover(emaFast, emaSlow)

IsUpTrendContinued = ta.crossover(close, emaFast)
IsDownTrendContinued = ta.crossunder(close, emaFast)

longCondition  := ReverseUp
shortCondition := ReverseDown

if not IsUpTrendContinued 
    longCondition  := ReverseUp

if not IsDownTrendContinued
    shortCondition := ReverseDown

// Store the crossover event if it happens ANYTIME (Pre-market)
if longCondition
    trendIsLong := true
    trendIsShort := false
    stoppedOut   := false // Reset SL protection on fresh cross

if shortCondition
    trendIsShort := true
    trendIsLong  := false
    stoppedOut   := false // Reset SL protection on fresh cross

// 3. Re-entry Logic: If stopped out, look for price to be on the "right side" of Fast EMA
// This allows the script to fire again if the trend is still strong.
reEntryLong  = trendIsLong  and IsUpTrendContinued
reEntryShort = trendIsShort and IsDownTrendContinued

// Logic: Use the stored trend for Market Open, or the re-entry trigger after SL
triggerLong  = (not stoppedOut and trendIsLong) or (stoppedOut and reEntryLong)
triggerShort = (not stoppedOut and trendIsShort) or (stoppedOut and reEntryShort)

// Entry Release Logic: Triggers if a cross happens now OR if a pending cross hits the Market Open
longEntry  = dir == 0 and is_rth and canTrade and (longCondition[CandleConfirmation] or reEntryLong or trendIsLong) and (not useMACD or hist > 0) and (not useChop or ci < chopThresh) and (not useTrend or close > emaFilter) and (not OnlyStrongVol or isStrongVol)
shortEntry = dir == 0 and is_rth and canTrade and (shortCondition[CandleConfirmation] or reEntryShort or trendIsShort) and (not useMACD or hist < 0) and (not useChop or ci < chopThresh) and (not useTrend or close < emaFilter) and (not OnlyStrongVol or isStrongVol)

isStrongTrend = (dir == 1 ? close > emaFast : close < emaFast)

if longEntry or shortEntry 
    // Clears the LAST trade's exit marker
    // --- STEP 1: Scrub the chart of the PREVIOUS trade entirely ---
    label.delete(lblEntrySig), line.delete(lnEntrySig)
    label.delete(lblAction) 
    line.delete(lnAction)
    cleanVisuals() // Cleans TP/SL lines

    // --- STEP 2: Start the new trade ---
    dir  := longEntry ? 1 : -1

    hit1 := false, hit2 := false, hit3 := false, hit4 := false, hit5 := false, hit6 := false 
    
    pEntry := open
    pSL  := pEntry * (1 - (dir * slPct))
    pTP1 := pEntry * (1 + (dir * tp1Pct))
    pTP2 := pEntry * (1 + (dir * tp2Pct))
    pTP3 := pEntry * (1 + (dir * tp3Pct))
    pTP4 := pEntry * (1 + (dir * tp4Pct))
    pTP5 := pEntry * (1 + (dir * tp5Pct))
    pTP6 := pEntry * (1 + (dir * tp6Pct))
    
    if not is_close_soon and not is_market_closed and IsShowSignals
        lEntry := line.new(bar_index, pEntry, bar_index + 10, pEntry, color=color.gray, style=line.style_dotted, extend=extend.none)
        lblE   := label.new(bar_index + 10, pEntry, "ENTRY: " + str.tostring(pEntry, "#.##"), color=color.new(color.gray, 20), style=label.style_label_left, textcolor=color.white, size=size.small)
        lSL    := line.new(bar_index - 10, pSL, bar_index + 10, pSL, color=slColor, style=line.style_dotted, extend=extend.none)
        lblSL  := label.new(bar_index + 10, pSL, "SL: " + str.tostring(pSL, "#.##") + " (" + str.tostring(slPctVal, "#.#") + "%)", color=slColor, style=label.style_label_left, textcolor=exitTextColor, size=size.small)
        lTP1   := line.new(bar_index, pTP1, bar_index + 10, pTP1, color=tp1Color, style=line.style_dotted, extend=extend.none)
        lblTP1 := label.new(bar_index + 10, pTP1, "TP1: " + str.tostring(pTP1, "#.##") + " (" + str.tostring(tp1PctVal, "#.#") + "%)", color=tp1Color, style=label.style_label_left, textcolor=color.white, size=size.small)

if dir != 0 and IsShowSignals
    label.set_x(lblE, bar_index + 10), label.set_x(lblSL, bar_index + 10)
    line.set_x2(lEntry, bar_index + 10), line.set_x2(lSL, bar_index + 10)
    
    if not na(lblTP1) 
        label.set_x(lblTP1, bar_index + 10)
        line.set_x2(lTP1, bar_index + 10)
    if not na(lblTP2) 
        label.set_x(lblTP2, bar_index + 10)
        line.set_x2(lTP2, bar_index + 10)
    if not na(lblTP3) 
        label.set_x(lblTP3, bar_index + 10)
        line.set_x2(lTP3, bar_index + 10)
    if not na(lblTP4) 
        label.set_x(lblTP4, bar_index + 10)
        line.set_x2(lTP4, bar_index + 10)
    if not na(lblTP5) 
        label.set_x(lblTP5, bar_index + 10)
        line.set_x2(lTP5, bar_index + 10)
    if not na(lblTP6) 
        label.set_x(lblTP6, bar_index + 10)
        line.set_x2(lTP6, bar_index + 10)
    
    bool isL = dir == 1
    
    // TP1 Hit -> Unlock TP2
    if not hit1 and (isL ? high >= pTP1 : low <= pTP1) 
        hit1 := true
        line.set_x1(lTP1, bar_index)
        createStatusLabel("TP1 HIT", hitColor, isL)

        if isStrongVol or isStrongTrend
            lTP2 := line.new(bar_index, pTP2, bar_index + 10, pTP2, color=tp2Color, style=line.style_dotted, extend=extend.none)
            lblTP2 := label.new(bar_index + 10, pTP2, "TP2: " + str.tostring(pTP2, "#.##") + " (" + str.tostring(tp2PctVal, "#.#") + "%)", color=tp2Color, style=label.style_label_left, textcolor=color.white, size=size.small)

    // TP2 Hit -> Unlock TP3
    if hit1 and not hit2 and (isL ? high >= pTP2 : low <= pTP2) 
        hit2 := true
        line.set_x1(lTP2, bar_index)
        createStatusLabel("TP2 HIT", hitColor, isL)
  
        if isStrongVol or isStrongTrend 
            lTP3 := line.new(bar_index, pTP3, bar_index + 10, pTP3, color=tp3Color, style=line.style_dotted, extend=extend.none)
            lblTP3 := label.new(bar_index + 10, pTP3, "TP3: " + str.tostring(pTP3, "#.##") + " (" + str.tostring(tp3PctVal, "#.#") + "%)", color=tp3Color, style=label.style_label_left, textcolor=color.white, size=size.small)

    // TP3 Hit -> Unlock TP4
    if hit2 and not hit3 and (isL ? high >= pTP3 : low <= pTP3) 
        hit3 := true
        line.set_x1(lTP3, bar_index)
        createStatusLabel("TP3 HIT", hitColor, isL)
       
        if isStrongVol or isStrongTrend
            lTP4 := line.new(bar_index, pTP4, bar_index + 10, pTP4, color=tp4Color, style=line.style_dotted, extend=extend.none)
            lblTP4 := label.new(bar_index + 10, pTP4, "TP4: " + str.tostring(pTP4, "#.##") + " (" + str.tostring(tp4PctVal, "#.#") + "%)", color=tp4Color, style=label.style_label_left, textcolor=color.white, size=size.small)

    // TP4 Hit -> Unlock TP5
    if hit3 and not hit4 and (isL ? high >= pTP4 : low <= pTP4) 
        hit4 := true
        line.set_x1(lTP4, bar_index)
        createStatusLabel("TP4 HIT", hitColor, isL)

        if isStrongVol or isStrongTrend
            lTP5 := line.new(bar_index, pTP5, bar_index + 10, pTP5, color=tp5Color, style=line.style_dotted, extend=extend.none)
            lblTP5 := label.new(bar_index + 10, pTP5, "TP5: " + str.tostring(pTP5, "#.##") + " (" + str.tostring(tp5PctVal, "#.#") + "%)", color=tp5Color, style=label.style_label_left, textcolor=color.white, size=size.small)

    // TP5 Hit -> Unlock TP6
    if hit4 and not hit5 and (isL ? high >= pTP5 : low <= pTP5) 
        hit5 := true
        line.set_x1(lTP5, bar_index)
        createStatusLabel("TP5 HIT", hitColor, isL)
        
        if isStrongVol or isStrongTrend
            lTP6 := line.new(bar_index, pTP6, bar_index + 10, pTP6, color=tp6Color, style=line.style_dotted, extend=extend.none)
            lblTP6 := label.new(bar_index + 10, pTP6, "TP6: " + str.tostring(pTP6, "#.##") + " (" + str.tostring(tp6PctVal, "#.#") + "%)", color=tp6Color, style=label.style_label_left, textcolor=color.white, size=size.small)

    if hit5 and not hit6 and (isL ? high >= pTP6 : low <= pTP6)
        hit6 := true
        line.set_x1(lTP6, bar_index)
        createStatusLabel("TP6 HIT", hitColor, isL)

    //  Exit Check 
    //  1. Stop Loss Hit Exit
    if (isL ? low <= pSL : high >= pSL)
        label.delete(lblEntrySig), line.delete(lnEntrySig)
        label.delete(lblAction) 
        line.delete(lnAction)

        stoppedOut := true // Set SL protection

        dir := 0
        tradeEndBar := bar_index // Record the time of death
        
        if not is_close_soon and not is_market_closed
            // Create new SL markers and SAVE to global variables
            lblAction := label.new(bar_index, isL ? high + offset : low - offset, "SL HIT", color=color.red, textcolor=color.white, style=isL ? label.style_label_down : label.style_label_up, size=size.small, text_formatting=text.format_bold)
            lnAction  := line.new(bar_index, isL ? high : low, bar_index, isL ? high + offset : low - offset, color=color.red, style=line.style_dotted)

    // 2. Trend Reversal / MACD Exit
    else if (isL and ReverseDown and hist < 0) or (not isL and ReverseUp and hist > 0)
        label.delete(lblEntrySig), line.delete(lnEntrySig)
        label.delete(lblAction) 
        line.delete(lnAction)
        
        stoppedOut := true // Set SL protection
        
        dir := 0
        tradeEndBar := bar_index // Record the time of death
        
        if not is_close_soon and not is_market_closed
            lblAction := label.new(bar_index, isL ? high + offset : low - offset, "EXIT", color=color.purple, textcolor=color.white, style=isL ? label.style_label_down : label.style_label_up, size=size.small, text_formatting=text.format_bold)
            lnAction  := line.new(bar_index, isL ? high : low, bar_index, isL ? high + offset : low - offset, color=color.purple, style=line.style_dotted)     
        
    // NEW: Force Exit at 4:00 PM Sharp
    if is_close_soon or is_market_closed
        dir := 0
        tradeEndBar := bar_index
        stoppedOut := true // Prevent re-entry until next day
        
        lblAction := label.new(bar_index, isL ? high + offset : low - offset, "⚠️ EOD FORCE EXIT", 
             color=color.gray, 
             textcolor=color.white, 
             style=label.style_label_down,text_formatting = text.format_bold)

        lnAction  := line.new(bar_index, isL ? high : low, bar_index, isL ? high + offset : low - offset, color=color.gray, style=line.style_dotted)     
        
        if bar_index == tradeEndBar + labelPersist    
            cleanVisuals() // Clean up your TP/SL lines for the day

// --- 5. Plotting ---
plot(emaFast, color=color.blue, title="9 EMA", linewidth = 2)
plot(emaSlow, color=color.orange, title="21 EMA", linewidth = 2)
plot(emaFilter, color=color.purple, title="50 EMA Trend", linewidth = 2)
plot(vwapVal, color=color.white, title="VWAP")

// --- 1. Define the Spacing (Add to Inputs if not already there) ---
labelGapVal = 1.5 // input.float(1.5, "Signal Label Spacing", minval = 0.0, step = 0.1, group="Style")

// --- 2. Calculate Offset ---
sigOffset = ta.atr(atrLen) * labelGapVal

// --- 3. BUY Signal with Dotted Connector ---
if longEntry and (not is_close_soon) and IsShowSignals
    // The Label
    lblEntrySig := label.new(bar_index, low - sigOffset, "BUY", 
         style=label.style_label_up, 
         color=color.green, 
         textcolor=buyTextColor, 
         text_formatting=text.format_bold,
         size=size.small)
    // The Dotted Line
    lnEntrySig := line.new(bar_index, low, bar_index, low - sigOffset, 
         color=color.green, 
         style=line.style_dotted, 
         width=1)

// --- 4. SELL Signal with Dotted Connector ---
if shortEntry and (not is_close_soon) and IsShowSignals
    // The Label
    lblEntrySig := label.new(bar_index, high + sigOffset, "SELL", 
         style=label.style_label_down, 
         color=color.red, 
         textcolor=sellTextColor, 
         text_formatting=text.format_bold,
         size=size.small)
    // The Dotted Line
    lnEntrySig := line.new(bar_index, high, bar_index, high + sigOffset,
