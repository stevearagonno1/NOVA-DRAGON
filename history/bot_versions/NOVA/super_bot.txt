def ema_series(vals, period):
    out = [None] * len(vals)
    if len(vals) < period:
        return out
    k = 2.0 / (period + 1)
    prev = sum(vals[:period]) / period
    out[period - 1] = prev
    for i in range(period, len(vals)):
        prev = vals[i] * k + prev * (1 - k)
        out[i] = prev
    return out

def sma_series(vals, period):
    out = [None] * len(vals)
    if len(vals) < period:
        return out
    s = sum(vals[:period])
    out[period - 1] = s / period
    for i in range(period, len(vals)):
        s += vals[i] - vals[i - period]
        out[i] = s / period
    return out

def rsi_series(closes, period=14):
    """RSI بطريقة وايلدر الأصلية — يعيد قائمة كاملة لكشف الدايفرجنس"""
    out = [None] * len(closes)
    if len(closes) < period + 1:
        return out
    gains = losses = 0.0
    for i in range(1, period + 1):
        ch = closes[i] - closes[i - 1]
        gains += max(ch, 0.0)
        losses += max(-ch, 0.0)
    ag, al = gains / period, losses / period
    out[period] = 100.0 if al == 0 else 100 - 100 / (1 + ag / al)
    for i in range(period + 1, len(closes)):
        ch = closes[i] - closes[i - 1]
        ag = (ag * (period - 1) + max(ch, 0.0)) / period
        al = (al * (period - 1) + max(-ch, 0.0)) / period
        out[i] = 100.0 if al == 0 else 100 - 100 / (1 + ag / al)
    return out

def macd_series(closes, fast=12, slow=26, signal=9):
    ef, es = ema_series(closes, fast), ema_series(closes, slow)
    line = [None if (ef[i] is None or es[i] is None) else ef[i] - es[i]
            for i in range(len(closes))]
    valid = [v for v in line if v is not None]
    sig_valid = ema_series(valid, signal)
    sig = [None] * len(closes)
    j = 0
    for i in range(len(closes)):
        if line[i] is not None:
            sig[i] = sig_valid[j]
            j += 1
    hist = [None if (line[i] is None or sig[i] is None) else line[i] - sig[i]
            for i in range(len(closes))]
    return line, sig, hist

def bollinger_series(closes, period=20, mult=2.0):
    up, mid, lo = [None] * len(closes), [None] * len(closes), [None] * len(closes)
    for i in range(period - 1, len(closes)):
        w = closes[i - period + 1:i + 1]
        m = sum(w) / period
        sd = statistics.pstdev(w)
        mid[i], up[i], lo[i] = m, m + mult * sd, m - mult * sd
    return up, mid, lo

def atr_series(highs, lows, closes, period=14):
    n = len(closes)
    out = [None] * n
    if n < period + 1:
        return out
    trs = [max(highs[i] - lows[i],
               abs(highs[i] - closes[i - 1]),
               abs(lows[i] - closes[i - 1])) for i in range(1, n)]
    atr = sum(trs[:period]) / period
    out[period] = atr
    for j in range(period, len(trs)):
        atr = (atr * (period - 1) + trs[j]) / period
        out[j + 1] = atr
    return out

def stoch_rsi_series(closes, rsi_period=14, stoch_period=14, smooth=3):
    rsi = rsi_series(closes, rsi_period)
    raw = [None] * len(closes)
    for i in range(len(closes)):
        if i < stoch_period - 1:
            continue
        w = rsi[i - stoch_period + 1:i + 1]
        if any(v is None for v in w):
            continue
        hi, lo = max(w), min(w)
        raw[i] = 50.0 if hi == lo else (rsi[i] - lo) / (hi - lo) * 100
    k = sma_opt(raw, smooth)
    d = sma_opt(k, smooth)
    return raw, k, d

def sma_opt(vals, period):
    out = [None] * len(vals)
    for i in range(period - 1, len(vals)):
        w = vals[i - period + 1:i + 1]
        if any(v is None for v in w):
            continue
        out[i] = sum(w) / period
    return out

def obv_series(closes, volumes):
    obv = [0.0]
    for i in range(1, len(closes)):
        if closes[i] > closes[i - 1]:
            obv.append(obv[-1] + volumes[i])
        elif closes[i] < closes[i - 1]:
            obv.append(obv[-1] - volumes[i])
        else:
            obv.append(obv[-1])
    return obv

def swing_indices(prices, mode="low", w=3, lookback=90):
    n = len(prices)
    start = max(w, n - lookback)
    idxs = []
    for i in range(start, n - w):
        left = prices[i - w:i]
        right = prices[i + 1:i + w + 1]
        if mode == "low" and prices[i] < min(left) and prices[i] <= min(right):
            idxs.append(i)
        elif mode == "high" and prices[i] > max(left) and prices[i] >= max(right):
            idxs.append(i)
    return idxs

def cluster_levels(vals, merge_pct=0.004):
    if not vals:
        return []
    vals = sorted(vals)
    clusters, cur = [], [vals[0]]
    for v in vals[1:]:
        if v <= cur[-1] * (1 + merge_pct):
            cur.append(v)
        else:
            clusters.append(sum(cur) / len(cur))
            cur = [v]
    clusters.append(sum(cur) / len(cur))
    return clusters

def support_resistance(highs, lows, live):
    sup_c = [lows[i] for i in swing_indices(lows, "low")]
    res_c = [highs[i] for i in swing_indices(highs, "high")]
    sups = cluster_levels(sup_c)
    ress = cluster_levels(res_c)
    support = max([s for s in sups if s <= live * 1.004], default=None)
    resistance = min([r for r in ress if r >= live * 0.996], default=None)
    return support, resistance

def bullish_divergence(closes, rsi):
    """السعر يصنع قاعاً أدنى بينما RSI يصنع قاعاً أعلى = انعكاس محتمل"""
    lows = swing_indices(closes, "low", w=3, lookback=70)
    if len(lows) < 2:
        return False, None
    i2, i1 = lows[-1], lows[-2]
    if len(closes) - 1 - i2 > 14:
        return False, None
    r1, r2 = rsi[i1], rsi[i2]
    if r1 is None or r2 is None:
        return False, None
    if closes[i2] < closes[i1] and r2 > r1 + 1.5:
        return True, (closes[i1], closes[i2])
    return False, None

def candle_pattern(opens, highs, lows, closes):
    """كشف أنماط الشموع الانعكاسية على آخر شمعة مغلقة"""
    o1, h1, l1, c1 = opens[-1], highs[-1], lows[-1], closes[-1]
    o0, h0, l0, c0 = opens[-2], highs[-2], lows[-2], closes[-2]
    rng = max(h1 - l1, 1e-12)
    body = abs(c1 - o1)
    upper = h1 - max(o1, c1)
    lower = min(o1, c1) - l1
    green, red0 = c1 > o1, c0 < o0
    body0 = abs(c0 - o0)

    if body <= 0.35 * rng and lower >= 2.0 * body and upper <= 0.6 * body:
        return "hammer", green
    if red0 and green and c1 >= o0 and o1 <= c0 and body > body0 * 1.05:
        return "engulf", True
    if body <= 0.35 * rng and upper >= 2.0 * body and lower <= 0.6 * body:
        return "star", green
    if c0 > o0 and c1 < o1 and c1 <= o0 and o1 >= c0 and body > body0 * 1.05:
        return "bear_engulf", False
    return None, green

def trend_of(closes):
    e9, e21, e50 = ema_series(closes, 9), ema_series(closes, 21), ema_series(closes, 50)
    if e9[-1] is None or e21[-1] is None:
        return "side"
    up = e9[-1] > e21[-1]
    above = e50[-1] is not None and closes[-1] > e50[-1]
    if up and above:
        return "up"
    if (not up) and (not above):
        return "down"
    return "side"

TICK_SIZES = {}

def binance_get(path, params=None, timeout=10):
    global STATE
    hosts = [STATE.get("host")] + [h for h in BINANCE_HOSTS if h != STATE.get("host")]
    hosts = [h for h in hosts if h]
    for host in hosts:
        try:
            r = SESSION.get(host + path, params=params, timeout=timeout)
            if r.status_code == 200:
                STATE["host"] = host
                return r.json()
        except Exception:
            continue
    return None

def load_tick_sizes():
    global TICK_SIZES
    symbols = list(set(COINS + [LEADER]))
    params = {"symbols": json.dumps(symbols, separators=(',', ':'))}
    data = binance_get("/api/v3/exchangeInfo", params)
    if not data or "symbols" not in data:
        log("⚠️ تعذر جلب دقة الأسعار — سأستخدم تقريباً تلقائياً")
        return
    for s in data["symbols"]:
        for f in s.get("filters", []):
            if f["filterType"] == "PRICE_FILTER":
                TICK_SIZES[s["symbol"]] = float(f["tickSize"])
    log(f"📐 دقة الأسعار جاهزة لـ {len(TICK_SIZES)} عملة")

def price_decimals(symbol, fallback_price):
    tick = TICK_SIZES.get(symbol)
    if not tick:
        p = fallback_price
        for dec in (8, 7, 6, 5, 4, 3, 2):
            if p >= 10 ** (3 - dec):
                return dec if dec > 0 else 2
        return 6
    d = 0
    t = tick
    while t < 0.999 and d < 9:
        t *= 10
        d += 1
    return d

def fmt_price(symbol, price):
    dec = price_decimals(symbol, price)
    return f"{price:.{dec}f}"

def fetch_klines(symbol, interval, limit=300):
    data = binance_get("/api/v3/klines",
                       {"symbol": symbol, "interval": interval, "limit": limit})
    if not data or not isinstance(data, list) or len(data) < 60:
        return None
    opens  = [float(k[1]) for k in data]
    highs  = [float(k[2]) for k in data]
    lows   = [float(k[3]) for k in data]
    closes = [float(k[4]) for k in data]
    vols   = [float(k[5]) for k in data]
    qvol   = [float(k[7]) for k in data]
    return {"o": opens, "h": highs, "l": lows, "c": closes,
            "v": vols, "qv": qvol, "live": closes[-1]}

def get_ticker(symbol):
    data = binance_get("/api/v3/ticker/price", {"symbol": symbol}, timeout=8)
    if data and "price" in data:
        return float(data["price"])
    return None

def analyze_btc():
    k1, k4 = fetch_klines(LEADER, "1h", 300), fetch_klines(LEADER, "4h", 260)
    if not k1 or not k4:
        return {"regime": "neutral", "rsi1h": 50, "t1h": "side", "t4h": "side", "live": 0}
    c1, c4 = k1["c"][:-1], k4["c"][:-1]
    t1h, t4h = trend_of(c1), trend_of(c4)
    rsi1 = rsi_series(c1)[-1] or 50
    live = k1["live"]
    if t1h != "down" and t4h != "down" and (t1h == "up" or t4h == "up") and rsi1 > 45:
        regime = "bullish"
    elif t1h == "down" and t4h == "down":
        regime = "bearish"
    elif t4h == "down" and rsi1 < 40:
        regime = "bearish"
    else:
        regime = "neutral"
    return {"regime": regime, "rsi1h": rsi1, "t1h": t1h, "t4h": t4h, "live": live}

# 7) بناء خطة الصفقة — وقف وأهداف ديناميكية حسب تقلب العملة (ATR)

def build_plan(symbol, an):
    live, atr = an["live"], an["atr"]
    sl_dist = max(1.4 * atr, live * 0.008)        # وقف لا يقل عن 0.8%
    sl_dist = min(sl_dist, live * 0.035)          # ولا يزيد عن 3.5%
    r = sl_dist
    plan = {
        "entry": live,
        "sl": live - r,
        "be": live + 1.0 * r,       # تصفير المخاطرة عند +1R
        "tp1": live + 1.5 * r,      # بيع 50%
        "tp2": live + 2.5 * r,      # بيع 30%
        "tp3": live + 4.0 * r,      # بيع 20%
        "sl_pct": r / live * 100,
        "rr": 2.5,
        "trail_pct": min(max(atr / live * 100 * 1.2, 1.5), 6.0),
    }
    plan["size_usd"] = min(MAX_POSITION_USD, RISK_PER_TRADE / (plan["sl_pct"] / 100))
    plan["actual_risk"] = plan["size_usd"] * plan["sl_pct"] / 100
    plan["rr"] = round((plan["tp2"] - live) / r, 1)
    return plan

def ask_gemini(prompt):
    models = ([STATE.get("gemini_model")] if STATE.get("gemini_model") else []) + \
             [m for m in GEMINI_MODELS if m != STATE.get("gemini_model")]
    for model in models:
        try:
            url = (f"https://generativelanguage.googleapis.com/v1beta/models/"
                   f"{model}:generateContent?key=__REDACTED__")
            res = SESSION.post(url, timeout=12, json={
                "contents": [{"parts": [{"text": prompt}]}]})
            js = res.json()
            if "candidates" in js and js["candidates"]:
                txt = js["candidates"][0]["content"]["parts"][0]["text"].strip()
                STATE["gemini_model"] = model
                return txt.replace("**", "").replace("*", "")[:600]  # تلجرام لا يفهم Markdown
        except Exception:
            continue
    return None

def tg(method, payload, files=None, retries=3):
    url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/{method}"
    for i in range(retries):
        try:
            if files:
                r = SESSION.post(url, data=payload, files=files, timeout=25)
            else:
                r = SESSION.post(url, json=payload, timeout=15)
            js = r.json()
            if js.get("ok"):
                return js.get("result")
        except Exception:
            time.sleep(2 + i)
    return None

def make_chart(symbol, an, plan, path):

    axr.grid(alpha=0.25)

def poll_commands():
    try:
        # أول تشغيل: تجاهل أي أوامر قديمة متراكمة من قبل التشغيل
        if not STATE.get("bootstrapped"):
            STATE["bootstrapped"] = True
            res = tg("getUpdates", {"offset": -1, "timeout": 0}, retries=1)
            if res:
                STATE["tg_offset"] = res[-1]["update_id"] + 1
            save_state()
            return
        res = tg("getUpdates", {
            "offset": STATE["tg_offset"], "timeout": 0,
            "allowed_updates": json.dumps(["message"])}, retries=1)
        if not res:
            return
        for upd in res:
            STATE["tg_offset"] = max(STATE["tg_offset"], upd["update_id"] + 1)
            msg = upd.get("message") or {}
            text = (msg.get("text") or "").strip()
            chat = str(msg.get("chat", {}).get("id"))
            if chat != CHAT_ID or not text.startswith("/"):
                continue
            cmd = text.split()[0].split("@")[0].lower()
            arg = text.split()[1] if len(text.split()) > 1 else ""
            log(f"📩 أمر وارد: {cmd} {arg}")
            handle_command(cmd, arg)
    except Exception:
        pass

def handle_command(cmd, arg):
    if cmd == "/help":
        send_alert("🤖 <b>أوامر NOVA PRO:</b>\n"
                   "/status — حالة البوت والصفقات النشطة\n"
                   "/coin XLM — تحليل فني فوري لأي عملة\n"
                   "/scan — فحص فوري للسوق الآن\n"
                   "/pause — إيقاف مؤقت\n/resume — استئناف\n"
                   "/close XLM — أغلق متابعة صفقة يدوياً")
    elif cmd == "/status":
        send_alert(build_status_message(analyze_btc()))
    elif cmd == "/scan":
        send_alert("🔎 جارٍ فحص السوق الكامل الآن...")
        run_scan(force=True)
        send_alert("✅ انتهى الفحص.")
    elif cmd == "/coin":
        if not arg:
            send_alert("✍️ اكتب رمز العملة هكذا: /coin XLM")
            return
        sym = arg.upper().replace("/", "")
        if not sym.endswith("USDT"):
            sym += "USDT"
        an = analyze_symbol(sym, analyze_btc())
        if an:
            send_alert(build_coin_report(an, analyze_btc()))
        else:
            send_alert(f"⚠️ تعذر جلب بيانات {sym} (تأكد من صحة الرمز على بينانس)")
    elif cmd == "/pause":
        STATE["paused"] = True
        save_state()
        send_alert("⏸️ توقف الفحص مؤقتاً. الصفقات النشطة تبقى تحت المراقبة. /resume للعودة")
    elif cmd == "/resume":
        STATE["paused"] = False
        save_state()
        send_alert("▶️ عاد الفحص للعمل.")
    elif cmd == "/close":
        sym = arg.upper().replace("/", "")
        if not sym.endswith("USDT"):
            sym += "USDT"
        if STATE["positions"].pop(sym, None) is not None:
            save_state()
            send_alert(f"✔️ أغلقت متابعة {sym}. لن تصلك تنبيهات إدارة لها.")
        else:
            send_alert(f"لا توجد صفقة نشطة لـ{sym}")
    else:
        send_alert("أمر غير معروف — جرب /help")

# 10) إدارة الصفقات النشطة (تصفير مخاطرة، أهداف، وقف، ركود)

def manage_positions():
    for sym in list(STATE["positions"].keys()):
        p = STATE["positions"][sym]
        price = get_ticker(sym)
        if not price:
            continue
        name = sym.replace("USDT", "/USDT")
        p["high"] = max(p.get("high", price), price)
        stage = p["stage"]
        entry, sl = p["entry"], p["sl"]

        if stage == 0 and price <= sl:
            send_alert(f"🛑 <b>ضرب وقف الخسارة — اخرج الآن</b>\n\n"
                       f"🔹 {name} بلغ <code>{fmt_price(sym, price)} $</code>\n"
                       f"الخسارة المتوقعة: <b>{p['risk_usd']:.2f} $</b>\n\n"
                       f"💡 انضباطك بالوقف هو ما يجعلك مربحاً على المدى الطويل. "
                       f"فرص أخرى قادمة.")
            del STATE["positions"][sym]
            STATE["cooldowns"][sym] = time.time()
        elif stage == 0 and price >= p["be"]:
            p["stage"] = 1
            send_alert(f"🛡️ <b>تصفير المخاطرة (Break-even)</b>\n\n"
                       f"🔹 {name} ارتفعت إلى <code>{fmt_price(sym, price)} $</code> "
                       f"(<b>{(price/entry-1)*100:+.1f}%</b>)\n"
                       f"↩️ انقل وقف الخسارة إلى سعر الدخول "
                       f"<code>{fmt_price(sym, entry)} $</code>\n"
                       f"🛡️ صفقتك الآن <b>بلا مخاطرة</b> — استمتع بالركب.")
        elif stage < 2 and price >= p["tp1"]:
            p["stage"] = 2
            p["sl"] = entry
            send_alert(f"🎯 <b>الهدف الأول تحقق!</b>\n\n"
                       f"🔹 {name} بلغ <code>{fmt_price(sym, price)} $</code> "
                       f"(<b>{(price/entry-1)*100:+.1f}%</b>)\n"
                       f"💵 <b>بِع 50%</b> من الكمية الآن\n"
                       f"↩️ ارفع الوقف إلى الدخول <code>{fmt_price(sym, entry)} $</code>\n"
                       f"🔄 فعّل التريلينج {p['trail_pct']:.1f}% على الباقي\n"
                       f"🎯 الهدف التالي: <code>{fmt_price(sym, p['tp2'])} $</code>")
        elif stage < 3 and price >= p["tp2"]:
            p["stage"] = 3
            send_alert(f"🚀 <b>الهدف الثاني تحقق!</b>\n\n"
                       f"🔹 {name} بلغ <code>{fmt_price(sym, price)} $</code> "
                       f"(<b>{(price/entry-1)*100:+.1f}%</b>)\n"
                       f"💵 <b>بِع 30%</b> إضافية\n"
                       f"🔄 الباقي 20% بوقف متحرك {p['trail_pct']:.1f}%\n"
                       f"🎯 الهدف الأخير: <code>{fmt_price(sym, p['tp3'])} $</code>")
        elif price >= p["tp3"]:
            send_alert(f"🏆 <b>الهدف الثالث الكامل! صفقة مثالية</b>\n\n"
                       f"🔹 {name} بلغ <code>{fmt_price(sym, price)} $</code> "
                       f"(<b>{(price/entry-1)*100:+.1f}%</b>)\n"
                       f"💵 بِع الكمية المتبقية (20%) واعتبر الصفقة مغلقة.\n"
                       f"🎉 هذا ما نسميه صفقة بلا أخطاء.")
            del STATE["positions"][sym]
            STATE["cooldowns"][sym] = time.time()
        elif stage >= 1 and price <= sl:
            zone = "نقطة الدخول (بدون خسارة)" if sl <= entry * 1.001 else "الوقف المتحرك"
            send_alert(f"✅ <b>إغلاق آمن على {zone}</b>\n\n"
                       f"🔹 {name} عادت إلى <code>{fmt_price(sym, price)} $</code>\n"
                       f"أُغلقت الصفقة مع حماية الأرباح. القمة المسجلة كانت "
                       f"<code>{fmt_price(sym, p['high'])} $</code>.")
            del STATE["positions"][sym]
            STATE["cooldowns"][sym] = time.time()
        else:
            age_h = (time.time() - p["time"]) / 3600
            if (stage == 0 and age_h > STALE_HOURS and price < entry
                    and not p.get("stale_warned")):
                p["stale_warned"] = True
                send_alert(f"😴 <b>صفقة راكدة</b>\n\n"
                           f"🔹 {name} منذ {age_h:.0f} ساعة تحت نقطة الدخول "
                           f"({(price/entry-1)*100:+.1f}%)\n"
                           f"💡 رأس المال النائم تكلفة فرصة — قرر: خروج أو صبر بوقف محكم.")
        save_state()

def should_signal(an, btc):
    if an["vetoes"]:
        return False, "veto"
    if an["ok_count"] < MIN_CONDITIONS:
        return False, "قليلة الشروط"
    if not (an["near_support"] or an["divergence"]):
        return False, "لا دعم ولا دايفرجنس"
    if not an["hist_rising"] and not an["macd_cross_up"]:
        return False, "الماكد لا يؤكد"
    need = MIN_SCORE_BT_BEAR if btc["regime"] == "bearish" else MIN_SCORE
    if an["score"] < need:
        return False, f"النتيجة {an['score']}% < {need}%"
    return True, "GO"

def run_scan(force=False):
    btc = analyze_btc()
    now = time.time()
    for symbol in COINS:
        # تحذير القمة للصفقات النشطة (بيعه/رفع الوقف قبل فوات الأرباح)
        if symbol in STATE["positions"]:
            p = STATE["positions"][symbol]
            an = analyze_symbol(symbol, btc)
            if an and not p.get("top_warned"):
                hist_falling = (an.get("hist_last") is not None and
                                an.get("macd_signal") is not None and
                                an.get("macd_line") is not None and
                                an["macd_line"] < an["macd_signal"])
                bear_pat = an["pattern"] in ("star", "bear_engulf")
                if ((an["rsi"] or 0) >= 72 and (hist_falling or bear_pat)) or (an["rsi"] or 0) >= 78:
                    p["top_warned"] = True
                    name = symbol.replace("USDT", "/USDT")
                    chg = (an["live"] / p["entry"] - 1) * 100
                    send_alert(
                        f"⚠️ <b>إشارة قمة — جني أرباح</b>\n\n"
                        f"🔹 <b>العملة:</b> {name}\n"
                        f"🔹 السعر: <code>{fmt_price(symbol, an['live'])} $</code> "
                        f"(<b>{chg:+.1f}%</b> من دخولك)\n"
                        f"📊 RSI: <b>{an['rsi']:.1f}</b> (تشبع شرائي) | "
                        f"ماكد: {'يتحوّل هبوطاً' if hist_falling else 'مرتفع'} | "
                        f"نمط: {an['pattern'] or 'عادي'}\n\n"
                        f"💡 <b>الإجراء:</b> بِع جزءاً الآن أو ارفع الوقف المتحرك إلى "
                        f"{p['trail_pct']:.1f}% لحماية المكاسب.")
                    log(f"⚠️ تحذير قمة {symbol}")
            continue
        if not force and now - STATE["cooldowns"].get(symbol, 0) < COOLDOWN_SEC:
            continue
        an = analyze_symbol(symbol, btc)
        if not an:
            continue
        ok, why = should_signal(an, btc)
        if ok:
            plan = build_plan(symbol, an)
            ai_text = ai_summary(symbol, an, plan, btc)
            send_alert(build_signal_message(symbol, an, plan, btc, ai_text))
            if SEND_CHARTS:
                chart_path = os.path.join(BASE_DIR, f"chart_{symbol}.png")
                if make_chart(symbol, an, plan, chart_path):
                    send_chart_photo(
                        chart_path, f"{symbol} 1H — score {an['score']}% ({an['grade']})\n"
                                    f"Entry {fmt_price(symbol, plan['entry'])} | "
                                    f"SL {fmt_price(symbol, plan['sl'])} | "
                                    f"TP2 {fmt_price(symbol, plan['tp2'])}")
            ts = datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M")
            log_signal_csv(f"{ts},{symbol},BUY,{plan['entry']:.6f},{plan['sl']:.6f},"
                           f"{plan['tp1']:.6f},{plan['tp2']:.6f},{plan['tp3']:.6f},"
                           f"{an['score']},{an['grade']}")
            STATE["positions"][symbol] = {
                "entry": plan["entry"], "sl": plan["sl"], "be": plan["be"],
                "tp1": plan["tp1"], "tp2": plan["tp2"], "tp3": plan["tp3"],
                "stage": 0, "trail_pct": plan["trail_pct"],
                "size_usd": plan["size_usd"], "risk_usd": plan["actual_risk"],
                "score": an["score"], "grade": an["grade"],
                "time": now, "high": plan["entry"],
                "stale_warned": False, "top_warned": False,
            }
            STATE["cooldowns"][symbol] = now
            log(f"🚨 إشارة {symbol} نتيجة {an['score']}% ({an['grade']})")
        else:
            log(f"↦ {symbol}: {an['score']}% ({an['grade']}) — {why}")
        time.sleep(0.15)
    return btc

def main():
    load_state()
    load_tick_sizes()
    send_alert("🚀 <b>تم تفعيل NOVA PRO v4.0</b>\n"
               "🔎 يقرأ الشارت على 3 فريمات (15د/1س/4س) ويربط 13 خيطاً فنياً\n"
               "🎯 أهداف متدرجة + وقف ديناميكي حسب تقلب كل عملة\n"
               "🛡️ تصفير مخاطرة تلقائي + إدارة مراحل كاملة\n"
               "📩 أوامر جديدة: /status /coin XLM /scan /pause /help\n\n"
               f"👁️ راقب العملات: {', '.join(c.replace('USDT','') for c in COINS)}")
    errors = 0
    while True:
        try:
            poll_commands()
            manage_positions()
            if not STATE["paused"]:
                run_scan()
            # رسالة نبض كل 12 ساعة
            if time.time() - STATE.get("last_heartbeat", 0) > HEARTBEAT_HOURS * 3600:
                STATE["last_heartbeat"] = time.time()
                send_alert(build_status_message(analyze_btc()))
            save_state()
            errors = 0
            time.sleep(SCAN_INTERVAL)
        except KeyboardInterrupt:
            send_alert("👋 أُوقف البوت يدوياً. الذاكرة محفوظة وسأكمل من حيث توقفت.")
            break
        except Exception as e:
            errors += 1
            log(f"❌ خطأ #{errors}: {e}\n{traceback.format_exc()[-400:]}")
            if errors == 5:
                send_alert(f"⚠️ البوت يواجه أخطاء متكررة ({errors}) لكنه ما زال يحاول... تحقق من الإنترنت.")
            time.sleep(min(30 * errors, 120))

if __name__ == "__main__":
    main()
