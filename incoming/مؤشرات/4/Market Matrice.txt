method calcRiskTargets(TradeState this, float _close, float _atLine) =>
    float _risk = math.max(math.abs(_close - _atLine), syminfo.mintick * 10)
    if this.isLong
        this.entry := _close
        this.sl    := _atLine
        this.tp1   := _close + (_risk * 1.0)
        this.tp2   := _close + (_risk * 2.0)
        this.tp3   := _close + (_risk * 3.0)
    if this.isShort
        this.entry := _close
        this.sl    := _atLine
        this.tp1   := _close - (_risk * 1.0)
        this.tp2   := _close - (_risk * 2.0)
        this.tp3   := _close - (_risk * 3.0)

f_momBar(float score) =>
    int    n = math.min(math.round(score / 3.0 * 10.0), 10)
    string s = ""
    for i = 1 to 10
        s += i <= n ? "█" : "░"
    s

f_momColor(float score) =>
    score >= 2.5 ? i_momHi : score >= 1.5 ? i_momMd : score >= 0.8 ? i_momLo : i_momNe

f_bar(float val, float maxVal) =>
    int    n = math.round(math.min(val / maxVal, 1.0) * 10)
    string s = ""
    for i = 1 to 10

    s + "  " + str.tostring(math.round(val / maxVal * 100)) + "%"
