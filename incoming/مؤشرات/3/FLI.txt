
//@version=5
indicator(shorttitle='FLI', title='Follow Line Indicator', overlay=true)
// 
BBperiod = input.int(defval=21, title='BB Period', minval=1)
BBdeviations = input.float(defval=1.00, title='BB Deviations', minval=0.1, step=0.05)
UseATRfilter = input(defval=true, title='ATR Filter')
ATRperiod = input.int(defval=5, title='ATR Period', minval=1)
hl = input(defval=false, title='Hide Labels')
//
BBUpper = ta.sma(close, BBperiod) + ta.stdev(close, BBperiod) * BBdeviations
BBLower = ta.sma(close, BBperiod) - ta.stdev(close, BBperiod) * BBdeviations
//
TrendLine = 0.0
iTrend = 0.0
buy = 0.0
sell = 0.0
//
BBSignal = close > BBUpper ? 1 : close < BBLower ? -1 : 0
// 
if BBSignal == 1 and UseATRfilter == 1
    TrendLine := low - ta.atr(ATRperiod)
    if TrendLine < TrendLine[1]
        TrendLine := TrendLine[1]
        TrendLine
if BBSignal == -1 and UseATRfilter == 1
    TrendLine := high + ta.atr(ATRperiod)
    if TrendLine > TrendLine[1]
        TrendLine := TrendLine[1]
        TrendLine
if BBSignal == 0 and UseATRfilter == 1
    TrendLine := TrendLine[1]
    TrendLine
//
if BBSignal == 1 and UseATRfilter == 0
    TrendLine := low
    if TrendLine < TrendLine[1]
        TrendLine := TrendLine[1]
        TrendLine
if BBSignal == -1 and UseATRfilter == 0
    TrendLine := high
    if TrendLine > TrendLine[1]
        TrendLine := TrendLine[1]
        TrendLine
if BBSignal == 0 and UseATRfilter == 0
    TrendLine := TrendLine[1]
    TrendLine
//
iTrend := iTrend[1]
if TrendLine > TrendLine[1]
    iTrend := 1
    iTrend
if TrendLine < TrendLine[1]
    iTrend := -1
    iTrend
//
buy := iTrend[1] == -1 and iTrend == 1 ? 1 : na
sell := iTrend[1] == 1 and iTrend == -1 ? 1 : na
//
plot(TrendLine, color=iTrend > 0 ? color.blue : color.red, style=plot.style_line, linewidth=2, title='Trend Line', transp=0)
plotshape(buy == 1 and hl == false ? TrendLine - ta.atr(8) : na, text='💣', style=shape.labelup, location=location.absolute, color=color.new(color.blue, 0), textcolor=color.new(color.white, 0), offset=0, size=size.auto)
plotshape(sell == 1 and hl == false ? TrendLine + ta.atr(8) : na, text='🔨', style=shape.labeldown, location=location.absolute, color=color.new(color.red, 0), textcolor=color.new(color.white, 0), offset=0, size=size.auto)
//
alertcondition(sell == 1, title='Sell', message='Sell')
alertcondition(buy == 1, title='Buy', message='Buy')
alertcondition(buy == 1 or sell == 1, title='Buy/Sell', message='Buy/Sell')


bullishCandle = close >= open[1] and close[1] < open[1]  //and high >= high[1] and low <= low[1]
bearishCandle = close <= open[1] and close[1] > open[1]  //and high > high[1] and low < low[1]
// RSI integration
rsiSource = input(title='rsiSource', defval=close)
rsiLenghth = input(title='rsi length', defval=14)
rsiOverBought = input(title='rsi overbought level', defval=70)
rsiOverSold = input(title='rsi over sold level', defval=30)
//rsiOverBoughtThreshold=input(title="rsiOBThreshold level", type=input.integer, defval=97)
//rsiOverSoldThreshold=input(title="rsiOSThreshold level", type=input.integer, defval=18)
//get RSI value
rsiValue = ta.rsi(rsiSource, rsiLenghth)
isRSIOB = rsiValue >= rsiOverBought and rsiValue
isRSIOS = rsiValue <= rsiOverSold and rsiValue

tradeSignal = (isRSIOS or isRSIOS[1] or isRSIOS[2]) and bullishCandle or (isRSIOB or isRSIOB[1] or isRSIOB[2]) and bearishCandle
//plot on chart
plotshape(tradeSignal and bullishCandle, title='bullish', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='BUY')
plotshape(tradeSignal and bearishCandle, title='bearish', location=location.abovebar, color=color.new(color.red, 0), style=shape.triangledown, text='SELL')


Timeframe = input.timeframe('D', title ='Time frame')
width = input.int(defval = 1, title='Line width', minval = 1)
fillzones = input(defval = true, title='Clouds')
showlast = input(title = 'Show only last period', defval = false)

////
r1 = 0.5
r2 = 0.618
r3 = 1
r4 = 1.382
resextsup = 2.74
///
s1 = -0.5
s2 = -0.618
s3 = -1
s4 = -1.382
sopextinf = -2.74
///
pHigh = request.security(syminfo.tickerid, Timeframe, high[1], lookahead=barmerge.lookahead_on)
pLow = request.security(syminfo.tickerid, Timeframe, low[1], lookahead=barmerge.lookahead_on)
pClose = request.security(syminfo.tickerid, Timeframe, close[1], lookahead=barmerge.lookahead_on)

islast = showlast ? request.security(syminfo.tickerid, Timeframe, barstate.islast, lookahead=barmerge.lookahead_on) : true
///
pvp = (pHigh + pLow + pClose) / 3
cp = (pHigh + pLow) / 2
dp = pvp + pvp - cp
fibrange(pv, h, l, ratio) =>
    pv + (h - l) * ratio


vS1 = fibrange(pvp, pHigh, pLow, s1)
vS2 = fibrange(pvp, pHigh, pLow, s2)
vS3 = fibrange(pvp, pHigh, pLow, s3)
vS4 = fibrange(pvp, pHigh, pLow, s4)
vSEX = fibrange(pvp, pHigh, pLow, sopextinf)
///
vR1 = fibrange(pvp, pHigh, pLow, r1)
vR2 = fibrange(pvp, pHigh, pLow, r2)
vR3 = fibrange(pvp, pHigh, pLow, r3)
vR4 = fibrange(pvp, pHigh, pLow, r4)
vREX = fibrange(pvp, pHigh, pLow, resextsup)
///
pvplot = plot(islast and pvp > 0 ? ta.change(pvp) != 0 ? na : pvp : na, color=color.new(#fb00ff, 0), title='POC', style=plot.style_linebr, linewidth=width)
///
res1 = plot(islast and vR1 > 0 ? ta.change(vR1) != 0 ? na : vR1 : na, color=color.new(#368a43, 0), title='R1', style=plot.style_linebr, linewidth=width)
res2 = plot(islast and vR2 > 0 ? ta.change(vR2) != 0 ? na : vR2 : na, color=color.new(#368a43, 0), title='RB1', style=plot.style_linebr, linewidth=width)
res3 = plot(islast and vR3 > 0 ? ta.change(vR3) != 0 ? na : vR3 : na, color=color.new(#368a43, 0), title='R2', style=plot.style_linebr, linewidth=width)
res4 = plot(islast and vR4 > 0 ? ta.change(vR4) != 0 ? na : vR4 : na, color=color.new(#368a43, 0), title='RB2', style=plot.style_linebr, linewidth=width)
resex = plot(islast and vREX > 0 ? ta.change(vREX) != 0 ? na : vREX : na, color=color.new(#368a43, 0), title='R3', style=plot.style_cross, linewidth=width)

///
sup1 = plot(islast and vS1 > 0 ? ta.change(vS1) != 0 ? na : vS1 : na, color=color.new(#e63c3c, 0), title='S1', style=plot.style_linebr, linewidth=width)
sup2 = plot(islast and vS2 > 0 ? ta.change(vS2) != 0 ? na : vS2 : na, color=color.new(#e63c3c, 0), title='SB1', style=plot.style_linebr, linewidth=width)
sup3 = plot(islast and vS3 > 0 ? ta.change(vS3) != 0 ? na : vS3 : na, color=color.new(#e63c3c, 0), title='S2', style=plot.style_linebr, linewidth=width)
sup4 = plot(islast and vS4 > 0 ? ta.change(vS4) != 0 ? na : vS4 : na, color=color.new(#e63c3c, 0), title='SB2', style=plot.style_linebr, linewidth=width)
supex = plot(islast and vSEX > 0 ? ta.change(vSEX) != 0 ? na : vSEX : na, color=color.new(#e63c3c, 0), title='S3', style=plot.style_cross, linewidth=width)

///
cpplot = plot(islast and cp > 0 ? ta.change(cp) != 0 ? na : cp : na, color=#ffffff00, style=plot.style_linebr, linewidth=width)
dpplot = plot(islast and dp > 0 ? ta.change(dp) != 0 ? na : dp : na, color=#ffffff00, style=plot.style_linebr, linewidth=width)

// Candle calculator
body_siz= math.abs(open - close)

//  Hammer or ShootingStar detector

wick_slows   =math.abs(low - close),wick_upows   =math.abs(high - open)


//  Hammer or ShootingStar detector
hammer_candle  = ( wick_llower > body_siz * 2 and low <= ta.lowest(low, 10) and wick_llower > wick_hupeer * 2 and close <= open[1]) and (low<=cur_dl or low<=cur_dl[1])
invert_hammer  = (wick_upows > body_siz * 2 and high >= ta.highest(high, 10) and wick_upows > wick_slows * 2 and open <= close[1]) and (high>=cur_dh[1] or high>=cur_dh)

//chart plotters

plotchar(en_inh?invert_hammer:na,char = "🟠", location=location.abovebar, color=color.red, text='Inverted Hammer',textcolor = color.orange,size = size.tiny)
