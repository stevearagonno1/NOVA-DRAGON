// Helper to get the day number from timestamp
get_day(ts) =>
    year_val = year(ts)
    month_val = month(ts)
    day_ = dayofmonth(ts)
    year_val * 10000 + month_val * 100 + day_





manageSessionLines(lines_l, lines_startTime, lines_active, lines_day, isHighLine) =>
    curDay = get_day(time)
    if array.size(lines_l) > 0
        i = 0
        while i < array.size(lines_l)
            l = array.get(lines_l, i)
            active = array.get(lines_active, i)
            lineDay = array.get(lines_day, i)
            // Remove lines older than 2 days
            if curDay - lineDay > 1
                line.delete(l)
                array.remove(lines_l, i)
                array.remove(lines_startTime, i)
                array.remove(lines_active, i)
                array.remove(lines_day, i)
                continue
            if active
                startTime = array.get(lines_startTime, i)
                y_level = line.get_y1(l)
                isCrossed = isHighLine ? high > y_level : low < y_level
                isExpired = time > startTime + 2 * 86400000 // 2 days in milliseconds
                if isCrossed or isExpired
                    line.set_x2(l, bar_index - 1)
                    array.set(lines_active, i, false)
                else
                    line.set_x2(l, bar_index)
            i += 1

// Function to calculate VWAP for a session
calculateSessionVWAP(session_volumes, session_prices, session_high, session_low) =>
    float vwap_result = na
    if array.size(session_volumes) > 0
        price_range = session_high - session_low
        bin_size = price_range / vp_bins
        temp_volumes = array.new_float(vp_bins, 0.0)
        temp_prices = array.new_float(vp_bins, 0.0)
        
        for i = 0 to vp_bins - 1
            array.set(temp_prices, i, session_low + (i + 0.5) * bin_size)
        
        for i = 0 to array.size(session_volumes) - 1
            vol = array.get(session_volumes, i)
            price = array.get(session_prices, i)
            bin_index = math.floor((price - session_low) / bin_size)
            if bin_index >= 0 and bin_index < vp_bins
                current_vol = array.get(temp_volumes, bin_index)
                array.set(temp_volumes, bin_index, current_vol + vol)
        
        total_volume = 0.0
        volume_price_sum = 0.0
        for i = 0 to vp_bins - 1
            vol = array.get(temp_volumes, i)
            price = array.get(temp_prices, i)
            total_volume += vol
            volume_price_sum += vol * price
        
        vwap_result := total_volume > 0 ? volume_price_sum / total_volume : na
    vwap_result

calculateSessionValueAreaHigh(session_volumes, session_prices, session_high, session_low) =>
    float val_high_result = na
    if array.size(session_volumes) > 0
        price_range = session_high - session_low
        bin_size = price_range / vp_bins
        temp_volumes = array.new_float(vp_bins, 0.0)

        for i = 0 to array.size(session_volumes) - 1
            vol = array.get(session_volumes, i)
            price = array.get(session_prices, i)
            bin_index = math.floor((price - session_low) / bin_size)
            if bin_index >= 0 and bin_index < vp_bins
                current_vol = array.get(temp_volumes, bin_index)
                array.set(temp_volumes, bin_index, current_vol + vol)
        
        max_vol = 0.0

        for i = 0 to vp_bins - 1
            vol = array.get(temp_volumes, i)
            if vol > max_vol
                max_vol := vol
                poc_index := i
        
        total_vol = 0.0
        for i = 0 to vp_bins - 1
            total_vol += array.get(temp_volumes, i)
        
        target_vol = total_vol * (vp_value_area_pct / 100.0)
        current_vol = 0.0
        val_low_index = poc_index
        val_high_index = poc_index
        
        while current_vol < target_vol and (val_low_index > 0 or val_high_index < vp_bins - 1)
            low_vol = val_low_index > 0 ? array.get(temp_volumes, val_low_index - 1) : 0.0
            high_vol = val_high_index < vp_bins - 1 ? array.get(temp_volumes, val_high_index + 1) : 0.0
            
            if low_vol >= high_vol and val_low_index > 0
                current_vol += low_vol
                val_low_index -= 1
            else if val_high_index < vp_bins - 1
                current_vol += high_vol
                val_high_index += 1
            else
                break
        
        val_high_result := array.get(temp_prices, val_high_index)
    val_high_result

// Helper to format seconds as HH:MM:SS
format_time_left(sec) =>
    h = math.floor(sec / 3600)
    m = math.floor((sec % 3600) / 60)
    s = math.floor(sec % 60)
    str.tostring(h, "00") + ":" + str.tostring(m, "00") + ":" + str.tostring(s, "00")

// Helper to get session open/close times in seconds from now
// sessionStr is like "0930-1600"
session_time_left(sessionStr, tz) =>
    t_now = timenow
    s = str.split(sessionStr, "-")
    startStr = array.get(s, 0)
    endStr = array.get(s, 1)
    sh = int(str.tonumber(str.substring(startStr, 0, 2)))
    sm = int(str.tonumber(str.substring(startStr, 2, 4)))
    eh = int(str.tonumber(str.substring(endStr, 0, 2)))
    em = int(str.tonumber(str.substring(endStr, 2, 4)))
    y = year(t_now, tz)
    mo = month(t_now, tz)
    d = dayofmonth(t_now, tz)
    t_start = timestamp(tz, y, mo, d, sh, sm)
    t_end = timestamp(tz, y, mo, d, eh, em)
    // If session ends on the next day (crosses midnight)
    t_end := t_end <= t_start ? timestamp(tz, y, mo, d + 1, eh, em) : t_end
    // If session already ended, move to next day
    t_start := t_now > t_end ? timestamp(tz, y, mo, d + 1, sh, sm) : t_start
    t_end := t_now > t_end ? timestamp(tz, y, mo, d + 1, eh, em) : t_end
    open_in = t_now < t_start ? t_start - t_now : 0
    close_in = t_now >= t_start and t_now < t_end ? t_end - t_now : 0
    result = array.new_float(2, 0.0)
    array.set(result, 0, open_in)
    array.set(result, 1, close_in)
    result

// Function to detect FVG using proper logic
detectFVG() =>
    // Check if we have enough bars
    if bar_index > 20
        // Bullish FVG: High[2] < Low[0] and Low[1] <= High[2] and High[1] >= Low[0]
        bull_fvg = high[2] < low and low[1] <= high[2] and high[1] >= low
        fvg_len_bull = bull_fvg ? low - high[2] : 0.0
        
        // Bearish FVG: Low[2] > High[0] and Low[1] <= High[0] and Low[2] <= High[1]
        bear_fvg = low[2] > high and low[1] <= high and low[2] <= high[1]
        fvg_len_bear = bear_fvg ? low[2] - high : 0.0
        
        // Calculate ATR for size check
        atr_value = ta.atr(20)
        atr_check_bull = fvg_len_bull > atr_value * fvg_cleanup_distance / 10
        atr_check_bear = fvg_len_bear > atr_value * fvg_cleanup_distance / 10
        
        // Check if it's a new FVG (not already detected)
        isNewFVG = true
        if array.size(fvg_bar_indexes) > 0
            for i = 0 to array.size(fvg_bar_indexes) - 1
                if array.get(fvg_bar_indexes, i) == bar_index
                    isNewFVG := false
                    break
        
        // Create bullish FVG
        if bull_fvg and atr_check_bull and isNewFVG
            array.push(fvg_bullish, box.new(bar_index - 1, high[2], fvg_extend_right ? bar_index + 500 : bar_index + 1, low, border_width=0, bgcolor=color.new(fvg_bullish_color, fvg_opacity)))
            array.push(fvg_bar_indexes, bar_index)
            array.push(fvg_highs, high[2])
            array.push(fvg_lows, low)
            array.push(fvg_types, "BULL")
        
        // Create bearish FVG
        if bear_fvg and atr_check_bear and isNewFVG
            array.push(fvg_bearish, box.new(bar_index - 1, high, fvg_extend_right ? bar_index + 500 : bar_index + 1, low[2], border_width=0, bgcolor=color.new(fvg_bearish_color, fvg_opacity)))
            array.push(fvg_bar_indexes, bar_index)
            array.push(fvg_highs, high)
            array.push(fvg_lows, low[2])
            array.push(fvg_types, "BEAR")

// Function to cleanup old FVGs
cleanupFVGs() =>
    atr_value = ta.atr(14)
    cleanup_threshold = atr_value * fvg_cleanup_distance
    
    // Only proceed if arrays have elements
    if array.size(fvg_bar_indexes) > 0
        // Cleanup arrays in reverse order to avoid index issues
        for i = array.size(fvg_bar_indexes) - 1 to 0
            fvg_bar = array.get(fvg_bar_indexes, i)
            fvg_high = array.get(fvg_highs, i)
            fvg_low = array.get(fvg_lows, i)
            fvg_type = array.get(fvg_types, i)
            
            // Check if price has moved away from FVG
            should_remove = false
            
            if fvg_type == "BULL"
                // For bullish FVG, remove if price is below the gap
                if close < fvg_low - cleanup_threshold
                    should_remove := true
            else if fvg_type == "BEAR"
                // For bearish FVG, remove if price is above the gap
                if close > fvg_high + cleanup_threshold
                    should_remove := true
            
            // Remove FVG if it's too old or price has moved away
            if should_remove or (bar_index - fvg_bar) > fvg_lookback
                // Delete the box
                if fvg_type == "BULL" and array.size(fvg_bullish) > i
                    box.delete(array.get(fvg_bullish, i))
                    array.remove(fvg_bullish, i)
                else if fvg_type == "BEAR" and array.size(fvg_bearish) > i
                    box.delete(array.get(fvg_bearish, i))
                    array.remove(fvg_bearish, i)
                
                // Remove from other arrays
                array.remove(fvg_bar_indexes, i)
                array.remove(fvg_highs, i)
                array.remove(fvg_lows, i)
                array.remove(fvg_types, i)

// Function to limit number of FVGs displayed
limitFVGs() =>
    max_fvgs = fvg_count
    
    // Only proceed if arrays have elements
    if array.size(fvg_bar_indexes) > 0

            array.remove(fvg_bearish, 0)
            array.remove(fvg_bar_indexes, 0)
            array.remove(fvg_highs, 0)
            array.remove(fvg_lows, 0)
            array.remove(fvg_types, 0)

                    should_remove := true

            if should_remove
                // Delete the box
                if invfvg_type == "BULL" and array.size(invfvg_bullish) > i
                    box.delete(array.get(invfvg_bullish, i))
                    array.remove(invfvg_bullish, i)
                else if invfvg_type == "BEAR" and array.size(invfvg_bearish) > i
                    box.delete(array.get(invfvg_bearish, i))
                    array.remove(invfvg_bearish, i)
                
                // Remove from other arrays
                array.remove(invfvg_bar_indexes, i)
                array.remove(invfvg_highs, i)
                array.remove(invfvg_lows, i)
                array.remove(invfvg_types, i)

            array.remove(invfvg_highs, 0)

            array.remove(invfvg_highs, 0)
            array.remove(invfvg_lows, 0)
            array.remove(invfvg_types, 0)

// Function to format the current time based on UTC offset
f_format_time(utc_offset) =>
    timestamp = timenow + utc_offset * 3600 * 1000
    hour_value = na(hour(timestamp)) ? 0 : hour(timestamp)
    minute_value = na(minute(timestamp)) ? 0 : minute(timestamp)
    second_value = na(second(timestamp)) ? 0 : second(timestamp)
    am_pm = hour_value < 12 ? "AM" : "PM"
    hour_display = hour_value % 12
    hour_display := hour_display == 0 ? 12 : hour_display
    str.tostring(hour_display, "00") + ":" + str.tostring(minute_value, "00") + ":" + str.tostring(second_value, "00") + " " + am_pm

// Function to get the day name
f_get_day_name(utc_offset, use_short) =>
    timestamp = timenow + utc_offset * 3600 * 1000
    days = array.new_string(7)
    short_days = array.new_string(7)
    array.set(days, 0, "Sunday")
    array.set(days, 1, "Monday")
    array.set(days, 2, "Tuesday")
    array.set(days, 3, "Wednesday")
    array.set(days, 4, "Thursday")
    array.set(days, 5, "Friday")
    array.set(days, 6, "Saturday")
    array.set(short_days, 0, "Sun")
    array.set(short_days, 1, "Mon")
    array.set(short_days, 2, "Tue")
    array.set(short_days, 3, "Wed")
    array.set(short_days, 4, "Thu")
    array.set(short_days, 5, "Fri")
    array.set(short_days, 6, "Sat")
    use_short ? array.get(short_days, dayofweek(timestamp) - 1) : array.get(days, dayofweek(timestamp) - 1)

// Function to format the current date based on UTC offset and chosen format
f_format_date(utc_offset, format) =>
    timestamp = timenow + utc_offset * 3600 * 1000
    year_num = na(year(timestamp)) ? 0 : year(timestamp)
    month_num = na(month(timestamp)) ? 0 : month(timestamp)
    day_num = na(dayofmonth(timestamp)) ? 0 : dayofmonth(timestamp)
    formatted_date = ""
    if format == "MM/DD/YYYY"
        formatted_date := str.tostring(month_num, "00") + "/" + str.tostring(day_num, "00") + "/" + str.tostring(year_num, "0000")
    else if format == "DD/MM/YYYY"
        formatted_date := str.tostring(day_num, "00") + "/" + str.tostring(month_num, "00") + "/" + str.tostring(year_num, "0000")
    else if format == "YYYY/MM/DD"
        formatted_date := str.tostring(year_num, "0000") + "/" + str.tostring(month_num, "00") + "/" + str.tostring(day_num, "00")
    else if format == "DD-MM-YYYY"
        formatted_date := str.tostring(day_num, "00") + "-" + str.tostring(month_num, "00") + "-" + str.tostring(year_num, "0000")
    else if format == "YYYY-MM-DD"
        formatted_date := str.tostring(year_num, "0000") + "-" + str.tostring(month_num, "00") + "-" + str.tostring(day_num, "00")
    else if format == "MMM DD, YYYY"
        month_names = array.from("Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec")
        month_str = array.get(month_names, month_num - 1)
        formatted_date := month_str + " " + str.tostring(day_num, "00") + ", " + str.tostring(year_num, "0000")
    else
        formatted_date := str.tostring(month_num, "00") + "/" + str.tostring(day_num, "00") + "/" + str.tostring(year_num, "0000")
    formatted_date

// Function to format the timeframe
f_format_timeframe(tf) =>
    tf_str = ""
    if tf == "1"
        tf_str := "1m"
    else if tf == "5"
        tf_str := "5m"
    else if tf == "15"
        tf_str := "15m"
    else if tf == "30"
        tf_str := "30m"
    else if tf == "60"
        tf_str := "1HR"
    else if tf == "120"
        tf_str := "2HR"
    else if tf == "180"
        tf_str := "3HR"
    else if tf == "240"
        tf_str := "4HR"
    else
        tf_str := tf
    tf_str

// Function to create the watermark text based on the selected display style
f_create_watermark_text(showSymbol, showTimeframe, showTime, showDate, showDay, useShortDay, UTC, dateFormat, displayStyle) =>
    separator = ""
    lineSeparator = "\n"
    if displayStyle == "Style 1"
        separator := " | "
    else if displayStyle == "Style 2"
        separator := " - "
    else if displayStyle == "Style 3"
        separator := "\n"
    else if displayStyle == "Style 4" or displayStyle == "Style 5" or displayStyle == "Style 6"
        lineSeparator := "\n"
        separator := " | "

    txt = ""
    if displayStyle == "Style 4"
        if showDate
            txt := txt + f_format_date(UTC, dateFormat) + lineSeparator

    if displayStyle == "Style 5"
        if showSymbol
            txt := txt + syminfo.ticker + separator
        if showTimeframe
            txt := txt + f_format_timeframe(timeframe.period) + separator
        if showDate
            txt := txt + f_format_date(UTC, dateFormat)
        txt := txt + lineSeparator

        if showDay
            txt := txt + f_get_day_name(UTC, useShortDay) + separator
        if showTime
            txt := txt + f_format_time(UTC)
        txt := txt + lineSeparator
        txt := txt + wm_text
    else if displayStyle == "Style 6"
        if showSymbol
            txt := txt + syminfo.ticker + separator
        if showTimeframe
            txt := txt + f_format_timeframe(timeframe.period) + separator
        if showDay
            txt := txt + f_get_day_name(UTC, useShortDay)
        txt := txt + lineSeparator

        if showTime
            txt := txt + f_format_time(UTC) + separator
        if showDate
            txt := txt + f_format_date(UTC, dateFormat)
        txt := txt + lineSeparator
        txt := txt + wm_text
    else
        if showSymbol
            txt := txt + syminfo.ticker
            if showTimeframe
                txt := txt + separator + f_format_timeframe(timeframe.period)
            if displayStyle != "Style 4"
                txt := txt + separator
            else
                txt := txt + lineSeparator

        if showTimeframe and not showSymbol
            txt := txt + f_format_timeframe(timeframe.period) + separator

        if showTime
            txt := txt + f_format_time(UTC) + separator

        if showDay
            txt := txt + f_get_day_name(UTC, useShortDay)
            if displayStyle != "Style 4"
                txt := txt + separator
            else
                txt := txt + lineSeparator

        // Remove trailing separator in Style 4 if not followed by another input
        if displayStyle == "Style 4" and str.length(txt) > 0 and str.length(separator) > 0
            if str.substring(txt, str.length(txt) - str.length(separator), str.length(txt)) == separator
                txt := str.substring(txt, 0, str.length(txt) - str.length(separator))

        if displayStyle == "Style 4"
            txt := txt + lineSeparator

        txt := txt + wm_text

        // Remove the last separator for non-stacked styles
        if displayStyle != "Style 3" and displayStyle != "Style 4" and str.length(txt) > str.length(separator)
            txt := str.substring(txt, 0, str.length(txt) - str.length(separator))

    txt
