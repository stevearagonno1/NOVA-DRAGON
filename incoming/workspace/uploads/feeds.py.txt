"""Module 1 — dual-mode data ingestion engine.

Mode A (backtest): Parquet → fully vectorized feature precompute (cached to
disk) → chronological k-way merge across all 12 symbols, tick-by-tick replay
with zero lookahead (only causal indicators are precomputed).

Mode B (live): one multiplexed Binance LIVE WebSocket (kline_1m + aggTrade +
bookTicker) feeding an asyncio.Queue; a separate worker task drains it so
network I/O is never blocked by math.
"""
from __future__ import annotations

import asyncio
import gc
import json
import logging
import os
import heapq
from dataclasses import dataclass

import numpy as np
import pandas as pd
import pyarrow.parquet as pq
import websockets

from . import config as C
from .indicators import compute_matrix

log = logging.getLogger("nova.feeds")

_TIME_COLS = ("open_time", "timestamp", "time", "date", "openTime", "time_open")


# --------------------------------------------------------------- Mode A
def _read_parquet(path: str) -> pd.DataFrame:
    """Patch 5: Termux-safe ingestion — pyarrow MEMORY-MAPPED read of only the
    6 required columns (no full-file materialisation, no page-cache copy).
    to_pandas(self_destruct=True) hands buffers over without duplication."""
    pf = pq.ParquetFile(path, memory_map=True)
    names = pf.schema_arrow.names
    lower = {n.lower(): n for n in names}
    tkey = next((lower[c] for c in _TIME_COLS if c in lower), None)
    wanted = ["open", "high", "low", "close", "volume"]
    cols = ([tkey] if tkey else []) + [lower[w] for w in wanted if w in lower]
    table = pf.read(columns=cols or None)
    df = table.to_pandas(split_blocks=True, self_destruct=True)
    del table
    return df
def _standardize(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df.columns = [str(c).lower() for c in df.columns]
    tcol = next((c for c in _TIME_COLS if c in df.columns), None)
    if tcol is not None:
        t = pd.to_numeric(df[tcol], errors="coerce")
        v = int(t.dropna().iloc[0])
        unit = "ns" if v > 1e16 else "us" if v > 1e13 else "ms" if v > 1e10 else "s"
        df.index = pd.to_datetime(t, unit=unit, utc=True)
    if not isinstance(df.index, pd.DatetimeIndex):
        raise ValueError("no usable timestamp column/index in parquet file")
    out = df[["open", "high", "low", "close", "volume"]].astype(float).sort_index()
    out = out[~out.index.duplicated(keep="last")]
    return out


def _source_path(symbol: str) -> str | None:
    for cand in (symbol, *C.SYMBOL_ALIASES.get(symbol, ())):
        p = C.ARCHIVE_DIR / f"{cand}_1m.parquet"
        if p.exists():
            return str(p)
    return None


def load_matrix(symbol: str) -> "Matrix | None":
    """Load + precompute one symbol's causal feature matrix (disk-cached)."""
    src = _source_path(symbol)
    if src is None:
        log.warning("archive missing for %s (tried aliases %s)",
                    symbol, C.SYMBOL_ALIASES.get(symbol, ()))
        return None
    cache = C.ARCHIVE_DIR / f"{symbol}__matrix_v{C.MATRIX_CACHE_VER}.parquet"
    sliced = bool(C.START_DATE or C.END_DATE)
    if not sliced and cache.exists() and os.path.getmtime(cache) >= os.path.getmtime(src):
        mat = pd.read_parquet(cache)
        log.info("%s matrix cache hit (%d bars)", symbol, len(mat))
        return Matrix(symbol, mat)

    df = _standardize(_read_parquet(src))
    if C.START_DATE:
        df = df[df.index >= pd.Timestamp(C.START_DATE, tz="UTC")]
    if C.END_DATE:
        df = df[df.index <= pd.Timestamp(C.END_DATE, tz="UTC")]
    if len(df) < 260:
        log.warning("%s: only %d bars after slicing — skipped", symbol, len(df))
        return None
    mat = compute_matrix(df)
    del df
    gc.collect()                                      # return RAM to Android
    if not sliced:
        try:
            mat.to_parquet(cache)
        except Exception as exc:
            log.warning("matrix cache write failed (%s)", exc)
    log.info("%s precomputed %d bars", symbol, len(mat))
    return Matrix(symbol, mat)


@dataclass
class Matrix:
    symbol: str
    df: pd.DataFrame

    def __post_init__(self) -> None:
        self.cols: dict[str, np.ndarray] = {
            c: self.df[c].to_numpy() for c in self.df.columns}
        self.ts = (self.df.index.astype("int64").to_numpy() // 1_000_000)  # ms
        self.n = len(self.df)


def stream_events(mats: dict[str, Matrix]):
    """Chronological k-way merge across the universe — memory-flat streaming."""
    heap = [(m.ts[0], s, 0) for s, m in mats.items() if m.n]
    heapq.heapify(heap)
    while heap:
        ts, s, i = heapq.heappop(heap)
        yield ts, s, i
        j = i + 1
        m = mats[s]
        if j < m.n:
            heapq.heappush(heap, (m.ts[j], s, j))


# --------------------------------------------------------------- Mode B
class LiveFeed:
    """Persistent multiplexed Binance live stream → asyncio.Queue.

    Market data ALWAYS comes from the live endpoint (order flow accuracy);
    execution routing is decided elsewhere by BINANCE_ENV.
    """

    def __init__(self, queue: asyncio.Queue):
        self.queue = queue
        self._stop = False
        self.dropped = 0

    def stop(self) -> None:
        self._stop = True

    @staticmethod
    def url() -> str:
        streams = "/".join(f"{s.lower()}@{st}" for s in C.SYMBOLS for st in C.STREAMS)
        return f"{C.WS_BASE}/stream?streams={streams}"

    def _enqueue(self, ev: tuple) -> None:
        try:
            self.queue.put_nowait(ev)
        except asyncio.QueueFull:
            self.dropped += 1
            if self.dropped % 1000 == 1:
                log.warning("queue full — dropped %d events so far", self.dropped)

    def _dispatch(self, raw: str) -> None:
        msg = json.loads(raw)
        stream = msg.get("stream", "")
        data = msg.get("data")
        if data is None:
            return
        sym, kind = stream.split("@", 1)
        symbol = sym.upper()
        if kind == "kline_1m":
            k = data["k"]
            if k["x"]:                              # closed bar ONLY → no repaint
                self._enqueue(("bar", symbol, (
                    int(k["t"]), float(k["o"]), float(k["h"]),
                    float(k["l"]), float(k["c"]), float(k["v"]))))
        elif kind == "aggTrade":
            self._enqueue(("agg", symbol,
                           (float(data["p"]), float(data["q"]), bool(data["m"]))))
        elif kind == "bookTicker":
            self._enqueue(("book", symbol, (float(data["b"]), float(data["a"]),
                                            float(data["B"]), float(data["A"]))))

    async def run(self) -> None:
        backoff = 1
        url = self.url()
        while not self._stop:
            try:
                async with websockets.connect(
                        url, ping_interval=180, ping_timeout=30,
                        max_queue=4096, close_timeout=5) as ws:
                    log.info("multiplexed stream connected (%d streams)",
                             len(C.SYMBOLS) * len(C.STREAMS))
                    backoff = 1
                    async for raw in ws:
                        self._dispatch(raw)
            except asyncio.CancelledError:
                raise
            except Exception as exc:
                log.warning("WS dropped (%s) — reconnecting in %ds", exc, backoff)
                await asyncio.sleep(backoff)
                backoff = min(backoff * 2, 60)
