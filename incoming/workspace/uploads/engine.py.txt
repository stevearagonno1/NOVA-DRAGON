"""NOVA_V8 orchestrator — one analytical pipeline for BOTH modes.

 Backtest: feeds.load_matrices (vectorized precompute) → k-way chronological
           replay → per-bar pipeline → SimExecutor.
 Live    : LiveFeed queue → worker drains → identical per-bar pipeline (rolling
           window) + tick-level CVD/OFI → LiveExecutor.

Concurrency model: one asyncio.Lock per symbol guards evaluate→open (Module
state-management rule); the event loop stays in deep idle in live mode —
100 % queue/event driven.
"""
from __future__ import annotations

import asyncio
import logging
import math
import time
from collections import deque
from dataclasses import dataclass, field
from typing import Any, Optional

import numpy as np
import pandas as pd

from . import config as C
from . import feeds
from .execution import LiveExecutor, Position, SimExecutor
from .indicators import compute_matrix
from .microstructure import FVGRegistry
from .orderflow import NaiveBayesOrderFlow, OrderFlowEngine
from .regime import classify as classify_regime
from .risk import PositionSizer, SymbolGovernor
from .strategies import StrategyDispatcher, build_telemetry
from .telegram_bot import BOOT_MSG, TelegramBot
from .telemetry import TelemetryDB

log = logging.getLogger("nova.engine")

PROGRESS_STEPS = (10, 25, 50, 75, 90, 100)


@dataclass
class SymState:
    lock: asyncio.Lock = field(default_factory=asyncio.Lock)
    fvg: FVGRegistry = field(default_factory=FVGRegistry)
    ofe: OrderFlowEngine = field(default_factory=OrderFlowEngine)
    nb: NaiveBayesOrderFlow = field(default_factory=NaiveBayesOrderFlow)
    ofi: float = 0.0                                  # signed imbalance −1..+1
    bid_q: float = 0.0                                # instant book quantities
    ask_q: float = 0.0                                #   (wall-guard inputs)
    win_ts: deque = field(default_factory=lambda: deque(maxlen=C.LIVE_WINDOW))
    win_bars: deque = field(default_factory=lambda: deque(maxlen=C.LIVE_WINDOW))


def _f(x: Any) -> Optional[float]:
    try:
        v = float(x)
        return v if math.isfinite(v) else None
    except (TypeError, ValueError):
        return None


class NovaEngine:
    def __init__(self) -> None:
        self.mode = C.MODE
        self.db = TelemetryDB()
        self.bot = TelegramBot(self)
        self.dispatcher = StrategyDispatcher()
        self.governor = SymbolGovernor()
        self.states = {s: SymState() for s in C.SYMBOLS}
        self.executing = C.AUTOSTART
        self.sim = SimExecutor()
        self.sim.on_closed = self._on_trade_closed
        self.live = LiveExecutor()
        self.live.on_closed = self._on_trade_closed
        self._progress_done = 0
        self._progress_fired: set[int] = set()
        self._warm_ready = False                      # live execution gate (Patch 4)
        self._event_now = time.time()                 # backtest: event clock
        self._queue: asyncio.Queue = asyncio.Queue(maxsize=50_000)
        self._feed = feeds.LiveFeed(self._queue)

    # ============================================================ lifecycle
    async def run(self) -> None:
        await self.db.init()
        self.bot.start()
        await self.bot.send(BOOT_MSG)
        try:
            if self.mode == "live":
                await self._run_live()
            else:
                await self._run_backtest()
        finally:
            await self.bot.stop()
            await self.db.close()

    def set_executing(self, flag: bool) -> None:
        self.executing = flag
        log.info("executing → %s", flag)

    async def reset_all(self) -> None:
        self.executing = False
        self._progress_done = 0
        self._progress_fired.clear()
        self.governor.reset()
        await self.db.reset_session()

    # ======================================================== trade closing
    async def _on_trade_closed(self, symbol: str, net_pct: float, pnl_usd: float,
                               pos: Position, reason: str, px: float) -> None:
        await self.db.log_trade(
            ts=int(pos.entry_ts * 1000), mode=self.mode, regime=pos.regime,
            symbol_setup=f"{symbol} | {pos.strategy}", telemetry=pos.telemetry,
            exit_trigger=reason, net_outcome=round(net_pct, 4),
            pnl_usd=round(pnl_usd, 6))
        now = self._event_now if self.mode == "backtest" else time.time()
        if self.governor.record(symbol, pnl_usd, now):
            log.info("GOVERNOR: %s parked for %ds (3 consecutive losses)",
                     symbol, C.GOVERNOR_COOLDOWN_SEC)
        log.info("CLOSE %-9s %-26s %-32s net=%+.3f%% pnl=%+.4f$",
                 symbol, pos.strategy, reason, net_pct, pnl_usd)
        # ----- progress milestones (10/25/50/75/90/100 %) -----
        self._progress_done += 1
        pct = self._progress_done / max(C.SESSION_TARGET_TRADES, 1) * 100.0
        for step in PROGRESS_STEPS:
            if pct >= step and step not in self._progress_fired:
                self._progress_fired.add(step)
                await self.bot.milestone_pct(
                    float(step), self._progress_done, C.SESSION_TARGET_TRADES)

    # ==================================================== shared bar pipeline
    async def _maybe_trade(self, symbol: str, snap: dict, now: float) -> None:
        """asyncio.Lock guards simultaneous trade evaluations per ticker."""
        st = self.states[symbol]
        async with st.lock:
            if not self.executing or not self.governor.allowed(symbol, now):
                return
            if self.mode == "backtest":
                busy = symbol in self.sim.positions or symbol in self.sim.pending
            else:
                if not self._warm_ready:             # NB primed before any order
                    return
                busy = symbol in self.live.positions or (
                    symbol in self.live.tasks and not self.live.tasks[symbol].done())
            if busy:
                return

            feats = st.ofe.features()
            if feats is not None:
                nb_cls, nb_prob = st.nb.predict(feats)   # observe ONLY — no veto
            else:
                nb_cls, nb_prob = "Unclassified", 50.0

            sig = self.dispatcher.evaluate(snap)         # incl. adverse-wall veto
            if sig is None:
                if feats is not None:
                    st.nb.update(feats, st.ofe.self_label())
                return
            side, strategy, trigger_price = sig

            # ---- Whale-Wake Bonus: extreme aligned aggTrade print → boost ----
            whale = st.ofe.whale_recent(side)
            conf = min(100.0, nb_prob + (C.WHALE_BOOST if whale else 0.0))

            vr = snap.get("vr") or 0.0
            notional = PositionSizer.notional(conf, vr)    # confidence sizing
            if notional <= 0.0:
                if feats is not None:
                    st.nb.update(feats, st.ofe.self_label())
                return

            regime = classify_regime(
                adx=snap.get("adx14"), close=snap["close"], alma=snap.get("alma200"),
                hurst=snap.get("hurst"), atr=snap.get("atr14"),
                atr_sma20=snap.get("atr14_sma20"), atr_sma100=snap.get("atr14_sma100"))
            telemetry = build_telemetry(snap, nb_cls, nb_prob)
            telemetry["confidence_score"] = round(conf, 2)
            telemetry["whale_wake"] = {"aligned": bool(whale),
                                       "boost": C.WHALE_BOOST if whale else 0.0}
            atr_pct = (snap.get("atr14") or 0.0) / snap["close"] if snap["close"] else 0.0
            if atr_pct <= 0:
                return
            log.info("SIGNAL %-9s %-26s side=%s px=%s conf=%.1f%s vr=%.2f nb=%s",
                     symbol, strategy, side, f"{trigger_price:g}", conf,
                     "🐋" if whale else "", vr, nb_cls)

            if self.mode == "backtest":
                await self.sim.request_entry(
                    symbol=symbol, side=side, limit=float(trigger_price),
                    strategy=strategy, regime=regime, telemetry=telemetry,
                    atr_pct=atr_pct, notional=notional,
                    ts_ms=int(now * 1000), bar_idx=snap["bar_idx"])
            else:
                self.live.fire(symbol=symbol, side=side, strategy=strategy,
                               regime=regime, telemetry=telemetry,
                               atr_pct=atr_pct, notional=notional,
                               limit_hint=float(trigger_price))
            if feats is not None:
                st.nb.update(feats, st.ofe.self_label())

    # --------------------------------------------------------- snap builders
    @staticmethod
    def _snap_common(st: SymState, row: dict, prev: dict, prev2: dict,
                     bar_idx: int) -> Optional[dict]:
        close = row["close"]
        atr14 = row.get("atr14")
        if not close or not atr14:
            return None
        bull_ce = st.fvg.entry(1, close)
        bear_ce = st.fvg.entry(-1, close)
        vsma = row.get("vol_sma20")
        return {
            "bar_idx": bar_idx, "close": close, "open": row["open"],
            "high": row["high"], "low": row["low"], "px": close,
            "atr14": atr14, "atr53": row.get("atr53"),
            "atr14_sma20": row.get("atr14_sma20"),
            "atr14_sma100": row.get("atr14_sma100"), "vr": row.get("vr"),
            "alma200": row.get("alma200"), "adx14": row.get("adx14"),
            "hurst": row.get("hurst"), "kama_slope": row.get("kama_slope"),
            "hyst_m": row.get("hyst_m"), "hyst_side": row.get("hyst_side"),
            "bar_delta": row.get("bar_delta"),
            "rsi_0": row.get("rsi7"), "rsi_1": prev.get("rsi7"),
            "rsi_2": prev2.get("rsi7"), "rsi_zone": row.get("rsi_zone"),
            "macd_hist_0": row.get("macd_hist"), "macd_hist_1": prev.get("macd_hist"),
            "macd_hist_2": prev2.get("macd_hist"),
            "vol_burst": bool(vsma and row.get("volume") and
                              row["volume"] > 1.5 * vsma),
            "sfp": int(row.get("sfp") or 0), "mss": int(row.get("mss") or 0),
            "fvg_entry": {"bull": bull_ce, "bear": bear_ce},
            "ofi": st.ofi,
            "bid_q": st.bid_q, "ask_q": st.ask_q,      # wall-guard (live; 0 in replay)
        }

    # ======================================================== MODE A: replay
    async def _run_backtest(self) -> None:
        log.info("precomputing matrices for %d symbols…", len(C.SYMBOLS))
        mats_list = await asyncio.to_thread(
            lambda: {s: m for s in C.SYMBOLS if (m := feeds.load_matrix(s))})
        if not mats_list:
            log.error("no archives found in %s — nothing to do", C.ARCHIVE_DIR)
            return
        total = sum(m.n for m in mats_list.values())
        log.info("replaying %s bar-events chronologically…", f"{total:,}")

        def row_at(m: feeds.Matrix, i: int) -> dict:
            out = {}
            for k, arr in m.cols.items():
                v = arr[i]
                out[k] = None if (isinstance(v, float) and not math.isfinite(v)) else v
            return out

        sim, states, governor = self.sim, self.states, self.governor
        done = 0
        try:
            for ts_ms, symbol, i in feeds.stream_events(mats_list):
                m = mats_list[symbol]
                cols = m.cols
                st = states[symbol]
                o, h = float(cols["open"][i]), float(cols["high"][i])
                l, c = float(cols["low"][i]), float(cols["close"][i])
                now = ts_ms / 1000.0
                self._event_now = now

                # ---- FAST PATH (every event, ~10 µs): fills/exits + state ----
                await sim.on_bar(symbol, o, h, l, ts_ms, i)
                st.fvg.on_bar(h, l, ts_ms, cols["fvg_bull_top"][i], cols["fvg_bull_bot"][i],
                              cols["fvg_bear_top"][i], cols["fvg_bear_bot"][i])
                st.ofi = cols["clv"][i] if cols["clv"][i] == cols["clv"][i] else 0.0
                st.ofe.on_bar_close(c, cols["bar_delta"][i])

                # ---- SLOW PATH only on pre-vectorized strategy candidates ----
                if (i >= 2 and cols["cand"][i] and self.executing
                        and symbol not in sim.positions and symbol not in sim.pending
                        and governor.allowed(symbol, now)):
                    snap = self._snap_common(st, row_at(m, i), row_at(m, i - 1),
                                             row_at(m, i - 2), i)
                    if snap:
                        await self._maybe_trade(symbol, snap, now)
                done += 1
                if done % 4096 == 0:                 # keep the loop responsive
                    await asyncio.sleep(0)
        finally:
            log.info("backtest finished: %d closed trades", self._progress_done)
            stats = await self.db.session_stats()
            await self.bot.send(
                f"✅ اكتملت المحاكاة الاستباقية ({done:,} حدث)\n"
                f"الصفقات المغلقة: {stats['trades']} | نسبة النجاح: {stats['win_rate']}%"
            )
            await self.bot.milestone_pct(100.0, self._progress_done,
                                         C.SESSION_TARGET_TRADES)
            await self.bot.stop()
            await self.db.close()

    # ========================================================= MODE B: live
    async def _run_live(self) -> None:
        if not (C.API_KEY and C.API_SECRET):
            log.error("BINANCE_API_KEY / BINANCE_API_SECRET required in live mode")
            return
        await self.live.init()
        log.info("execution routing: %s (%s)", C.BINANCE_ENV.upper(), C.REST_BASE)
        await self._warm_up()                         # Patch 4: prime before live
        feed_task = asyncio.create_task(self._feed.run(), name="live-feed")
        worker = asyncio.create_task(self._ws_worker(), name="queue-worker")
        try:
            await asyncio.gather(feed_task, worker)   # deep idle; event-driven
        finally:
            self._feed.stop()
            await self.live.close()

    # ------------------------------------------------- Patch 4: NB warm-up
    async def _warm_up(self) -> None:
        """Cold-start priming BEFORE any execution is allowed.

        Per symbol (public live-data REST, no signing):
          1) last 1000 trades → replayed through on_agg_trade (primes CVD,
             quote-volume Welford stats, whale detector);
          2) last 500 × 1m klines → seeded into the rolling window AND into
             the OrderFlowEngine, each bar running features()+self-label so
             the Naive-Bayes sums arrive pre-filled — not an empty memory.
        Failure of any symbol degrades to a log line, never a crash."""
        PUBLIC = "https://api.binance.com"
        log.info("warm-up: priming %d symbols from last 1000 trades + 500 bars…",
                 len(C.SYMBOLS))
        sem = asyncio.Semaphore(3)

        async def one(symbol: str) -> int:
            st = self.states[symbol]
            updates = 0
            try:
                async with sem, self.live.session.get(
                        f"{PUBLIC}/api/v3/trades",
                        params={"symbol": symbol, "limit": 1000}) as r:
                    trades = await r.json(content_type=None)
                if isinstance(trades, list):
                    for t in trades:
                        st.ofe.on_agg_trade(float(t["price"]), float(t["qty"]),
                                            bool(t["isBuyerMaker"]))
                st.ofe.bar_delta = 0.0               # don't dump bulk into bar 1

                async with sem, self.live.session.get(
                        f"{PUBLIC}/api/v3/klines",
                        params={"symbol": symbol, "interval": "1m",
                                "limit": 500}) as r:
                    kl = await r.json(content_type=None)
                if isinstance(kl, list):
                    for k in kl:
                        ts = int(k[0])
                        o, h, l = float(k[1]), float(k[2]), float(k[3])
                        c, v = float(k[4]), float(k[5])
                        st.win_ts.append(ts)
                        st.win_bars.append((o, h, l, c, v))
                        rng = h - l
                        proxy = v * (c - o) / rng if rng > 0 else 0.0
                        st.ofe.on_bar_close(c, proxy)
                        feats = st.ofe.features()
                        if feats is not None:        # prime NB sums per bar
                            st.nb.update(feats, st.ofe.self_label())
                            updates += 1
            except Exception as exc:                  #  noqa: BLE001
                log.warning("warm-up %s failed: %s", symbol, exc)
            return updates

        results = await asyncio.gather(*(one(s) for s in C.SYMBOLS))
        self._warm_ready = True
        log.info("warm-up complete — NB primed (%s updates/symbol); execution gate OPEN",
                 results)

    async def _ws_worker(self) -> None:
        while True:
            ev = await self._queue.get()              # decoupled worker
            kind, symbol = ev[0], ev[1]
            st = self.states.get(symbol)
            if st is None:
                continue
            if kind == "agg":
                p, q, maker = ev[2]
                st.ofe.on_agg_trade(p, q, maker)      # true tick CVD delta
            elif kind == "book":
                bid, ask, bidq, askq = ev[2]
                tot = bidq + askq
                st.ofi = (bidq - askq) / tot if tot > 0 else 0.0
                st.bid_q, st.ask_q = bidq, askq
                self.live.on_book(symbol, bid, ask, bidq, askq)  # exit engine + micro guard
            elif kind == "bar":
                await self._on_live_bar(symbol, ev[2], st)

    async def _on_live_bar(self, symbol: str, bar: tuple, st: SymState) -> None:
        ts, o, h, l, c, v = bar
        st.win_ts.append(ts)
        st.win_bars.append((o, h, l, c, v))
        st.ofe.on_bar_close(c)                        # live: delta from ticks
        if len(st.win_bars) < 250:                    # ALMA(200)/hyst(200) warm-up
            return
        arr = np.asarray(st.win_bars, dtype=float)
        df = pd.DataFrame(arr, columns=["open", "high", "low", "close", "volume"],
                          index=pd.to_datetime(np.asarray(st.win_ts), unit="ms", utc=True))
        mat = await asyncio.to_thread(compute_matrix, df)
        i = len(mat) - 1
        last = mat.iloc[i]
        st.fvg.on_bar(h, l, ts, _f(last.fvg_bull_top) or np.nan,
                      _f(last.fvg_bull_bot) or np.nan,
                      _f(last.fvg_bear_top) or np.nan, _f(last.fvg_bear_bot) or np.nan)

        def row_at(j: int) -> dict:
            r = mat.iloc[j]
            return {k: (None if isinstance(r[k], float) and not math.isfinite(r[k])
                        else r[k]) for k in mat.columns}

        snap = self._snap_common(st, row_at(i), row_at(i - 1), row_at(i - 2), i)
        if snap:
            await self._maybe_trade(symbol, snap, time.time())
