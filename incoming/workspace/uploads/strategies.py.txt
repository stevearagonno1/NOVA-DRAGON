"""Module 6 — Strategy Tournament Engine (multi-trigger dispatcher).

Any strategy may fire; the first valid trigger wins the dispatch tuple:
    (Action, Strategy_Name, Trigger_Price)
Regime and Naive-Bayes are observers (zero-blocking) — they only tag.
"""
from __future__ import annotations

from typing import Any, Optional

from . import config as C

BUY, SELL, HOLD = 1, -1, 0


class StrategyDispatcher:
    name_a = "Monotonic_MSS_FVG"
    name_b = "Liquidity_Sweep_Reversal"
    name_c = "Micro_Burst_Momentum"

    def evaluate(self, snap: dict[str, Any]) -> Optional[tuple[int, str, float]]:
        sig = self._strategy_a(snap)
        if sig is None:
            sig = self._strategy_b(snap)
        if sig is None:
            sig = self._strategy_c(snap)
        if sig is None:
            return None
        if self._adverse_wall(sig[0], snap):     # hard microstructure veto
            return None                          # → HOLD (no order emitted)
        return sig

    # ---------------- Adverse Wall Guard (order-book veto, live mode only)
    @staticmethod
    def _adverse_wall(side: int, s: dict[str, Any]) -> bool:
        """Do not fire a BUY straight into a massive sell wall (and mirror):
        if the opposing book side carries > WALL_RATIO× our side, the wall
        crushes the micro-move → HOLD. In backtest no book quantities exist
        (bid_q/ask_q are falsy) → guard is inert."""
        bq, aq = s.get("bid_q") or 0.0, s.get("ask_q") or 0.0
        if bq <= 0.0 or aq <= 0.0:
            return False
        if side == BUY:
            return aq > C.WALL_RATIO * bq
        return bq > C.WALL_RATIO * aq

    # ----------------------------- A: Monotonic MSS + Shrink-on-Fill FVG
    def _strategy_a(self, s):
        mss = s.get("mss", 0)
        if mss == 0:
            return None
        ce = s.get("fvg_entry", {}).get("bull" if mss == 1 else "bear")
        if ce is None:
            return None
        return (BUY if mss == 1 else SELL, self.name_a, ce)

    # ------------- B: 1m Liquidity Sweep + heavy tick-level OFI on reversal
    def _strategy_b(self, s):
        sfp = s.get("sfp", 0)
        if sfp == 0:
            return None
        ofi = s.get("ofi", 0.0)              # signed imbalance −1..+1 (see engine)
        if sfp == 1 and ofi >= 0.30:         # bear-trap sweep → reversal long
            return (BUY, self.name_b, s["px"])
        if sfp == -1 and ofi <= -0.30:       # bull-trap sweep → reversal short
            return (SELL, self.name_b, s["px"])
        return None

    # --------------- C: 1m MACD-hist rising + RSI(7) curving + volume burst
    def _strategy_c(self, s):
        h2, h1, h0 = s.get("macd_hist_2"), s.get("macd_hist_1"), s.get("macd_hist_0")
        r2, r1, r0 = s.get("rsi_2"), s.get("rsi_1"), s.get("rsi_0")
        vol_ok = s.get("vol_burst", False)
        if None in (h2, h1, h0, r2, r1, r0) or not vol_ok:
            return None
        rising = h0 > h1 > h2
        falling = h0 < h1 < h2
        curve_up = (r0 - r1) > (r1 - r2) > 0
        curve_dn = (r0 - r1) < (r1 - r2) < 0
        if rising and curve_up:
            return (BUY, self.name_c, s["px"])
        if falling and curve_dn:
            return (SELL, self.name_c, s["px"])
        return None


def build_telemetry(snap: dict[str, Any], nb_class: str, nb_prob: float) -> dict:
    """Indicators_Matrix JSON: exact value AND direction prediction per indicator."""
    def pred_buy_sell(cond):
        return "BUY" if cond else "SELL"

    return {
        "kama_slope": {"value": snap.get("kama_slope"),
                       "prediction": pred_buy_sell((snap.get("kama_slope") or 0) > 0)},
        "alma200_vs_close": {"value": snap.get("alma200"),
                             "prediction": pred_buy_sell(snap.get("close", 0) > (snap.get("alma200") or 0))},
        "rsi7": {"value": snap.get("rsi_0"), "zone": snap.get("rsi_zone"),
                 "prediction": "BUY" if snap.get("rsi_zone") == "BULL_ZONE" else
                               "SELL" if snap.get("rsi_zone") == "BEAR_ZONE" else "HOLD"},
        "adx14": {"value": snap.get("adx14"),
                  "prediction": "HOLD" if (snap.get("adx14") or 0) < 20 else "TREND"},
        "hurst": {"value": snap.get("hurst"),
                  "prediction": "TREND" if (snap.get("hurst") or 0) > 0.55 else
                                "MEAN_REVERT" if (snap.get("hurst") or 1) <= 0.45 else "HOLD"},
        "hysteresis": {"mid": snap.get("hyst_m"),
                       "prediction": pred_buy_sell((snap.get("hyst_side") or 0) > 0)},
        "cvd_delta_bar": {"value": snap.get("bar_delta"),
                          "prediction": pred_buy_sell((snap.get("bar_delta") or 0) > 0)},
        "naive_bayes": {"class": nb_class, "probability_pct": round(nb_prob, 2),
                        "prediction": "BUY" if nb_class == "Bull" else
                                      "SELL" if nb_class == "Bear" else "HOLD"},
        "volatility_ratio": snap.get("vr"),
        "note": "Zero-Threshold: scores recorded, never used to veto direction.",
    }
